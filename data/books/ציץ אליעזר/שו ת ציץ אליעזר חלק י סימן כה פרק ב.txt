 שו"ת ציץ אליעזר חלק י סימן כה פרק ב 

פרק ב גר שמל ולא טבל והוא חושיב"ס =חולה שיש בו סכנה= אי מותר לחלל עליו את השבת. יש לברר אם מותר מעיקרא דדינא לחלל שבת במלאכה דאורייתא עבור גר שמל ועדיין לא טבל כשהוא חושיב"ס אי שנסתכן גלל המילה ואי שנסתכן גלל סיבה אחרת, וזה מצוי לקרות בזמן שמתרבים הגרים דהרי הדין הוא דלאחר המילה משהין אותו עד שיתרפא רפואה שלימה ואחר כך מטבילין אותו, כי מיא מרזו מכה, כדאיתא בגמ' יבמות ד' מ"ז ע"ב ונפסק להלכה ברמב"ם פי"ד מאיסו"ב ה"ה ובטויו"ד סי' רס"ח סעיף ב'. וכך נוהגים שבגדול מחכים כמה שבועות מהמילה עד הטבילה. (א) ולכאורה מסתברא דאין מחללין עליו את השבת בהיות שפסקינן דלעולם אינו גר עד שימול ויטבול וכמה דלא טבל נכרי הוא, כדאיתא בברכות ד' מ"ז ע"ב, וביבמות ד' מ"ז, ונפסק כך להלכה ברמב"ם פי"ג מה' איסו"ב ה"ו ובטויו"ד סי' רס"ח סעי' ב', וא"כ יש להחיל עליו מעיקרא דדינא את הנפסק בזה באו"ח סי' ש"ל סעי' ב' שהוא אפי' בנכרי שאינו עע"ז כיעו"ש. (ב) אבל לאחר העיון נראה דמותר לחלל עליו את השבת, וחילא דידי ממה דמצינו בשו"ת תשב"ץ ח"א סי' כ"א דמדי דברו בדינא דאין מפליגין בספינה פחות מג' ימים קודם השבת, כותב בתוך דבריו וז"ל: ומכאן אני אוסר למול את הגר בחמישי בשבת כדי שלא יבא יום ג' למילה בשבת ויצטרכו לחלל עליו שבת כדאיתא במס' שבת פר"ע (פ"ו ע"א). וה"ה לתינוק שחלה ונתרפא בחמישי בשבת והיה אפשר למול חד בשבת שממתינין לו עד למחר כיון שלא יכלו להמולו בזמנו עכ"ל, הרי שהיה פשוט להתשב"ץ דגר שמלוהו בחמישי בשבת מחללין עליו את השבת אע"פ שבודאי עוד לא הספיקו גם לטובלו עד אז כי במשך יומיים שלשה בודאי עוד לא נתרפא ממכת - המילה ובפרט בגדול, ואילו הוה ס"ל להתשב"ץ שהדין הזה הוא דוקא כשגם כבר טבל אזי בלי שום ספק שלא היה נמנע מלבאר זאת כדי שלא לתת מכשול לרבים ללמוד מסתימת דבריו לחלל את השבת עבור גר שמל ולא טבל במקרה ומסתכן. וזה גם לא מסתבר כלל לומר דכוונת התשב"ץ היא דיכנסו למצב שיצטרכו לחלל עליו שבת ולא יוכלו בגלל שלא טבל עוד. דאין זה במשמעות דבריו כלל, וגם דומיא דתינוק שחלה קתני שיצטרכו ויחללו. ועוד דאילו כן אין לזה שייכות לדינא דאין מפליגין שמשם הוא שלומד זאת, דבשם הרי המדובר שיצטרכו לחלל ויחללו, ופשוט. ודברי תשב"ץ אלה הובאו בב"י בטור יו"ד סי' רס"ח בב"ה ובב"ח בקו"א לסי' רס"ו ובש"ך בסי' רס"ו ס"ק י"ח. ולא העירו כלום על עיקרו של דבר זה אם בכלל מחללין עליו. וכמו"כ כל הפוסקים המיימינים והמשמאילים שדנים בדברי תשב"ץ דנין רק על הפרט אם מלין אותו ביום חמישי ולא חיישינן למה שיצטרכו לחלל, או שאין מלין, אבל בעצם הדין הזה שיוצא מהתשב"ץ שבמקרה ויסתכן שמחללין עליו את השבת, לא נגע אף אחד לפקפק בזה, ומשמע איפוא שבזה כו"ע מודים שכן מחללין עליו ואע"פ שעדיין לא טבל מכיון שמיהת כבר מל א"ע לשם יהודות. ולכן לא מצאו לנכון לדבר כלל על ההנחה הזאת שהניח לפנינו התשב"ץ ז"ל כלאחר יד. (ג) וצריכים רק להבין אבל טעמו של דבר מדוע שמחללין עליו, הא הרי כנכרי נחשב עוד (כלשון הגמ' בברכות) כל עוד שגם לא טבל. ונראה להסביר זאת עפ"י מה שראיתי בשו"ת בנין ציון סימן צ"א שהשיב בעובדא שהיה בירושלים בשנת תר"ח שגדול אחד כפה לגר שמל ולא טבל לעשות מלאכה ביום ש"ק, ולעזה עליו כל העיר, דאותו גדול לא יפה הורה ומעולם לא הקפידו ע"ז בשום מקום שלא ישמור הגר שבת קודם הטבילה, וכתב להסביר זאת, מפני, דאחרי שמילת הגר נקרא ברית שמברכים עלי' כורת הברית כדאמרי' שבת קל"ז וגם שבת נקרא ברית כדאמרינן שם ד' קל"ב, איך נאמר דאחר שנכנס לברית האחת יהי' מוכרח להפר ברית האחרת שכרת הקב"ה עם ישראל מקיימי מצוותיו, ואעפ"י שעדיין לא נכנס לכלל ישראל גמור עד שטבל, מ"מ משעה שנכנס לברית מילה כבר נבדל מכלל ב"נ =בן נח=, וכעין שכתבו התוס' בכריתות ד' ט' וכו', וע"כ אין עליו עוד מצות יום ולילה ל"י =לא ישבותו= של ב"נ, ולכן לא לבד שמותר לגר כזה לקיים שבת אלא אפשר לצדד ג"כ שחובה עליו לקיים וכו', ובשבת ד' פ"ו אמרינן דעל שבת נצטוו ישראל במרה, וטבילה לא היה עד סיני, ע"כ קיימו ישראל שבת כשמלו ולא טבלו וכו' עיי"ש. ואם נקבל את ההנחה הזאת דגר שמל ולא טבל אע"פ שעדיין לא נכנס לכלל ישראל אבל לענין שבת שאני דהוקש ברית לברית, ולא רק שיכול כבר לשמור את השבת אלא שיתכן שכבר מחויב בכך, א"כ נוכל להוסיף ולומר על הדרך הזאת גם סברא שמותר ממילא גם לחלל עליו את השבת, והוא עפ"י מה דמצינו בגמ' בע"ז ד' כ"ו ע"א דאיתא: סבר רב יוסף למימר אולודי עובדת כוכבים בשבתא בשכר שרי משום איבה, א"ל אביי יכלה למימר לה דידן דמינטרי שבתא מחללינן עלייהו דידכו דלא מינטרי שבת לא מחללינן. וא"כ אפשר לקחת לנו מזה אמת - מדה כדי להגדיר על מי שמחללין שבת ולומר דכל מי דמינטרי שבתא מחללין עליה את השבת, ואמרינן עליה חלל עליו שבת אחת כדי שישמור שבתות הרבה, שכידוע אע"פ שאנן תפסינן לעיקר הלימוד של שמואל ביומא ד' פ"ה ע"ב שלומד מקרא וחי בהם ולא שימות בהם, הוא זה כדי ללמוד שמחללין גם על ספק פקו"נ =פקוח נפש=, אבל להרבה ענינים משתמשים גם בלימוד של ר"ש בן מנסיא ביומא שם שלומד מקרא דושמרו בני ישראל את השבת, אמרה תורה חלל עליו שבת אחת כדי שישמור שבתות הרבה שגם רשב"ג בשבת ד' קנ"א ע"ב דורש ג"כ דרשה זאת. (ויעוין מ"ש בזה בספרי שו"ת צ"א ח"ח סי' ט"ו פ"ג עיי"ש). (ד) ולדברי הבנין ציון הנ"ל כדאי להוסיף כי בויכוח ההוא בשנה ההיא הוסיף גם המהר"י שווארץ ז"ל בספרו שו"ת דברי יוסף ח"ג סי' כ"ד להביא מקור לכך כי שמירת שבת תלוי במצות מילה מדברי המד"ר ריש דברים דאיתא: גוי ששמר את השבת עד שלא קיבל עליו את המילה חייב מיתה וכו' לפיכך כל גוי שבא ומכניס עצמו ביניהם (בין השבת בין ישראל ובין הקב"ה) עד שלא קיבל עליו לימול חייב מיתה ע"ש, הרי דהמדרש תלה ענין שמירת שבת בעיקר באם קיבל עליו את המילה. וכאשר נשנה מעשה כזאת וויכוח כזה בירושלים בשנת תרמ"א הביא בס' שמח נפש אות גרים סייעתא לדעתו שאין לצוות את הגר שמל ולא טבל לחלל את השבת מדברי הדברי יוסף הנ"ז ע"ש [וכשלעצמי חשבתי לסתור הרא' מהמדרש די"ל דכוונתו כשמקבל עליו המילה ולאחר מיכן גם הטבילה, וכך מצאתי אח"כ בספר חמדת ישראל ח"א בדיני ב"נ אות נ"ח שכותב ג"כ בכזאת שהכוונה שקיבל עליו להתגייר ע"ש. ולפלא שבספרו כלי חמדה בקו"א אות כ"ב העתיק שכאילו בספרו שם הוכיח מהמדרש שתלוי רק במילה כיעו"ש, כן יש לדון מהא דאין למדין הלכה ממדרש, וכתב מזה בשומר ציון הנאמן תמוז תרי"ג עיי"ש]. ויש להוסיף הסבר נוסף בזה בהלכה עפ"י מה שראיתי בירחון המאסף שנה עשירית חוברת ד' שמובא שם בשם רבה של ירושלים הגר"ש סלנט ז"ל שג"כ הורה בעובדא הנ"ל שאסור לו לחלל שבת, והביא ראי' לזה מדברי הרשב"א בשבת ד' ע"א, דמקשה שם על דברי הגמ' שאו' אלא לאתויי גר שמל ולא טבל, דא"כ היינו עכו"ם ואע"פ שמל הרי הוא כערל דהו"ל כערבי מהול, ומתרץ דשאני הכא דמילתו לשם יהדות ואע"פ שלא נגמר גירותו מ"מ כבר התחיל ונכנס קצת בדת יהודית שאינו צריך אלא טבילה ע"ש. ואחרי שכבר התחילה יהדותו איך יחלל יהדותו עיי"ש. ואחרי הגלות נגלות דברי הרשב"א דמאז שמל עצמו לשם יהדות כבר התחיל ונכנס קצת בדת יהודית, שוב י"ל דמכיון שכן כבר כלול הנהו במצות הקרא של וחי בהם דדרשינן ולא שימות בהם, ומעניין הדבר שהקרא של וחי בהם (בויקרא י"ח - ה) כתוב בלשון אשר יעשה אותם האדם וחי בהם, וכבר ראיתי מקשים דהרי לישנא דהאדם כולל גם עכו"ם כדאיתא תוס' יבמות ד' ס"א ע"א ד"ה ואין. אבל לפי הנ"ז יתיישב שפיר, שכלול בקרא גם גר שמל ולא טבל אע"פ שליתר דיני התורה דינו עוד כעכו"ם שנקרא בשם האדם, [כן ראיתי גם בשו"ת אבני נזר חיו"ד ח"ב סי' שנ"א שכותב נמי לחדש בכנ"ל דאף דאינו גר עד שימל ויטבול מ"מ הואיל שבמרה נצטוו במצוות שבת וזה הי' קודם טבילה שהיה בשעת מ"ת =מתן תורה=, ע"כ לדורות נמי תיכף לאחר מילה מצווה על השבת, ומוכיח כן גם מדברי הזוה"ק, אלא שלבסוף לא הכריע את ההלכה בזה וכותב שלדינא אם נאסר גר שמל ולא טבל במלאכת שבת צע"ג עיי"ש. אבל כפי שראיתי הרי יתר הגדולים הנ"ל כן נטו להכריע ההלכה בזה]. (ה) הגדרה שלישית אפשר לומר בזה בעפ"י מה שמצינו בשו"ת רדב"ז סי' תתקי"ז (תע"ט) שנשאל אודות גר שמל ולא טבל ונגע ביין אם אסרו, וכן אם מזמנין עליו או מצטרף לעשרה, ואם קדש אשה אם חוששין לקדושיו, והשיב ובירר דאפילו למאן דאית ליה גר תושב אוסר היין במגעו מודה הוא בגר שמל ולא טבל שאינו אוסר, ולענין להצטרף לכל דבר שבקדושה ולענין אי חיישינן לקדושיו הדבר ברור שאינו כישראל לשום דבר דקיי"ל מל ולא טבל כאילו לא מל, אלא שיצא מכלל עכו"ם שמצוה החיותו ואין מגעו ביין טמא ולכלל ישראל לא בא עד שיטבול, ומסיים שדברים ברורים הם עיי"ש. הרי לנו הגדרה חדשה על קביעת מצבו המהותי של זה שמל ולא טבל, והיינו שכבריה בפני עצמו הוא, דיצא מכלל עכו"ם ולכלל ישראל לא בא, ולכן אין דנין אותו כעכו"ם אבל לא חל עליו עדנה דין ישראל. וא"כ נהי דבפרט הדין של אם עושה יין נסך סתם השו"ע ביו"ד סי' קכ"ד סעי' ב' לפסוק דלא כהרדב"ז אלא דגר שמל ולא טבל מגעו אוסר בשתייה (והמקילין שם מקילין גם בגר תושב, ע"ש ברמ"א ובדר"ת סק"ג בשם הכנה"ג). י"ל דזהו מפני שסוברים דגזירת היי"נ =היין נסך= היתה כללית על כל מי שאיננו מוגדר עדנה בשם ישראל, אבל בנוגע אם לחלל שבת עבורו יש לומר דעל זה שכבר יצא מכלל עכו"ם וכבר קיבל עליו להיות כמונו אלא שנאנס ולא היה יכול לגמור הדבר (כלשונו של הרדב"ז שם) שפיר מותר כבר לחלל עליו את השבת ומצוה להחיותו כדי שישמור שבתות הרבה, ולפי הרדב"ז אפשר להגדיר את זה שמל ולא טבל שהוא בדומה לעובר העומד להולד דמחללין עליו את השבת. שהרמב"ן מסביר טעמו של דבר משום דחשוב כילוד ורק הדלת ננעל בפניו (הובא בביאור הלכה סוף סי' ש"ל), והא הרי כל עוד שלא יצא רובו או ראשו עדיין אין דינו כישראל ואין נהרגין עליו כדאיתא בפ"ז דאהלות מ"ו וברמב"ם פ"א דרוצח ה"ט ובתוס' נדה ד' מ"ד ע"א ד"ה איהו, ועוד. ומכל מקום מחללין עליו כבר את השבת בהיות ומעותד כבר לצאת לאויר העולם ורק הדלת ננעל בפניו, וא"כ אותה ההגדרה י"ל גם במל ולא טבל דאז הוא נמצא במצב של עיבור (ובאמת הרי אמרינן בזה דגר שנתגייר כקטן שנולד דמי) ומכיון שהוא מצידו כבר קיבל עליו להיות כמונו אלא שנאנס עדנה מלטבול, א"כ הרי דינו כאותו העובר שרק הדלת ננעל בפניו שמחללין עליו את השבת (שו"ר בשו"ת חלקת יואב ח"ב סי' ט' שתולה דעת התשב"ץ באי ילפינן מושמרו וגו' כדי שישמור שבתות הרבה או מוחי בהם והרמב"ם שפוסק לילף מוחי בהם אסור לחלל ע"ש, ולענ"ד לא יתכן לומר כן דהרי גם הב"י וכן יתר הפוסקים מביאין ג"כ בסתמא דברי התשב"ץ וזהו אע"פ דסברי להלכה דאין מטבילין גר בשבת, והחלקת יואב שם הרי מבאר דאי אין מטבילין אין מחללין). (ו) שוב מצאתי לשנים מהגדולים שבדור הקודם שדנו בשאלתינו אם מותר לחלל את השבת עבור גר שמל ולא טבל, והמה הגאב"ד מבראדי ז"ל בספרו שו"ת מחזה אברהם חאו"ח סי' נ"ד והגר"מ אריק ז"ל בספרו שו"ת אמרי יושר ח"ב סי' ק"ל, וכתובות המה להגאב"ד מטשעבין ז"ל שפנה אליהם בשאלה זאת גלל מעשה שהיה בגר שמל ולא טבל עדין ונסתכן לרגלי כאב המילה ושום נכרי לא אבה לעשות שום שירות עבורו לסיבת שנאת הדת, ושאל איפוא אם מותר לישראל לחלל שבת עבורו. והנה מתוך דבריהם שם יוצא לנו שהגאב"ד מטשעבין ערך גם לפניהם תשובה להוכיח שמותר לחלל עליו את השבת והביא בדבריו גם סייעתא מדברי התשב"ץ והרשב"א ביבמות שהבאנו. אלא דהגאב"ד מבראדי כתב לדחות כל דבריו, ודחק א"ע להעמיד דברי התשב"ץ בדיחוקין, או דמיירי כשגם טבל, או שהכוונה שימהרו עבור כן להטבילו בשבת, או דס"ל להתשב"ץ כהרמב"ן דטבילה קודם המילה, וס"ל להל' שלא לחלל שבת עבור גר שמל ולא טבל כיעו"ש, ולפענ"ד כל זה אינה במשמעות כלל, ופשטות דברי התשב"ץ מורים בבירור דסבר בפשטות דמחללין שבת עבור גר שמל ולא טבל, ואפילו למטותי' דהמחזה אברהם שם שס"ל לעיקר להעמיד דס"ל להתשב"ץ כהרמב"ן דטבילה קודם מה יעשה אבל על זה שכל הפוסקים הבאים אחריו ובראשם הב"י הביאו ג"כ דברי התשב"ץ בסתמא ולא פקפקו כלל על עיקר הוראתו שמותר לחלל את השבת עליו, ואנן הרי פסקינן דמילה קודם, וכמו"כ מ"ש לדמותו בדבריו להצלת עובר פחות מבן מ' יום שפסקינן שאין מחללין לפענ"ד יותר יש לדמותו לעובר שאמו יושבת כבר על המשבר וחסר רק פתיחת דלת וכנ"ל. ומשובחת יותר תשובתו של האמרי יושר שנטה קו בדבריו להתיר לחלל שבת עבורו וכתב שדברי התשב"ץ מפורשים להיתר, ולא מובן ע"כ מדוע שסיים שם שעכ"ז למעשה יש לעשות ע"י קטן או בשינוי או ע"י שנים להקל חומר האיסור, דהרי יוצא לנו דלא רק שדברי התשב"ץ מפורשים להיתר, אלא דיוצא מפורש כן מעשרות פוסקים שדנו בדברי התשב"ץ על שלא התיר למולו בחמישי, ולא עלה על דעתם כלל לפקפק על עצם התירו לחלל שבת עבורו אם יסתכן. וכמו"כ מה שרוצה האמרי יושר שם לחלק עוד ולומר דדוקא בנסתכן מחמת המילה שפיר אנו מצוין לרפאותו אף בשבת בכדי שיתרפא ויטבול ויהיה גר גמור, דבדבר הנצרך לענין הגירות אין דינו כנכרי, ומביא ראיה לכך מהא דכשבא להתגייר גם בטרם שהתגייר לומדין אותו אע"פ שאסור ללמד לגוי תורה וה"נ לענין חילול שבת עבור לרפאותו כדי שיטבול י"ל דשרי כיעו"ש, ולא אבין לענ"ד, ראשית לא מסתבר לתת הדברים לשיעורין ואם הדין הוא שמותר לחלל שבת עבורו צריך להיות מותר בכל גוונא, שנית לא מובנת ההוכחה מהא דמלמדין לגר שבא להתגייר, הרי שם המדובר כשאפי' לא מל עדנה ואז הרי בודאי עדיין דינו כעכו"ם גמור לכל דבריו אלא ודאי דלימוד תורה שאני שהיא הכשרה לגירותו, ולכן אין ללמוד מזה לפרטי קיום המצוות, [אלא דלענין זה שיהא מותר לו לשבות מצינו בתוספות ישנים יומא ד' מ"ח ע"ב ד"ה זה גר תושב דס"ל דלענין זה גם קודם המילה מצי כבר לשבות מכיון דדעתו ולהתגייר ע"ש], ושלישית, זיל בתר טעמא, אם מתירים לחלל שבת לרפאותו מהמילה בהיות וזה נצרך להגירות כדי שיתרפא ויטבול ויהיה גר גמור, א"כ הרי מאותו הטעם יש גם להתיר לחלל שבת לרפאותו גם כשהסתכן בינתיים מחמת דבר אחר, ולא רק מחמת המילה, וג"כ מאותו הנימוק כדי שעי"כ יתרפא ויוכל לטבול ולהיות גר גמור. (ז) על כן מסתברת לפענ"ד ההלכה בזה כפי שחיוה דעתו הגאב"ד מטשעבין ז"ל בדברי התשובה שערך לפני הגדולים הנ"ל שהיתה דעתו להתיר לחלל שבת עבור גר שמל ולא טבל, וכמו כן כפי מה שהציע בנו של הבעל מחזה אברהם האבד"ק מפרעמישלאן ז"ל בהשמטות שבסוף הספר ללמוד בעפ"י חידושו של האבני נזר שהבאנו לעיל דבגר שמל אע"פ שלא טבל נצטוה על שבת, דלפי"ד בגר שמל ולא טבל מותר לחלל עליו את השבת ושייך בו שפיר לומר חלל עליו שבת אחת כדי שישמור שבתות הרבה עיי"ש, והדברים מקבלים חיזוק וסותרים מה שיש לכאורה להשיב עליהם, בעפ"י ההגדרות שכתבנו בזה לעיל. (ח) אולם נ"ל דכ"ז כשהמדובר על גר צדק, אבל היכא שהגירות היא רק בגלל נימוק של מוטב שיאכל רוטב ולא שומן עצמו שכותב על זה הרמב"ם בתשובותיו, והדבר ידוע שאינו חוזר אלא בשביל דבר, בכל כגון דא אין מחללין עליו את השבת במל ועדיין לא טבל, ודיינו בזה מה שפוסק הרמב"ם בפי"ג מה' איסורי ביאה הי"ז (וכן בשו"ע יו"ד סי' רס"ח סעי' י"ב) וז"ל: גר שלא בדקו אחריו או שלא הודיעוהו המצות וענשן ומל וטבל בפני ג' הדיוטות ה"ז גר. אפילו נודע שבשביל דבר הוא מתגייר הואיל ומל וטבל יצא מכלל העכו"ם וחוששין לו עד שיתבאר (לשון השו"ע: עד שיתברר) צדקתו, ואפילו חזר ועבד כו"ם הרי הוא כישראל מומר שקדושיו קדושין, ומצוה להחזיר אבידתו מאחר שטבל נעשה כישראל עכ"ל. ודיינו בזה מה שמודגש מאחר שמל וטבל וכולי האי ואולי, וכמה קשים כדרבנות המה דברי הרמב"ם בהלכה י"ח שם שממשיך וכותב, דמפני זה אמרו חכמים קשים להם גרים לישראל כנגע צרעת, שרובן חוזרין בשביל דבר ומטעין את ישראל וקשה הדבר לפרוש מהם אחר שנתגיירו ע"ש, וד"ל.
