 שו"ת ציץ אליעזר חלק ז סימן א - קונ' קטן למפטיר פרק יב 

ענף י"ב (מד) גדול המאספים הבעל כנסת הגדולה הסכים ג"כ לפסוק להלכה כהב"י והרמ"א ודעימייהו, ולא עוד אלא שמעיד על עצמו שעשה מעשה רב בזה לשנות מהמנהג קודם ולהנהיג לכתחילה בכזאת. וכותב: ואני בבואי לעיר תירי"א מצאתי המנהג כדברי הריב"ש והרלנ"ח ז"ל ולא ישר בעיני ומנעתיו והנהגתי שקטן יקרא בפרשת המוספין ויפטיר זולת בפרשת זכור דלכ"ע הוא דאורייתא שיקרא גדול ויפטיר בה גדול עכ"ל. 

הן אמנם דיש להבין דנהי דסובר דאפשר להעלות גם קטן אבל למה לא ישר בעיני הכנה"ג החומרא שנהגו בעיר ההיא להתנהג כדעת הריב"ש עד שמצא לנחוץ לשנותה ולהנהיג לקרוא לקטן והרי לכו"ע גדול עדיף, וכבר עמדו להקשות על הכנה"ג בכך בספר דבר משה ח"א סי' כ' ובפרי האדמה ח"א פי"ב מתפלה הי"ז, ובאמת ליעקב בפרק תורת העולין לקס"ת אות כ"ח, ובספר בני חיי על הטור סי' רפ"ב או"ח א', ובברכי יוסף סי' רפ"ד אות ב', עיין בדבריהם. 

ואולי יש ליישב דהוא זה מפני שהבעל כנה"ג ראה עי"כ שנהגו כהריב"ש עזובה בחינוך הקטנים לקריאה בתורה, ומשום כך לא ישר זה בעיניו ועמד והתקין שהקטן יקרא בפרשת המוספין ויפטיר, ועי"ז חשב לעשות תיקון גדול לחנך את קטניהם בקריאת ס"ת, וכעדותו של הריב"ש בעצמו בסי' ל"ה שהיו מעלים המפטיר בדמים וקונים אותה בעד קטניהם להרגילם. 

ואפשר אולי לומר גם זאת דהחשש למנהגו של הריב"ש שהוא לקרוא מקודם גדול לקרוא בפרשה דחובת היום ולומר קדיש ולחזור ולקרוא לקטן המפטיר בפרשה היא, משום הנכנסים והיוצאים שיאמרו שמוסיפים על מנין העולין ביו"ט, והבעל כנה"ג הוא מכת הסוברים דאין להוסיף ביו"ט למנין העולים, או ביותר שחשש מפני הטועים שיבואו לחשוב שחובת היום בששה, וכדראיתי בספר שו"ת מים רבים חאו"ח סי' כ"ב שכתב באמת לחשוש משום כך עיי"ש. ולכן בהיות שלא היה אפשר בידו לבטל שלא יקראו לגמרי קטן במפטיר דמוספין, כשם שלא עלה בידי הריב"ש שם, או שהכנה"ג לא רצה לבטל לגמרי משום חינוך הקטנים, לכן העדיף ביותר שיקראו לקטן בלבד שזה מותר מעיקרא דדינא ממה שיקראו קודם גם גדול ויבואו לטעות כנ"ז. 

ושמחתי כשמצאתי אח"כ בספר שו"ת ראש משביר ח"א סי' י' שכותב נמי לפרש כמקצת דברי הנ"ל בכוונת הכנה"ג שלהנהיג לקרוא לגמרי רק גדול לא היה באפשרותו דלא היו שומעין לו כמ"ש הריב"ש שהם מוכרים ההפטרה למי שמעלין בדמים והם קונים אותם לקטניהם כדי להרגילם ולכך היה ישר בעיניו שיהיו נוהגים כהתוס' והמרדכי והרא"ש שקטן יקרא במוספין עיי"ש. 

(מה) וכמו"כ ראיתי בספר פרי האדמה שם שכותב להקשות דהכנה"ג סותר א"ע בפסקו הנ"ל ממה שמצינו לו בשירי כנה"ג או"ח סי' קל"ה הגה"ט אות ח' שכותב שם שבתשובה אחת השיב בלמי שיש לו החסרונות שהמנהג הוא שכל מי שקונה אותה חסר מן הקהל או הוא בקהל ואינו יכול לעשותה בעל החסרונות נכנס במקומו, וראובן קנה ההפטרות של כל השנה לבנו קטן פחות מי"ג ולפי שמוציאין שני ספרים אינו מן הדין שיקראו בפרשת המוספין מפטיר קטן, עולה בעל החסרונות וקורא בתורה ומפטיר, ויהי היום יום שני של פסח קדם אחיו של קטן וקרא בתורה והפטיר, ובעל החסרונות היה רוצה שיפרע שכר הברכות, והשבתי דאע"ג דלכאורה היה נראה שישלם כמ"ש רש"ל ז"ל במי שקנה ג' או ו' או ז' ליתא דכל מידי דלית ביה חסרון כיס לא מגבינן בבבל ואפילו אם תפס מפקינן מיניה כיון שענה אמן אחריו עיי"ש. הרי דבשירי כנה"ג מדבר הכנה"ג כעל דבר הפשוט שאינו מן הדין שיקראו בפרשת המוספין מפטיר קטן, וזהו ההיפך הגמור ממ"ש בכנה"ג הנ"ל. ואיך סתם איפוא בשיריו לכתוב דאינו מן הדין ונשאר בקושיא. 

אך בספר אמת ליעקב שם הרגיש ג"כ בזה, אבל ויהי כמשיב כלאחר יד דמ"ש בשכנה"ג לאו דאיהו ס"ל הכי אלא הרב כתב שם תשו' שכתב בתשובותיו ששאל השואל ודעת השואל היה כן לפי שכך נהגו במקומו, אבל איהו לדידיה אזיל ומורה להוראת מרן ז"ל שלא כדברי הריב"ש כמ"ש בכנה"ג עיי"ש. 

כן מ"ש שם הכנה"ג דדעת המבי"ט בתשובה בח"א סי' ק"ט כדעת הב"י ולחלק על הריב"ש [ואחריו נגרר לכתוב כזאת בשם המבי"ט באמת ליעקב שם] ג"כ כבר השיגו עליו כל הפוסקים הנ"ל שהזכרתי לעיל דבדברי המבי"ט מפורש ההיפך מזה וכי אדרבא מחמיר בזה עוד ביותר מהריב"ש. [ועיין מ"ש לדחוק א"ע בישובם של דברים בשו"ת ראש משביר ח"א סי' י' עיי"ש]. 

(מו) בשו"ת מים רבים שם סובר נמי לעיקר לפסוק כדעת רבותינו המתירים לקרוא לקטן למפטיר דמוספין ואפי' בד' פרשיות ומכ"ש עכשיו שהחזן קורא הפרשה ומוציא הציבור, ומאריך בזה, ולבסוף מביא עדות שליח א' מצפת ח"ר שמואל הלוי שמו שהעיד שמנהג ארץ ישראל להעלות תינוק אפילו לכתחילה בפרשת המוספין, ולשאלת הרב המחבר שם ששאלו איך נוהגים בעיר הקודש ירושלם פירט להעיד שכן נוהגים בשלשה ארצות של ארץ ישראל עיי"ש. 

אמנם הברכי יוסף באו"ח שם בסי' רפ"ד אות ד' כותב על עדותו זאת של שליח צפת שלא כוון יפה בעדותו על עיה"ק ירושלם ת"ו ועיה"ק חברון ת"ו ע"ש. [ועיין גם מ"ש בזה בספר שער המפקד ה' קס"ת ובנהר פקוד שם]. 

אבל אין לתפוס את השליח בזה כי כפי שמעידים במפורש הבעל פרי האדמה והבעל אמת ליעקב היו בזה מנהגים מחולקים בבתי הכנסיות אשר בירושלים זו בכה וזו בכה, וא"כ יכול להיות שהשליח דצפת איקלע בירושלים לביכ"נ שנהגו בכזאת וכן אותו הדבר בחברון, ועל יסוד זה העיד מה שהעיד. ומלבד הברכ"י העתיקו כל הפוסקים [עיין בפרה"א ובמזבח אדמה, ובאמת ליעקב, ושו"ת ראש משביר ח"ב סי' מ"ד וכן עיין גם מ"ש בזה בספר ארץ חיים] את דברי המים רבים מבלי לערער ע"ז כלל. 

(מז) עכ"פ ברור הדבר אליבא דכו"ע שאין להעיד על מנהג קבוע בירושלים שלא לקרוא לקטן, ובודאי דעביד כמר עביד ודעביד כמר עביד, ואולי יש מקום אפילו להכריע להתיר לקרוא לכתחילה לקטן בהיות וראינו לרובא דהפוסקים שהכריעו ככה לדינא, ואפילו מהצד שכנגד הרבה מהם רק מפקפקים משום חומרא בעלמא אבל לא מעיקרי דדינא ורק לפ' זכור יש מקום אולי להחמיר. ואולי גם לפ' פרה. [עיין שו"ת ראש משביר ח"ב סי' מ"ד ע"ש]. 

והכי רואים שנוטה להכריע בכזאת גם הברכי יוסף שם מלבד פ' זכור שראוי שיפטיר גדול אבל אם עלה הקטן אף בפ' זכור לא ירד. ובספרו לדוד אמת סי' ה' אות כ"ח פסק ג"כ דבמקום שנהגו שקטן אולה בפ' המוספין ומפטיר אין לשנות מנהגם רק פ' זכור לכתחילה יעלה גדול ואם עלה קטן לא ירד. 

גם בספר אמת ליעקב שם בתורת העולין לקס"ת אות כ"ח אחרי שמעיד שלא ראה בזה מנהג קבוע בעיה"ק ירושלים תוב"ב, כותב שלענין הלכה רבים הם החולקים על הריב"ש. וביותר מזה מצינו לו להלן מזה בפרק משפט קריאת המפטיר בס"ת אות ה' דאחרי שהביא דעת מרן הב"י דלא כהריב"ש הכריע במפורש דלענין הלכה אנן בתר מרן ז"ל גרירנן בין להקל בין להחמיר ע"ש. 

(מח) והנה בספר אמת ליעקב שם מוסיף וכותב: והרב דבר משה בא"ח סי' כ' כתב דהסכמת הרב מר אביו בק"ק שאלוניקי להפטיר קטן בפ' המוספין והוא העלה שם דלכתחילה לא יקראו לקטן דשב וא"ת =ואל תעשה= עדיף מיהו אם עלה לא ירד מאחר דרבים המה העולין בהסכמה שלא כדברי הריב"ש עכ"ל. 

ובברכי יוסף שם משיג עליו ע"ז וכותב: ודבריו קשים דמבואר בס' דבר משה דהרב אביו הסכים והוא ז"ל חלק עלינו את השוים ברא לגבי אבא עכ"ל. ועיין ביד אהרן ח"ב סי' רפ"ב הגהב"י אות ג' ושו"ת ראש משביר ח"ב סי' מ"ד שביארו ג"כ בכזאת כוונת הדבר משה. 

אכן עיינתי בגוף דברי הדבר משה וראיתי ששם כתוב בלשון זה: ולמעשה נראה דלכתחילה לא יקראו לקטן דשב וא"ת עדיף אמנם אם כבר עלה לא ירד מפני כבוד כמה מרבוותא דכתבו דקטן עולה למפטיר וכן הסכים הרב אדוני המלך עכ"ל. ולפי לשון זה שבדבר משה אין באמת הכרע דכוונתו דהרב אביו הסכים על ידו שלכתחילה לא יקראו לקטן, אלא יש לפרש בכוונת דבריו דבהיות דכמה מרבוותא כתבו דקטן עולה למפטיר אפילו לכתחילה וכן הסכים על ידיהם גם אביו הרב ז"ל לכן מפני כבודם אם כבר עלה לא ירד, וזוהי איפוא הכוונה שפירש בדבריו גם הבעל אמת ליעקב הנ"ל ומשו"ה למד מזה שדעת הרב מר אביו של הדבר משה להפטיר קטן בפ' המוספין, ורק הדבר משה כשלעצמו הוא זה שסובר דלכתחילה לא יקראו לקטן. 

וראיתי שכנראה גם הפרי האדמה שם הבין ג"כ כוונת הדבר משה כדברי האמת ליעקב, דמביא שם מ"ש בזה בס' דבר משה, ומסיים בלשון וכתב הרב יצ"ו דמנהג קוסט' שלא להעלות לקטן והעלה דלכתחילה לא יקראו לקטן דשב וא"ת עדיף אמנם אם כבר עלה לא ירד משום דכמה רבוותא הסכימו דקטן עולה למפטיר בהסכמת מר אביו ז"ל. ולשון זה שכותב הפרה"א דהדבר משה כתב דאמנם אם כבר עלה לא ירד משום דכמה רבוותא הסכימו דקטן עולה למפטיר בהסכמת מר אביו ז"ל, משמע לכאורה דר"ל בכוונתו כנ"ל, דהעלה כן משום דעתם של כמה רבוותא ובתוכם גם אביו ז"ל שהסכים על ידם כי בכלל קטן עולה למפטיר. וידוע דאביו של הדבר משה הוא הבעל כרם שלמה. 

(מט) והפרי האדמה שם סובר נמי מעיקרא דדינא כהב"י הרמ"א הב"ח והט"ז דקטן יכול לעלות אפי' לפ' זכור וקורא קילוס לדברי הב"ח בתשובה וכותב דדחה דברי הריב"ש בטוב טעם, אלא דמסוף דבריו נראה דסובר דמיהא למעשה יש להחמיר שלא לקרוא לקטן לכתחילה למפטיר דחובה, דמסיים בלשון אם כן גם בנ"ד דכבר קנו לו ההפטרות דמצי לקרות בתורה על ידי ש"ץ. אבל בהיות דנראה מדבריו שם דזה שאין לקרותו לכתחילה הוא רק מהיות טוב לצאת ידי כל הדעות, לכן נראה דבמקום צורך קצת יודה דיש להקל לקרותו אפילו לכתחילה. 

והכי מתבאר ביותר בספרו מזבח אדמה, (באו"ח סימן רפ"ד), הוא מביא שם אמנם חילוקי מנהגי הקהלות וכן חילוקי מנהגי הבתי כנסיות בירושלים בנוגע להעלאת קטן למפטיר דמוספין, אבל הוא כשלעצמו מבליע לנו בדבריו להשמיענו בנעימה דיותר טוב לקרות הקטן לחוד אפילו בפרשת מוספין כמ"ש מרן סימן זה להתיר [ר"ל. ממה שסתם לפסוק דקטן עולה למפטיר, ומשמע בכל מפטיר שהוא] ולהבית חדש אפילו בפרשת זכור. וחוזר ומדגיש בדבריו להשמיענו דעל אחת כמה דיש לקרות הקטן לחוד אם הוא כדי לזכות לקטן בפרטות שהי"ל שבת זה יום ייארצ"ט ומצוה רב לזכותו בעד אביו וכו' עיי"ש. 

ואפשר ללמוד מזה דלפי"ז ה"ה דמודה הבעל פרה"א גם לזאת דיש לקרות לקטן אפי' לכתחילה למפטיר דחובה כשזה בשבת שלפני הבר מצוה שלו דיש בה ג"כ ענין של קיום מנהג וחינוך לזכות לקטן בקריאה זו [עיין במג"א סי' רפ"ב ובשערי אפרים שער ב' סעי' י' ובביאור הלכה סי' קל"ו במה דנמנה שם בסדר החיובים לעליה לתורה נער שנעשה בר מצוה באותה שבת, ובמ"ש שם דנער שנעשה בר מצוה באותו שבוע הוא קודם לכל החיובים כיון שהוא זמן חינוכו עיי"ש].
