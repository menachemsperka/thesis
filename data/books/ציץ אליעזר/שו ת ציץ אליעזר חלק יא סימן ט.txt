 שו"ת ציץ אליעזר חלק יא סימן ט 

אם מותר לקרוא לתורה בר - מצוה שאינו נימול. 

ראיתי בספר הנדמ"ח נקרא בשם שו"ת רע"ז הילדסהיימער בחאו"ח סי' ד' שנשאל אם מותר לקרות לתורה בר - מצוה שאינו נימול ובסי' ה' נדפסה תשובה על כך מאת הגאון רי"א ז"ל מקאוונא, ולפי שלדאבונינו הרב מצוי לקרות גם בכאן בעיא זאת של העלאת ערל לקרה"ת =לקריאת התורה= עקב הקיבוץ הגדול מבין הגוים מארבע כנפות הארץ, ויש לי כמה הרהורי דברים על התשובה הזו לכן אעתיק בזה את לשונה: ע"ד מה שדרש כת"ר ממני להגיד חות דעתי ע"ד אותן שאינן מלין א"ע אם להרחיקן שלא לקוראן לעליות ס"ת ולא לצרפן למנין עשרה וכן לבר מצוה שלהן או לא כו'. הנה ע"פ דין הא קיי"ל מומר לערלות לא הוי מומר לכל התורה וכמבואר בחולין ד' ה' וביו"ד סי' ב' סעי' ז' ובש"ך יו"ד סי' רס"ד סק"ד. ומבואר בשו"ע או"ח סי' נ"ה סעיף י"א דאף מי שעבר עבירה כ"ז שלא נידוהו נמנה למנין עשרה, וע"כ /וע"פ/ הדין יש לצרפו לעשרה ולכל הנ"ל, ולכן אין לנו להרחיקן ויש לחוש שמא יצאו לתרבות רעה ולצאת מן הכלל, ואף דעכשיו הן פורשין א"ע מכלל הציבור עכ"ז יש לחוש דשמא יצאו ויהיו רודפין חלילה לכלל עמנו ודתינו כאשר עינינו ראו כזה בעו"ה בעתים הללו, וע"כ לדעתי אין להרחיקן לגמרי ואולי בתוך כך יראו קצת לשוב מן דרכם הרעה מעט ומעט עד שישובו לגמרי בעזר ה'. וכן בענין זה כתב הט"ז ביו"ד סי' של"ד סק"א, ובנ"ד אף הרמ"א דשם יודה למה שכתבתי, דהא שם דעת הרמ"א לעשות כפי הדין ולנדותן (באם האפשרי בידינו לנדותן ולא כמו בצוק העתים שלנו) וע"כ אין לנו לחוש דשמא יצאו לתרבות רעה כו'. אבל בנ"ד דע"פ דין מצטרף לעשרה ולכל הנך ע"כ אין לנו לעשות מעצמינו שלא כפי הדין ובפרט כיון דיש לחוש שיצאו לתרבות רעה ביותר חלילה ויבדלו מכלל ישראל ח"ו לגמרי, וע"כ יעשו כפי הדין ולקרותן לס"ת וכיוצא, ויש לי עוד מה להאריך בכ"ז אך לפי שהנני חלש מאד וכן מחמת חולשת עיני ל"ע אין לי כח לכתוב כלל והנני יושב כעת מחוץ לעירי וכפי אזהרת הד"ר עלי וגם אין לי ספרים הנצרכים לזה ע"כ נכתב מכתבי בקצרה, ובעיקר הענין יעשה כת"ר כפי ראות עיניו הטהורים וד' יהי' בעזרתינו ולהאיר עינינו בתורתו הק' ולהורות לנו הדרך הנכון ולהסיר מכשול מעינינו ויעבדו אותו כולם ולשמור מצותיו כד"ת וכהלכה ונזכה לשמחת הכלל ולהרמת קרן התוה"ק ויראתו בשלימות עכ"ד התשובה. 

ולפענ"ד נראה דחרפה היא לצרף למנין ולקרוא לתורה לאיש אשר לו ערלה, דבבואו ככה בקרב עדת ישראל בביהכ"נ הוא בבחינה של ויקרב אל אחיו את המדינית לעיני משה ולעיני כל עדת בני ישראל, ובהעיזו לגשת אל תוה"ק לברך עליה הוא בבחינה של טובל ושרץ בידו, וגרוע הוא בבחינה זאת יותר מכל עבריין אחר איזה שהוא כי הרי נגש הוא יחד עם חרפת העבירה, ובאותה שעה שעובר בכל רגע על איסור כרת באי הסרתה ממנו. ואיך יתכן שיעלה לקרות בתורה בציבור והלא אין לך חילול השם וכיבוי מאור הדת גדול מזה, וכל העומד ונמצא שם באותו פרק באותו מקום ואינו מוחה ילכד בעוונו ובחבלי חטאתו יתמך (כלשון שו"ת חכם צבי סי' ל"ח), ואין לשעות בכגון דא לחששא לפן יבאישנו ביושב הארץ, כי הכזונה יעשה את אחותינו לבוא ולכבוש את המלכה עמי בבית, ואנו נחשה ועוד ננענע לו בראש? גם אין לחשוב לפן ואולי עי"ז שנקרבהו ישוב מעט, אלא אדרבה בכגון זה יש לנו לומר: הלעיטהו לרשע ימות /וימות/, וזה שאמרינן בעבר עבירה כ"ז שלא נידהו נמנה למנין עשרה, עד כאן לא אמרינן הכי אלא בעובר לפרקים אבל העושים ביד רמה בשאט בנפש ופורקים מעליהם מצות אפי' מצוה אחת קלה הרי הוא מאותן שאין להם חלק לע"ה וכו' (כדמבאר בשו"ת חכם צבי שם) מכש"כ בכגון נידוננו שאין לתלות שעושה לעבור ע"ז משום בצע כסף, אלא עושה כן רק כדי להכעיס את בוראו ולשם כפירה בתורה מן השמים ח"ו, א"כ בודאי שאסור לשבוע בחרפה ולהרשות לאיש רשע כזה ליקרב אל המזבח ולברך עה"ת, כי אין זה מברך אלא מנאץ ומלגלג בפועל באותה שעה בערלות לב וערלות בשר על מה שמוציא בפיו מטבע הברכות שהמה שבח והודאה לנותן התורה שבחר בנו מכל העמים ונתן לנו תורתו תורת אמת, וסו"ס הא אין באי קריאתו לתורה איזה ביטול מ"ע עד שנצטרך להעמיק אם זהו בגדר של אין אומרים לאדם רשע הוסף רשע והמנע מן המצות שכותב הרמב"ם בפט"ו מנשיאת כפים ה"ו. [ויעוין במג"א סי' קכ"ח ס"ק נ"ד ושיוכנה"ג הגהב"י אות כ"ד בפלוגתא אם מומר לערלות עולה לנ"כ =לנשיאת כפיים=, ובמה שצוין בכה"ח ס"ק רכ"ב, וכן מ"ש בזה בשו"ת פני מבין חאו"ח סי' כ"ה נתיב ג' אות ו' עיי"ש]. 

אפרוש עוד קצת את היריעה בזה ואעידה לי עדים נאמנים לצדקת אמרינו בדברים הנלמדים בקל וחומר ובגזירה שוה, ואשר בריח תיכון אחד מבריח אותם מן הקצה אל הקצה להגביל את הר ה' ולקדשו שכל ערל בשר לא יקרב אליו לנגוע אל קצהו. 

בספר שו"ת מים רבים (מילדולה) חיו"ד סי' נ"א נשאל על אחד מאנוסי הזמן שהציל את נפשו מני שחת ארץ גזרה ובא לעיר שדרים בה ישראלים ובסוד ישרים ועדה מודה ה' בכל לבב וזמנין תלתא ביומא ברך על ברכוהי ומצלי קורא ק"ש וברכותיה ומתעטף בציצית ומתעסק במצות כאשר הוא מזרע אברהם אבינו, אלא שעדיין לא נכנס בבריתו לסיבת ממונו הנמצא מפוזר ומפורד בידי גוים באותה העיר אשר יצא משם וגברא שהי בה למידע היצלח ה' אורחיה אם לא, ויצטרך לשוב שם להציל ממונו, לא לכוונה נפסדת ח"ו, ופגע בו הרגל ובבית הכנסת קנה מצות פתיחת שערים להוציא ס"ת מארון הקדש וליתנו מיד ליד המובילו לבימה. אם כפי הדין מותר להניחו שיתעסק במצוה זו ולאחוז בדבר קדושה כזה כל עוד שלא מל ולא טבל ואין בזה נדנוד איסור, או אם יש לדחותו ולמחות עליו ולעכב על ידו כשם שאין מספיקים בידו להניח תפילין עד היותו מהול. 

והחכם השואל צידד לומר שלא יעכב עליו איחור הכנסתו בברית דאנוס מיקרי לענין זה ורחמנא פטריה דיש לך אדם שממונו חביב עליו יותר מגופו ומה גם כי יש לחוש שלא ישוב להטמע בין הגוים, וגם ע"ז שאין מניחים אותו להניח תפילין מהנדז דמה טעם בציצית לא קפדו והרי כתיב גם ביה והיה לכם לציצית, כשם שבתפילין כתוב והיה לך לאות ודרשינן ולא לאחרים לאות, וגדולה מצות ציצית ששקולה ככל המצות, ומביע דעתו שאף בתפילין מותרים (אלה האנוסים) דלאו אחרים מיקרו. 

הרהמ"ח מסר את השאלה לטיפול לחכמי בני הישיבה הכלית /הכללית/ של ק"ק ליוורנו, והמה בתשובתם חילקו את הדבר לשלש מחלקות, והמה: (א) בנוגע לאחיזת ס"ת הסכימו להחכם השואל שמותר בזה מעיקרא דדינא, ברם אם כי שהדין כך אין השעה צריכה לכך ואין להשוותם לישראלים גמורים להתיר להם ההשתמשות בדברי קדושה כל עוד שלא נמולו כי נפיק ממנו חורבא כי יתעצלו מהכנס בבריתו של אברהם אבינו בראותם שעם היותם ערלים אין מונעים מהם דבר וכו', זולת כאשר יהיה אנוס כמשמעו. (ב) בנוגע בהנחת תפילין יש להחמיר אפילו באונס ממש, יען שכבר ידענו כי על פי שנים עדים יקום דבר ועד א' לאו כלום הוא ואדרבה בזיון יהיה לו לבא להעיד יחידי מאחר שדבריו לא יהיו נשמעים כי לא יקום עד א' באיש וכן הנה כל איש ישראל החתום באות ברית קודש בהניחו תפילין שנקראו אות הרי הוא כמצרף ומייחד ב' עדים שיעידו על היותו מאמין באלקים ובתורתו ועדותם נכונה וצודקת, אבל הבלתי נימול מה יועילהו הנחת התפילין והרי הוא כמביא עד א' להעיד עליו לזכותו שלא יועילהו עדותו ויהיה בזיון לעד, כמו"כ יהי' בזיון לתפילין בהעידם על האיש הזה לבדם, מה שאין כן מצות ציצית שלא נקראת אות. (ג) והמונעים עצמם ומאחרים להמול לא לסיבת אונס כי אם לסיבת עצלות יש להחמיר מאד עליהם ולמונעם מהשתמשם בשום דבר של קדושה ושלא יעטפו בציצית ולא יבואו להתפלל ביהכנ"ס כי אם יחידים בביתם עד שיהיו נמולים כמונו והיינו לעם א'. 

גם הרהמ"ח בעצמו, אם כי שהחכם השואל והמציע היה זקנו, בכל זאת לא הסכים להיתרו, ובא אחר החיתום של חכמי הישיבה, וכתב ג"כ לדחות ראיותיו שכתב בזה, ועוד הוסיף נימוק חשוב לאיסור, והוא זה, דמסתברא דאם ממקום טינופת צריך להרחיק הס"ת משום שנראה שמבזה התורה כשמקריבין אותו במקום נמאס ומטונף ובעינן והיה מחנך קדוש וצריך לנהוג כבוד גדול, ואסור ליכנס בס"ת לבית המרחץ או לבית הקברות וכמה דקדוקים דקדקו הפוסקים ז"ל על נקיות וטהרת האוחז ס"ת וכו' כ"ש שיש להרחיקו ממגע ערל שאין דבר מאוסה כערלה ומי הוא יותר ערום מהאיש הזה שהוא ערום ממצות מילה ואעפ"י שכל כונתו ומגמתו של אותו האיש אינו אלא לחבב התורה ולכבדה שכן קונה פתיחת השערים בדמים יקרים וסוף סוף ד"ת אינן מקבלים טומאה וכו', מ"מ לנו ולבנינו יחשב לשמצה לפי שאין אנו חרדים לכבוד התורה ולהיות קנאים לעמוד על משמרתינו אשר לא יקרב פגום וטמא אצלה, כי הגם שהיא אינה מקפדת על עצמה ואינה מתפעלת לפי שאינה מקבלת טומאה, אפילו הכי חובה עלינו להקפיד ולחוש שישמשוה בני אדם מהוגנים ונקיים ולכן מכל הטעמים הללו דעתי נוטה להחמיר, ע"כ תוכן הדברים שם בסימנים נ"ב נ"ג עיי"ש. 

אנו רואים כמה רעדו אמות הסיפים של גדולי חכמי בני הישיבה דכוללות ק"ק ליוורנו שלא לתת לאיש אשר ערלה לו אפילו לפתוח שערי ארון הקודש להוציא ולהחזיק את הס"ת בלבד, וראשם הגהמ"ח ז"ל החרה והחזיק וחיזק איסורם זה, ומבלי פנות אל החששא לשמא יגרמו עי"ז שיחזור ויטמע בין העכו"ם ואדרבא חששו יותר על להיפך שאם נכבדם יתעצלו מהכנס בבריתו של אאע"ה =אברהם אבינו עליו השלום=, וזאת גם זאת, אפילו כשהמדובר על כזה שמקודם היה אנוס ממש להיות ערל. ורק אחרי שנשתחרר מאונסו איננו ממהר עוד למול א"ע, ולא בשרירות לב אלא גלל סיבות ממוניות. וחוץ מעון זה גלל בצע כסף הוא רד עם קל מחזיק באמונת אבותינו ועובד אותו בלבב שלם ומחזר אחר מצוות, בכל זאת החילו גם על שכזה איסורם. כדמפורט בהדיא בספר שם וכנז"ל. 

אם כן נלמד מזה בק"ו בן בנו של ק"ו על כגון נידוננו שהמדובר גם על לעלות לתורה לאחוז בה ולברך עליה, וגם על בכזה שמתוך קשיחות ושרירות לב ורשע וזדון לא מלו אותו אבותיו וממשיך לאחוז ברשעותו זה, שודאי ובודאי שאסור לקרותו לתורה וצא טמא יאמר לו למען ישמעו ויראו, ומבלי לחשוש לכל תוצאות שהם, פן תחגור חלילה עלינו התורה שק. 

עוד חידוש גדול נלמד מדברי הספר, דבר שלא עלה על דעתנו, כי נהוג היה מאז מקדם שלא לעזוב למי ששב אל אמונת אבותיו להניח עדנה תפילין כל עוד הוא עדנו ערל, וחכמי הישיבה בליוורנא /בליוורנו/ חיזקו מנהג זה ולא נתנו לערערו, וזאת אפילו כשהוא אנוס ממש לכך וטעמם ונימוקם עמם כנז"ל. ונהי דעל אנוס ממש יש הרבה לדון על הנחתם זאת עפ"י סברתם שהמציאו ודימו לומר דצריכים לזה ב' עדים ושזהו דבר המעכב, ומה נעשה לערל שמתו אחיו מחמת מילה? (עיין יו"ד סי' רס"ג סעי' ב' ופ"ת שם ושו"ת עין יצחק חיו"ד סי' כ"ה ע"ש ואכמ"ל), אבל על אנוס רק משום ממון וכ"ש על פושע ובן ממר בודאי שיש מקום רב לסברתם והנחתם ואולי גם להנחתם הנוספת למנוע מלהשתמש בשום דבר של קדושה וגם כלה גרש יגורש מנחלת ה' דמי ביקש לפושעים כאלה לבוא ולרמוס בחצרות בית ד'. אע"פ שמעיקרא דדינא יש לשקול גם את הדבר הזה כי בדרך כלל הא אין אומרים לאדם רשע הוסף רשע והמנע מן המצוות כדברי הרמב"ם הנ"ל בפט"ו מנשיאת כפים, אבל משום גדר וסייג בודאי יש בזה להבדילו לרעה. 

וזה לא ניתן כלל לדחות ולומר שהאמור בספר שם הוא מפני זה שהמדובר במי שהיה מן האנוסים שזקוק גם טבילה אחרי המילה מדי חזרו אל אמונת אבותיו, דזה אינו כי הרי מפורש נאמר ונשנה שם שכל העיכוב הוא רק מפני שהוא ערל בשר, ועוד הא הטבילה שם אפי' להמצריכים אותה אינו אלא מתורת חומרא בעלמא (יעוין יו"ד סי' רס"ח סעי' י"ב ופ"ת סק"י ומ"ש בשם שו"ת רשב"ש עיי"ש) ואיך אפשר לומר שזהו הדבר המעכב והמונע ממנו מלקיים מצות הנחת תפילין וכו' שמחויב בזה מעיקרא דדינא, אלא בודאי שהמונע בזה הוא רק בגלל היותו ערל בשר בלבד וזאת מפני שהיא אות חתום בשרינו /בבשרינו/ שמעידה בנו שבחר בנו ה' מכל העמים ומאוסה היא הערלה שנתגנו בה הגוים, וכל המפיר בריתו של אאע"ה אין לו חלק לעולה"ב, ונכרתו עליה י"ג בריתות (עיין נדרים ל"א ל"ב ורמב"ם פ"ג ממילה וטויו"ד סי' ר"ס). 

זאת שנית, בשו"ת שואל ומשיב מהדו"ת ח"ב סי' ס"ד נשאל על כגון דא ממש, על אחד מהזדים שלא רצה למול את בנו באמרו שאף שלא ימול מ"מ לא יצא מכלל דת יהדות לכל הדברים, ורב אחד רצה להוציא אותו הילד שלא נימול מכלל יהדות שלא יעלה לתורה והוא ערל, ופנה לבקש הסכמתו, והגאון בעל שו"מ ז"ל השיב לחזק את ידו ובירר דמילה חמיר משאר עובר ל"ת שיש בה כרת משום דמצות מילה עוד נצטווה א"א ע"ה ובניו אחריו המתייחסים לזרעו של יצחק, ובדת יהדות מתיחסים אנו לא"א שהי' ראש לגרים, ועד איתא במדרש תנחומא פ' משפטים שכן השיב עקליס לאדרינוס המלך דודו שכל שאינו נימול א"י ללמוד תורה וכו', לכן כל שהוא ערל א"א לו להשיג תורה וא"כ אינו בכלל יהדות, וגם ראינו שמשה רבינו נענש על איחור מילת בנו, וע"כ דאינו יהודי כלל יעו"ש ביתר אריכות, וכן במה שמציין לעיין בהמובא מזה גם בקו' מנחת קנאות להמהר"צ חיות ז"ל עיי"ש, ואולי כמשפט החמור הזה שהוציא השו"מ סבר לה נמי להבעל כנה"ג ומשום כך בשיוכנה"ג סי' קכ"ח הגהב"י אות כ"ד חרץ בפשי' משפטו של ערל כהן וכתב וז"ל: ולדידי הדבר פשוט דלא ישא כפיו בעוד שלא נכנס בבריתו של אברהם אבינו. אך יש לעיין ממ"ש הפר"ח שם בס"ק ל"ט עיי"ש. 

כן ראיתי בספר שו"ת רשב"ן (שיק) חאו"ח סי' ס"ז שיושב ג"כ על מדוכת זאת השאלה, והגם שמתמרמר מאד על הוראת השו"מ שאינו יהודי כלל וכן על אחרים שהורו גם שלא לסדר לו חו"ק (וידוע שכמה מגדולי הונגריא רננו אחרי מחבר זה הספר יעוין לדוגמא בשו"ת לבושי מרדכי מהדו"ק חיו"ד סי' פ"ח ע"ש) עם כל זאת סמך גם הוא את שתי ידיו שלא לקרותו לתורה, כי חמור הוא מכל חייבי כריתות, לפי שמי שלא נימול בכל עת ובכל שעה שבא אל הקודש עליו ועל גופו הוא העון, והוא הוא שעליו נאמר ותהי עונותם על עצמותם, ואיך יברך אשר בחר בנו וכו' ועונו בגופו, לזאת כתורה פסק בעל שואל ומשיב להבדילו מן הקהל ושלא לצרפו לכל דבר שבקדושה ושלא יעלה לתורה, בחיים ישב מחוץ למחנה ובמותו לא ישכב בין התמימים כי עונו בגופו. יעו"ש ביתר אריכות. 

עוד זאת ראיתי בשו"ת מהר"ם שיק חיו"ד סי' רל"ח שהעלה דהאידנא סתמא דמילתא מי שאינו מניח למול בנו מסתמא הוא גם מומר לשבת ואפיקורס גמור, ומסתמא גם בניו יהיו כמוהו, ואין לנו לחוש עליהם, עיי"ש. ולפי"ז נסתר מחמתו גם עצם ההשוואה לדין מומר לערלות דלא הוי מומר לכה"ת =לכל התורה=, כי בזמנינו ברור שאיש שכזה ומשפחה שכזאת בודאי הנם גם מומרים לחלל שבת שהוו מומרים לכה"ת. 

כן ראיתי כעת בספר שו"ת מהרש"ג ח"ב סי' ר"ג שהעלה ג"כ בדבריו מעין הסברא שכתבנו לעיל דשעת עבירה שאני, דבשעת עשיית עבירה ממש אז האדם הוא מרוחק מן הקדושה בתכלית ההתרחקות. ולכן פסק בנידונו שם דאם הנער שאינו מהול הוא גדול דכבר מוטל עליו החיוב מילה אזי אין ללמדו תורה אפילו בינו לבין עצמו כיון דהוי בשעת עבירה ממש עיי"ש, ודון מינה ומינה. 

אוסיף לציין מה שראיתי כתוב בזה עוד בשו"ת מלמד להועיל חיו"ד סימנים ע"ט וקט"ו, לעיין /לענין/ נשואין וקבורה. וגם לרבות מ"ש בזה בשו"ת מנחת יצחק ח"ד סי' י' עיי"ש. 

ונראין הדברים שגם הגרי"א מקאוונא בתשובתו שם עדנה היה מהסס למעשה בפסקו, ואם לא לדינא עכ"פ משום גדר וסייג למען לא תפוג תורה ולא ירופפו עמודיה, ומשום כך השמיע בסוף דבריו נימא. שהנהו יושב חוץ לעירו ואין לו ספרים הנצרכים לזה והשאיר כף ההכרעה לזה בידי הרב השואל בהשמיע לו שבעיקר הענין יעשה כפי ראות עיניו הטהורים. ובאמת הכריע כנראה להרחיק כי כך כותב בשו"ת מלמד להועיל בסי' ע"ט שם שבשטאטוטען דעדת ישראל בברלין ודעדת ישראל בפפד"מ ניתקן שכל איש ערל או מי שלא ימול את בנו נגד דת תוה"ק לא יוכל להיות חבר לקהל יראים ושזה נעשה ע"פ ה"ה ש"ר הירש ומו"ה עזריאל הילעסהיימער /הילדעסהיימער/ ז"ל. ובשו"ת רשב"ן שם, ומובא גם במלמד להועיל שם, מעתיק בשם הש"ר הירש שהתקין מפורש שלא לקרוא בתורה אינו נימול עיי"ש. 

מכל האמור ומתבאר נראה דאין להעלות לתורה ערל שלא מלו אותו אבותיו מתוך מרד ומעל, או אפילו אם תחילתו באונס אבל סופו ברצון וממשיך בזדון בנשיאת חרפת ערלתו עליו, אם מדינא, ואם משום גדר וסייג, ובמקום לשאול אם להעלות שכזה לתורה, יש לשאול יותר אם בכלל יש אפילו מקום לערוך לשכזה חגיגת בר מצוה הבאה בעבירה. ולכאלה יש להשיב תשובה ברורה חד משמעית לאמור: אך בזאת נאות לכם את תהיו כמונו למול לכם כל זכר.
