 שו"ת ציץ אליעזר חלק י סימן כה פרק כז 

פרק כז דיני אדם טריפה לח"א. (א) אודות דיני אדם טריפה, אמנם ההלכה היא שההורג את הטריפה אינו נהרג בידי אדם, אבל אין זה מוציא את האדם ההורגו משם רוצח כדביארתי במקו"א (בצ"א ח"ט סי' י"ז פ"ו וביותר לעיל פ"ה), ופרט להאמור דינו של הטריפה כחי לכל דבר, ונעלה מכל ספק שמחללים עליו את השבת כעל שלם, כדהבאתי בספרי שם מדברי גדולי הפוסקים בזה, וכדלא כדברי המנ"ח במצוה רצ"ו כיעו"ש, והמנ"ח בעצמו כתב במקו"א, במצוה ת"ר, שבדבריו בזה בא רק לעורר ולמעשה צ"ע עיי"ש, ולפי הנתבאר בהפוסקים בזה אין מקום גם לעיון וברור שמחללין, וכך ראיתי גם בשו"ת אחיעזר ח"ג סי' ס"ה אות י"ד שכותב דחלילה להעלות על הדעת סברא כזאת שלא יחללו על הטריפה את השבת, דמחללין אפילו על חיי שעה ומכש"כ בטריפה שיכול לחיות זמן רב עיי"ש. גם לענין אם מותר להורגו כדי להציל ממות את שאינו טריפה ג"כ בארתי הדק היטב בספרי שם את ההלכה בזה שאסור כדיוצא בפירוש מדברי גדולי הפוסקים ודלא כהמנ"ח וגם ביארתי שאף המנ"ח לא אמר את דבריו כי אם כשלמעשה דורשים בפה גם דמו של הטריפה או דמו של זה או דמו של זה ולא כשדמו של הטריפה לא נדרש בכלל, וכן רק כשהמסוכן בעצמו מוסר אותו, וגם מוסרו להאנס הדורש בתוקף זאת כיעו"ש. ולאור מה שביררתי בספרי שם נדחית מהלכה גם הדברים לפילפולא שמצינו שכותב בזה בשו"ת בית יצחק חאו"ח סי' י"ט וביותר בחיו"ד ח"ב סי' קס"ב כיעו"ש, והבית יצחק בעצמו כותב שם (בסי' קס"ב) דלשיטת התוס' דאדם חי יותר מי"ב חדש לא מקרי איפוא חיי טריפה רק חיי שעה יעו"ש, והגע בעצמך האם נאמר שגזל ממונו של טריפה יותר יותר משל ברי? וא"כ האם ממונו חביב עלינו יותר מגופו? וחוץ מכל האמור יש לשים לב למציאות של זמנינו שנשתנו הטבעיים ונתגלו גם סממנים ואופני ניתוחים להחיות בהם גם אנשים שדינם לפי ההלכה כדין טריפה, וכדמבאר מזה החזון איש נשים סי' כ"ז אות ג' עיי"ש. ולפעמים ע"י ניתוח מוציאים אותו גם מידי מצב טריפותו, ואין לזה שייכות לדיני טריפות בבהמה ובמה שנחלקו הפוסקים אי אמרינן בטריפה חוזרת לכשרותה או לא, עיין בדע"ת סי' מ"ד ס"ק י"ט כ', וכן מ"ש בזה בספר אשל אברהם חולין בפירות גינוסר סי' צ"ו עיי"ש, ופשוט. ואכמ"ל. (ב) ובדבר מה שהגאון הגרי"ש נאטאנזאן ז"ל בהסכמתו לספר שו"ת שבות יעקב מביא בשם שו"ת הרב מהר"ח אלפנדרי בספרו מגיד מראשית חיו"ד ס"ס ב' שהעלה בעפ"י המבואר בנדה ד' כ"ג, דבאחות אשה דלא מיתסר אלא בחייה כל שטריפה אינה חיה אין איסור אחות אשה ע"ש. במחכ"ת לא דק, דעיינתי בגוף דברי ספר מגיד מראשית וראיתי שאדרבא העלה להיפך מזה דכותב וז"ל: ויש ראיה לזה ממ"ש בפ' המפלת לר"מ דס"ל המפלת כמין בהמה תשב לזכר ולנקבה, ופריך קבל בה אביה קדושי מהו למאי נפ"מ למיתסר באחתיה והדר פריך ומי חיי וכו' וכתבו התוס' שם דהוה מצי למבעי אם בא עליו אי חשיב כרובע בהמה וכו' ומשמע מדבריהם דמאי דקאמר תלמודא ומי חיי היינו שסופו למות וקס"ד למיתסר באחתיה ה"ה לטרפה עכ"ל, הרי שהעלה לאיסורא, וראיתו היא מפני שהתוס' שם מפרשים דלענין גופה לא בעי אם הבא עליה חייב משום א"א =אשת איש= או אם חייב ההורגה, משום, דפשיטא ליה דאינה חיה ג' שנים שתהא בת ביאה ולא שנה שלא תהא טרפה, חזינן שהתוס' פירשו דגם בהס"ד ידע שפיר שהיא טריפה ובכל זאת מיבעי' ליה אי מיתסרי באחותה, עד דדחי לה דלא חיי בכלל, א"כ הרי ראיה דגם טריפה מיתסרא באחותה. והכי ראיתי בספר עין הרועים להמהרש"ם ז"ל ערך טריפה שהביא בשם המגיד מראשית דפשט לה לאיסורא ע"ש. ויעוין בספר ברכי יוסף אה"ע סי' ט"ו אות ד' שאחרי שהביא ספיקו של המגיד מראשית, הוסיף וכתב, דאיברא י"ל דלמסקנא כיון דלא חיי לא מתסר וה"ה לטרפה אך מדברי הרמב"ן והר"ן שם בחידושיהם יש ללמוד דמתסר עיי"ש. ואמנם הרמב"ן והר"ן מפרשים שם שכוונת הדחייה מפני שאי אפשר לה לחיות כלל ואין קדושין תופסין בה יעו"ש. וכך מבאר גם בחידושי המאירי שם. שוב ראיתי שהגרי"ש ז"ל בעצמו בחיבורו הגדול שואל ומשיב מהדו"ק ח"ג סי' ר"א העתיק ג"כ בשם המגיד מראשית שהעלה להוכיח מההיא דנדה לאיסורא, ורק כותב דיש לדחות הראי' ולומר דזהו דקמקשה הש"ס ומי קא חיי וכו', וזהו כעין שרצה לדחות הבר"י, אבל נרתע מזה מכח דהראשונים לא ביארו כן וכנ"ל. ומה צדק הגאון בעל חקרי לב ז"ל בחאה"ע סי' י"א דאחרי שהעתיק ג"כ בשם המגיד מראשית שהעלה בזה לאיסורא, נו"נ באמונה והוכיח דאין תיבת חי ממעט אלא מת גמור ולא טריפה וכי כתיב עליה בחייה לא התיר כי אם אחר מיתה ולא כשהיא בחיים והיא טריפה ופשוט הדבר דבאשתו טריפה אסור באחותה ע"ש. (ג) ויעוין עוד בספר פתח הדביר סי' קצ"ט אות ט' וסי' שכ"ט אות ו' שכתב אריכות דברים בדיני דטריפת אדם וזאת הלכה העלה ג"כ דפשוט הדבר דמחללין עליו את השבת וכן דחזי לאיצטרופי למנין עשרה לכל דבר שבקדושה וקדוש ה' דכל עוד נשמתו בו חי הוא לכל מילי, ועלה ונסתפק רק אי חזי להיות ש"ץ כשהוא טריפות הניכר ונראה לעין וכתב שבמחלוקת הוא שנוי עיי"ש, ועוד לו בסי' רי"ט אות ג' כתב להסתפק בחולה שנתרפא מחוליו אבל הרופאים אמרו באומדנא דמוכח שנעשה טריפה באברים הפנימיים אי יברך הגומל, מי נימא כיון דעכ"פ החולי חלף עבר ונתרפא ועמד על רגליו מברך, או"ד כיון דנעשה טריפה מחמת החולי וקי"ל דטריפה אינה חיה לא נתחייב בברכת הודאה כיון שעדיין מעותד למות והוי כלא נתרפא, ובדברי תשובתו חזר להביא מדברי הפוסקים שהעלו דאדם טריפה חשיב כחי לכל מילי, ולכן הגם שנטרף כיון שניצול מצרתו חייב להודות על חיי שעה, ומה גם דאדם דאית ליה מזלא יכול לחיות הרבה, וכן הא איכא מ"ד דטריפה חיה, והגם דהלכתא דטריפה אינה חיה הרי כתב הגאון רש"ל ז"ל דהיינו ברוב אבל במיעוט חי. וכמו"כ הביא משם הרא"ח ז"ל כ"י בסוף ה' טריפות שכ' דמ"ש רז"ל טרפה אינה חיה י"ב חדש כמה פעמים אנו רואים שמתרפאת וחיה לפיכך יש מפרשים אינה חיה בטריפותה אבל מתרפאת וחיה יותר מיב"ח וכו' ולא אמרו אינה חיה אלא בעודה טריפה כל עוד שרפואות תעלה אין לה, ולזאת העלה להלכה דמעיקרא דדינא זה החולה שנתרפא ונשאר טריפה מחמת חולי מצי שפיר לברך ברכת הגומל, אלא משום חומר לא תשא ובהיות שמחמירים בס' ברכות שלא לברך, ואפילו באיכא כמה ספיקות לברך אמרי' שואת"ע =שב ואל תעשה= עדיף הוסיף לכתוב דמן הטוב והישר שיברך ברכת הגומל בלא שם ומלכות יעו"ש באריכות דבריו דיש ללמוד מהם כמה וכמה הלכתא גבירתא ואין להאריך יותר, ואעיר בקצרה רק זאת בנוגע לברכת הגומל כשנשאר טריפה, דנראה לומר דאם החולה אינה /אינו/ יודע מזה וחושב שאכן הבריא, ובא לשאול אם לברך ברכת הגומל, דאז יש להעמיד הדבר על עיקר הדין ולהורות לו שפיר לברך בשם ומלכות, (ומכ"ש שאין למחות בידו אם נגש בעצמו לברך הגומל) דאם יורו לו לברך בלי שו"מ =שם ומלכות= ירגיש בדבר ומחשבותיו יתחילו להטרידו שחולה הנהו במחלה שאין סיכויים שיבריא ממנה, ובמקום להיות מעודד שקם ממטתו תשבר רוחו בקרבו ויגרמו בעקיפין להגברת מחלתו. ולכן מוטב הדבר ומן הראוי להורות לו מיד לברך ברכת הגומל בשו"מ שניצול עתה מצרתו וקם ממטתו שהיה מרותק אליה, או מניתוח שנותח בו ומרגיש א"ע כעת בטוב וכאילו הנהו כאחד האנשים הבריאים. ויה"ר מלפני אבינו שבשמים שיגמלנו תמיד כל טוב בשלימות שאין אחריו כל מחסור, ויקוים בנו מקרא כתוב: ויאמר אם שמוע תשמע וגו' כל המחלה אשר שמתי במצרים לא אשים עליך כי אני ה' רופאך.
