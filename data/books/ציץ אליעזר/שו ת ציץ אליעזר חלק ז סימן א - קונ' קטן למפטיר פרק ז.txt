 שו"ת ציץ אליעזר חלק ז סימן א - קונ' קטן למפטיר פרק ז 

ענף ז' (יט) ולמען בירורה של ההלכה בשלימותה נחזור בזה להדברים שכתבתי לעיל באות א' דיש מקום לומר דמפטיר דמוספין קיל יותר ממפטיר דר"ח וחנוכה ודד' פרשיות, והבאתי יסוד לזה מדברי רבינו ישעיה המובאים בשבלי הלקט סי' ע"ט וכן בב"י סי' רפ"ב שכותב במפורש לחלק ביניהם דקריאת המוספין אינה חובה דאינו אלא תיקון הגאונים ואילו ארבע פרשיות ופרשת ר"ח שהן חובה ליום ועודפות עוד מחובת קריאת פרשת היום ע"ש. 

ולפי"ז יכלנו להכריז עוד ביותר מהב"ח ולומר דלא נמצא לשום מחבר שיכתוב כן בשם ר"ת דקטן אינו עולה למפטיר דמוספין דיו"ט, ואפילו לא בספר המנהיג, דנהי דבה' חנוכה סי' ק"נ מביא המנהיג בשם ר"ת בסתמא שכתב דפרשת חובה אין לקרא בה קטן אבל כפי שהבאנו הרי בה' שבת סי' ל"ד מביא הספר המנהיג בשם ר"ת שפירט כוונתו בפרשת חובה שהיא כגון של ר"ח וחנוכה ופרשיות ארבע והשמיט מלהזכיר גם של יו"ט, וא"כ ניתן לומר דזה שבה' חנוכה לא פירט מזה בספר המנהיג הוא מפני שסמך על זה שהמדובר שם בהדיא על כגון דא דהיינו על פרשת חובה של חנוכה, אבל גם שם כוונתו רק על פרשת חובה של כיוצא בזה דחנוכה דהיינו של ר"ח ופרשיות ארבע, כדפירט בהדיא בשמו בה' בשבת, ולאפוקי מפטיר דיו"ט דבזה יש שפיר לומר דגם ר"ת יודה שרשאין להקרות בה לקטן ולהפטיר בהיות דקריאה זאת אינה חובה דאינה נזכרת לא במשנה ולא בתוספתא ולא בתלמוד בבלי ולא בירושלמי ואינו אלא תיקון הגאונים, וכדכותב רבינו ישעיה שם. 

ועוד זאת דאפי' בריב"ש גופא נמי לא מצינו כמעט בדבריו שלא התיר העליה לקטן אפי' למוספים דיו"ט, דבסי' שכ"ו שהוא הסימן המרכזי אשר מדבר שם בנושא זה בהרחבה לא נזכר שם ממפטיר דיו"ט כלל, והשאלה היתה על כגון פרשת שקלים ושאר הפרשיות, ובכל התשובה מדבר הריב"ש בסתמא על פרשה שהיא חובת היום, ואין איפוא הוכחה גלויה דהוה סבר זה גם לענין מפטיר דיו"ט שאינה חובה הנזכרת בתלמוד. ובסי' שכ"א כתוב נמי בסתמא על ביום שיש בו שני ספרים שהמפטיר קורא במה שהוא חובה ליום, ומעתיק זה רק ממה שכתוב בספר המנהיג בשם ר"ת, והרי הבאנו דהספר המנהיג במקו"א =במקום אחר= בדבריו בא ופירש בהדיא דמה שמצא בשם ר"ת הוא על כגון של ר"ח וחנוכה ופרשיות ארבע, ואין איפוא הוכחה גלויה שהריב"ש ביאר זה דה"ה במפטיר דיו"ט. ורק בסי' ל"ה כתוב בריב"ש בנוסחת השאלה שהיא במפטיר דחובת היום כמו בראשי חדשים ופרשיות והמועדים ובתוך דברי התשובה נרמז ג"כ שמוסבת גם על מועדים בכתבו בלשון: ואפשר שמפני שנהגו למכור ההפטרה למי שמעלה בה בדמים והם קונים אותה בעד קטניהם להרגילם ואם לא יתכבדו בהפטרות המועדים לא יתנו בה פרוטה וכו'. והתשובה הזאת היא התשובה שאליה מכוון הב"י בסי' רפ"ב בכתבו שם דגם הוא ז"ל כתב בתשובה שלא רצו לו לשמוע לו, ויתכן על כן שרק בקשר למועדים לא רצו לשמוע לו מפני שסברו שעל כגון דמועדים גם ר"ת מודה דאפשר להעלות לקטן, דבין אלה שלא רצו לשמוע לו היו גם ת"ח ברמזו הריב"ש על כך בכתבו בלשון א"צ לומר אם יהיה שם איזה יודע ספר יחזיק בידם לקיים המנהג אף אם אינו גדול בחכמה. ויתכן גם שעל כן לא עמד בזה הריב"ש בתוקף דעתו לשנות מנהגם, בהיות ומפטיר דמועדים לא נזכר בתלמוד כנ"ל, ובהיות דאין הוכחה מפורשת בדברי ר"ת שסבר דה"ה דאין לקרוא לקטן גם במפטיר דמועדים. 

(כ) ועל פי הנ"ל היה מתיישב גם כמין חומר מה שדקדקנו באות י"ז בדברי התשב"ץ במה שהוא כותב בתשובה בח"ג בלשון אני משתדל דמשמע שאין זה אבל לעיכובא, ואילו בנו הרשב"ש כותב בתשובה בשמו בלשון צריך דמשמע לעיכובא, והיינו משום דבח"ג שם מדבר בסתמא מיום שמוציאים שני תורות שכלול בזה גם מועדים לכן כותב רק בלשון אני משתדל שהוא לשון הכולל את כולם, משום דבמועדים יש רק להשתדל על כך שלא יעלה קטן אבל אין זה לעיכובא. אבל בנו הרשב"ש מדבר בהדיא בהיכא שחל ר"ל בשבת דבזה הרי הקריאה במפטיר דר"ח חיובא מדינא דתלמודא לכן בזה כותב שפיר בשם אביו התשב"ץ ז"ל בלשון צריך שהוא לישנא דעיכובא. 

(כא) לעומת זאת אם נחלק כנ"ל בין ר"ח חנוכה וד' פרשיות לבין מפטיר דמוספין דיו"ט תצטמצם הראיה של הב"ח ז"ל שכותב שם בתשובה להוכיח מדברי המרדכי דמגילה שהבאנו לעיל באות א' רק לגבי קריאת מוספין דיו"ט, כי י"ל שבזה המדובר שם במרדכי, ואין משם גילוי גם לענין קריאה דחנוכה וד' פרשיות. ואמנם בב"ח שם בתשובה המדובר בעיקר להוכיח לענין עליית קטן לקריאת מוספים דיו"ט שעל זאת השאלה שם. אלא שמוסיף בדברי התשובה שם להתרחב ולבאר שסובר ללמוד על כך גם לענין ד' פרשיות, וכן מתבאר מדברי הב"ח בטור שבסוף סי' תרפ"ה וכנ"ל באותו ו' ע"ש, ולפי האמור אין ללמוד על זה מהמרדכי, ואפשר ללמוד זה רק מדברי המהרי"ל שמזכיר שם הב"ח בתשובה, וכנ"ל באות ג'. 

ומצאתי באליהו רבא או"ח סי' רפ"ב סק"ח שכותב לדחות בכזאת ראית הב"ח מהמרדכי די"ל דהמרדכי מיירי בשאר מוספים וס"ל כלא חיובא הן כמ"ש ב"י בשם ש"ל אבל ד' פרשיות ור"ח דחיובא אסור ע"ש. 

(כב) וכמו"כ להאמור מה שהב"ח בסוף דברי תשובתו שם מסיים במעשה רב וכותב: ומעיד אני עלי שאדוני אבי ומורי קנה לי ההפטרה בדמים יקרים במועד אחרון של פסח בהיותי קטן וקראתי בעצמי בספר מוספין בבית הכנסת של מהר"ש לורי' וכמדומה לי שהרב ז"ל היה אז בב"ה =בבית הכנסת= ולא מיחה בדבר עכ"ל. ולהאמור העדות היא רק לענין מפטיר דמוספים דיו"ט ואין אבל ללמוד מזה לענין מפטיר דד' פרשיות וכדומה. 

[ואגב שלא כדת הגיב על עדות זאת של הב"ח בספר משה ידבר שם בכתבו דאין מביאין ראיה מן הקטן עיי"ש, וכל גדולי האחרונים מזכירים מעדות זאת ומתחשבים בזה, עיין באמת ליעקב בפרק תורת עלית קס"ת אות כ"ח, ובשו"ת מים רבים חאו"ח סי' כ"ב, ובפרי האדמה ח"א פי"ב מה' תפלה הי"ז. ולדעתי אין בעדותו של הב"ח כי אם גילוי מילתא בעלמא ובדבר דרבנן, כי הרי כידוע דעת הרש"ל היא שלא לקרות לקטן רק לפ' זכור, ומינה הא בשאר מקומות מכש"כ במוספים דיו"ט סובר שאפשר לקרות לקטן, עיין מה שמובא בשמו בב"ח סי' תרפ"ה ובט"ז בשו"ע שם סק"ג, וכפי שכבר הבאנו קריאת המוספים ביו"ט אינה אלא מתקנת הגאונים ועכ"פ אליבא דכו"ע רק מדרבנן] עיין במרדכי מגילה שם. וכמו"כ עיין בהעמק שאלה שאילתא קס"א סק"ב מה שרצה לומר בדעת השאילתות, ושוב חזר בו בהוספה בכי"ק שנדפס בשולי הגליון בההוצאה החדשה ע"י מוסד הרב קוק עיי"ש ואכמ"ל [וא"כ העדות שמעיד הב"ח ממעשה רב שנהגו אתו בקטנותו בביהכ"נ של הרש"ל ושכמדומה שגם הרש"ל בעצמו היה נוכח בה, אינה אלא גילוי מילתא בעלמא וגם בדבר דרבנן, ובכל כה"ג קי"ל דכל קטן נאמן עיין בכתובות ד' כ"ח וברמב"ם פי"ד מה' עדות ה"ג ובשו"ע חו"מ סי' ל"ה סעי' ו' וסמ"ע ס"ק י"ז עיי"ש, ומכ"ש עדות כזאת מרבינו בעל הב"ח ז"ל. ואכמ"ל יותר]. 

גם בברכי יוסף או"ח סי' רפ"ד אות ב' כותב על הדחיה הנ"ל של האליהו רבא הנ"ל מההיא דהמרדכי, דאין באמת ראי' חותכת מדברי המרדכי ויש מקום לדוחה לדחות כן, אלא דכותב דמ"מ נראין הדברים והאמת יורה דרכו שמנהגם שקטן קרא בחובת היום ואף בד' פרשיות ע"ש. 

ולזה דנראין הדברים שהיה מנהגם שקטן קורא בחובת היום ואף בד' פרשיות, מוכח באמת גם משו"ת מהרי"ל שהבאנו אלא שמפקפק רק על פ' זכור, וכן מוכח כן מהב"י והרמ"א ויתר הפוסקים שהבאנו לעיל. 

והנה האליהו רבא שם כותב לדחות בעוד גוונא דאין ראי' מההיא דהמרדכי משום די"ל דקאי אמ"ש קודם שכבר קראו גדול אלא שהקטן מפטיר חוזר וקוראה מפני כבוד התורה ובזה גם הריב"ש מודה בתשובה שם, אבל שיקרא הקטן בלא שיקרא הגדול קודם י"ל דמודה מרדכי דאסור ע"ש. 

אלא דלזה יפה כבר השיב הברכי יוסף שם דאינו מחוור דמ"ש קודם הוא מנהג שהנהיגו רבינו אליקים ורבינו משולם והוא סיים ואין להקשות על מנהגינו והדבר ברור דמנהגו היה דהשביעי מסיים פ' השבוע והמפטיר קורא בחובת היום ואינו קורא מה שקרא הז' ובא ליישב מנהגו שהוא היפך מנהג רבינו אליקים ורבינו משום /משולם/ עיי"ש. ועיין גם לעיל באות א' מ"ש לכוון דברי המרדכי דאיכא הוכחה אפי' אם נפרש שכוונת דבריו שהיא כהא"ר.
