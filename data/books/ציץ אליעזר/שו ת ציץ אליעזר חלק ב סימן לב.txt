 שו"ת ציץ אליעזר חלק ב סימן לב 

הניתנין על מזבח החיצון שנתנן מתנה אחת. 

בזבחים ד' ל"ו ע"ב וד' ל"ז ע"א. תנו רבנן מנין לניתנין על מזבח החיצון שנתנן מתנה אחת כיפר תלמוד לומר ודם זבחיך ישפך, ובתוס' ד"ה מנין, כותבים, ונראה דאיתא אפילו כב"ה וכו', וכן השט"מ להלן בד' ל"ז ע"ב אות ג' מביא בשם המהר"פ דאיתא אפילו כב"ה. 

ועיין בספר ברכת הזבח שמיישב בזה דברי הרמב"ם (בפ"ב מה' פסולי המוקדשין ה"א) שיליף הדין הזה שאם נתנן במתנה אחת שכיפר מקרא דדם זבחיך ישפך. ומתמה עליו הכ"מ שם, דכיון דפוסק כב"ה דאף בחטאת כיפר, ולדידיה הרי מחטאת ילפינן לה וכדמסיק הגמ' בד' ל"ז ע"ב. ואמאי מייתי לה מודם זבחיך ישפך, אבל לפי התוס' הנ"ל מיושב שפיר ואף לב"ה לא נוכל מילף מחטאת ע"ש. 

אמנם עיין בלח"מ שם, שגם הוא רוצה מתחילה ליישב דברי הרמב"ם עפ"י התוס' הנ"ל, אבל דוחה לה דאי אפשר לומר כן בדברי הרמב"ם, ומוכרח למילף מחטאת דהא איצטריך קרא לדעתו ז"ל לומר דכל הניתנין בשפיכה שנתנן בזריקה יצא עיי"ש. 

(א) ולענ"ד נראה ליישב קושיא זו של הלח"מ, דהנה הרמב"ם שם בה"א כשיליף, הדין שאם נתנן מתנה אחת לבד כיפר, כותב בלשון שנאמר ודם זבחיך ישפך על מזבח ה' אלקיך שפיכת הדם על המזבח היא העיקר. ובהלכה ב' כשיליף הדין דכל הניתנין בזריקה שנתנן בשפיכה, יצא כותב שנאמר ודם זבחיך ישפך, ואינו מסיים כל הנ"ל שמסיים בהלכה א', אנו רואים דכשיליף הדין של מתנה אחת כיפר מביא גם סיומא דקרא של ודם זבחיך על מזבח ה' אלקיך וגם חוזר ומדגיש שפיכת הדם על המזבח הוא העיקר. ואילו כשיליף הדין דניתנין בזריקה שנתנן בשפיכה יצא, אינו מעתיק אלא רישא דקדא /דקרא/, והיינו ודם זבחיך ישפך. 

ונראה לי לדייק מזה דהדין של נתן מתנה אחת כיפר יליף הרמב"ם העיקר מסיפא דקרא, והיינו מדכתיב, ישפך על מזבח ה' אלקיך, ויליף מזה שהעיקר הוא שישפך על המזבח. וכן עיין בפירוש המשניות להרמב"ם על המשנה הזו דריש פרק בית שמאי, שכותב: אבל אם נתן מתנה אחת בכל קרבן וקרבן כפר לפי שנאמר ודם זבחיך ישפך על מזבח באיזה צד שיהיה. הרי שמדייק להביא סיפא דקרא דעל מזבח אע"פ שבגמ' לא מובא, והיינו מפני שיליף דין זה מהועל מזבח, וגם זהו שכותב באיזה צד שיהיה, ר"ל שילפינן מהועל מזבח שהעיקר בעינן שישפך על המזבח, ולא איכפת לן דווקא בנתינה, ויהיה באיזה צד שיהיה והעיקר שיבוא על המזבח. 

ולפי"ז הרי אייתר שפיר הרישא דקרא היינו הודם זבחיך ישפך לדרשא, ויליף מזה שפיר דכל הניתנין בזריקה שנתנן בשפיכה יצא. 

(ב) ומקור לזה יש לומר, דהיה לו להרמב"ם מסוגיית הגמ' במנחות ד' מ"ד ע"ב. דאיתא שם בסתמא: ת"ר מנין לכל הניתנין על מזבח החיצון שנתנן במתנה אחת שכיפר שנאמר ודם זבחיך ישפך על מזבח ה' אלקיך. והתוס' שם בד"ה ודם זבחיך כותבים פ' בית שמאי מעיקרא דריש מהאי קרא ולא קאי לבסוף. ומדהגמ' במנחות חזרה והעתיקה הברייתא הזו דדריש מהאי קרא, אע"ג דהסוגיא דהזבחים /דזבחים/ שם לא קאי הכי לבסוף. הוציא מזה הרמב"ם דהסוגיא דמנחות סוברת דשפיר אפשר למילך /למילף/ מהאי קרא דודם זבחיך, ואין זו סתירה להא דילפינן מהאי קרא גופא דזריקה שנתנן בשפיכה יצא, והיינו משום דדין שנתנן במתנה אחת כיפר ילפינן מסיפא דקרא והיינו מעל מזבח ה' אלקיך, ואייתר שפיר רישא דקרא למידרש לכל הניתנין בזריקה שנתנן בשפיכה יצא. וכנ"ל, ופסק כהאי סתמא דמנחות, ועיין בכללי הרמב"ם שהובא בשם הב"ח דדרכו של הרמב"ם לסמוך על פשוטו של משנה וברייתא ומימרא דאמרואי /דאמוראי/ ולא על שינויא דמשני, ובפרט כאן דמתיישב שפיר הילפותות מהאי קרא, ודוק שכאן בזבחים מביאה הגמ' רק הרישא דקרא, תלמוד לומר ודם זבחיך ישפך. ואילו הסוגיא דמנחות מביאה גם הסיפא דקרא שנאמר ודם זבחיך ישפך על מזבח ה' אלהיך. והיינו כנ"ל, דהסוגיא דמנחות סוברת דהדין של אם נתנן במתנה אחת כיפר ילפינן בעיקר מהא דכתיב על מזבח ה' אלקיך. שהעיקר הוא שישפך הדם על המזבח וכמו"ש הרמב"ם. ומיושב שפיר דברי הרמב"ם עפ"י התוס' והשט"מ הנ"ל שכותבים דאיתא שפיר אפילו כב"ה. ושגם לדידיה איצטריך למילף מודם זבחיך ישפך. 

ובזה יתיישב גם הקושיא השניה שמקשה שם הכ"מ. דכיון דאמרינן דמאן דיליף אם נתנן במתנה אחת כיפר מודם זבחיך ישפך סבר לה כרבי עקיבא דאמר ניתנין בזריקה שנתנן בשפיכה לא יצא, האיך רבינו מזכא שטרא לבי תרי. אבל לפי הנ"ל מיושב שפיר. משום דזהו רק אי גם הדין דאם נתנן מתנה אחת כיפר ילפינן ג"כ מרישא דקרא, היינו מודם זבחיך ישפך, ולכך מוכרח לסבור כר"ע, אבל אי ילפינן הדין דנתנן מתנה אחת כיפר מסיפא דקרא, דהיינו מעל מזבח יכול למיזל אפילו דלא כר"ע. ושפיר נוכל לילף מהאי קרא דניתנין בזריקה שנתנן בשפיכה דיצא. והיינו מרישא דקרא וכנ"ל. 

(ג) ומה שמקשה הלח"מ (בפ"ה מה' מעשי הקרבנות הלכה י"ז) קושיא אלימתא מהא שפוסק הרמב"ם (בפ"ח מה' חמץ ומצה ה"ז), דלא ברכת הפסח פוטרת של זבח ולא של זבח פוטרת של פסח והיינו משום דפוסק כר"ע דלית ליה כל הניתנין בזריקה שנתן בשפיכה יצא וכמו שאמרו בריש פרק בית שמאי ע"ש. 

נלענ"ד דלאו קושיא היא, דבאמת בהדין הזה דכל הניתנין בזריקה שנתן בשפיכה פוסק הרמב"ם דלא כר"ע, וכמו"ש בהדיא שם (בפ"ב מה' פסולי המוקדשין ה"ב), והא דפוסק כר"ע דלא ברכת הפסח פוטרת של זבח ולא של זבח פוטרת של פסח. הוא מטעמא אחרינא. משום דהא טעמא דר' ישמעאל דסובר, דברכת הפסח פטר את של זבח איתא בפסחים (ד' קכ"א ע"א) דהוא משום דסובר דזריקה בכלל שפיכה. ולומר דזריקה בכלל שפיכה, אפשר רק אם הדין הוא דאף לכתחילה אפשר לעשות בשפיכה, ואז שייך לומר דזריקה הוא בכלל שפיכה, אבל אם הדין הוא דלכתחילה לא יעשה כן, הרי לא שייך לומר דזריקה הוא בכלל שפיכה, דהא לכתחילה אסור לעשות כן, ומוכח דשפיכה אינה זריקה גמורה, ואי אפשר למיכלל זריקה בכלל שפיכה, ורק בדיעבד יצא, וא"כ מכיון שהגמ' בפסחים (ד' פ"ט ע"א) מסיק, דזה שאמרינן דכל הניתנין בזריקה שנתנן בשפיכה יצא הוא רק בדיעבד, אבל לכתחילה לא, וכן פוסק הרמב"ם (בפ"ג מה' קרבן פסח הלכה ט') דהניתנין בזריקה אל יתנם בשפיכה לכתחילה. א"כ הרי תו אי אפשר לומר דזריקה בכלל שפיכה מכיון דהא לכתחילה אסור לעשות כן וכנ"ל. ולכך פוסק הרמב"ם שפיר דלא ברכת הפסח פוטרת של זבח ולא של זבח פוטרת של פסח. דמכיון דלכתחילה אסור ליתנם בשפיכה תו לא נחשבת זריקה בכלל שפיכה, ולכך אינה פוטרת. אבל זאת באמת סובר הרמב"ם שפיר דכל הניתנין בזריקה שנתנן בשפיכה יצא. וכמו"ש בהדיא בפ"ב מה' פסולי המוקדשין. ואין משם שום סתירה להדין הזה. 

(ד) ולפי"ז ובצירוף עם מה שנתבאר לעיל, יש להוסיף ביותר על מה שכתבתי לעיל לתרץ הקושיא השניה של הכ"מ. וגם על עיקר דבריו של הלח"מ שכותב שם להוכיח מהסוגיא דזבחים דר"ע לית ליה הך דכל הניתנין בזריקה שנתנן בשפיכה יצא, והיינו מהא דאומרת הגמ' שם. סבר לה כר"ע דאמר לא זריקה בכלל שפיכה וכו', כלומר ולית ליה דכל הניתנין בזריקה שנתנם בשפיכה יצא. 

אבל לפימ"ש יתיישב שפיר [וזהו על דרך תירוצו של הלח"מ שם אבל בהסבר ובאופן אחר], ויתיישב גם הברייתא דיליף דנתנן במתנה אחת כיפר מהקרא דודם זבחיך, ולא תיקשי מהא דילפינן מזה דהניתנין בזריקה שנתנן בשפיכה דיצא, והיינו דיש לומר דזה שהוכרחה הגמ' לומר דר"ע סובר כן, הוא מפני שהסוגיא בכאן דזבחים סוברת דהדין שנתנן במתנה אחת כיפר ילפינן מהרישא דקרא, דהיינו מודם זבחיך ישפך. ולהכי בכדי שלא תיקשי על הברייתא הא איצטריך הודם זבחיך דהניתנין בזריקה אם נתנן בשפיכה יצא. הוצרכה הגמ' לומר דהברייתא סוברת דטעמא דר"ע דלא זו פוטרת את זו ולא זו פוטרת זו, הוא משום דלית ליה לר"ע בכלל הדין של הניתנין בזריקה שנתנן בשפיכה יצא. שלפי"ז ממילא לא תיקשי על הברייתא מההיא דכל הניתנין וכו' משום דסברא כר"ע. אבל לפי מה שיליף הרמב"ם הדין דנתנן במתנה אחת כיפר מהסיפא דקרא, והיינו מעל מזבח, ושכן מוכח מהסוגיא דמנחות וכנ"ל שלפי"ז לא תיקשי על הברייתא מהא דילפינן מהקרא הזה דניתנין בזריקה שנתנן בשפיכה יצא, מכיון שהדין הקודם הרי ילפינן מסיפא דקרא וכנ"ל, א"כ שפיר י"ל דבאמת גם ר"ע סובר הדין דכל הניתנין בזריקה שנתנן בשפיכה דיצא, והא דסובר דלא זו פוטרת את זו ולא זו פוטרת זו, הוא משום דסובר דמכיון דלכתחילה אסור לעשות כן לא יוכל להקראות זריקה בכלל שפיכה וכנ"ל, ולכן אינן פוטרות זו את זו, וסברת רבי ישמעאל היא, או משום שסובר דמכיון דבדיעבד יצא מיקרי שפיר זריקה בכלל שפיכה, או משום דסובר דבאמת אף לכתחילה מותר לעשות כן, וכמו שרצתה הגמ' בפסחים (בד' מ"ט) לומר כן מתחילה, והרמב"ם פוסק כר"ע, וגם אין שום סתירה להא דפוסק דכל הניתנין בזריקה שנתנן בשפיכה יצא. מפני שגם ר"ע יוכל לסבור כן. 

(ה) וממילא לפי"ז יש ליישב גם דברי הרמב"ם שם (בפ"ה מה' מעשה הקרבנות הלכה י"ז) שיליף מהקרא שנא' בבכור ואת דמם תזרוק על המזבח, שהוא הדין במעשר ובפסח שנותן דמן מתנה אחת כבכור, ויש לומר שפיר כמו שרצה לומר שם הלח"מ, שנאמר. דאמר כולם בשפיכה בדיעבד, אם עשה בכור ומעשר בשפיכה יצא. [וכמו"ש שם הלח"מ בעצמו עפ"י דרכו זה עיי"ש]. 

גם עיין בספר חק נתן בסוגיין שמביא מהתוספתא דזבחים בסוף פ"ו דקתני הבכור ומעשר והפסח בשפיכה אחת כנגד היסוד עיי"ש, ויש מקור לדברי הרמב"ם הללו מהתוספתא הלזו, וכן מציין התוספתא הזו כמקור לדברי הרמב"ם בספר אור שמח, גם עיין בשיטה מוקבצת /מקובצת/ בהשמטות שבסוף המס' לד' ל"ז ע"א באות ל"ג מ"ש בזה ע"ש, ועיין בירושלמי פסחים (פ"ה ה"ו) ובמראה הפנים שם עיי"ש, ומיושבים שפיר כל פסקי הרמב"ם בזה.
