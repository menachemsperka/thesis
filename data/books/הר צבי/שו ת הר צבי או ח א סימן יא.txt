 שו"ת הר צבי או"ח א סימן יא 

ברכת להתעטף בציצית אם היא דוקא בשעת עטיפה. 

על דבר הערתו בדינו של הארצות החיים (סימן ט): במי שהיה לבוש טלית עם ציצית והפקיר את טליתו וחזר וזכה בו אם חייב לברך שנית אחר שהיה כאן הפסק, דקודם שחוזר וזוכה נסתלק ממנו חיוב ציצית. וכת"ר העיר ע"ז שהברכה שהיא להתעטף לא תקנו אלא על שעת עטיפה, ואיך יברך עכשיו להתעטף על טלית שהוא כבר עטוף מכבר. 

הנה, גם לדברי כת"ר, הרי בטלית - קטן שאין מברכים להתעטף אלא על מצות ציצית, יתחייב לברך שנית אם חזר וזכה בו, וא"כ איזה סברא היא לומר שבטלית קטן חייב לברך אחרי שחזר וזכה בו ובטלית גדול לא יתחייב. ועוד אחרי שיש ברכה על מצות ציצית, למה לא יברך ברכה זו גם על טלית גדול. 

וכל זה אני אומר לשיטתו, אבל כפי שאבאר לקמן האמת היא שיכול לברך גם להתעטף. ועל עצם הערתו, לא ידעתי למה העיר כן רק על הארצות החיים ולמה לא העיר כן על דינו של המחבר בשו"ע (סימן ח סעיף ט): הלן בטליתו בלילה צריך לברך עליו בבקר (וברמ"א שם: וכן יעשה מי שלובש טליתו קודם שיאיר היום), והברכה שם היא להתעטף, שהרי מקור דין זה של המחבר הוא בשו"ת הרא"ש (כלל ב סימן יג) מההיא דמנחות (דף מג), רבי יהודא רמי תכילתא לפרזומא דאינשיה ביתיה ומברך כל צפרא להתעטף בציצית. הרי מפורש דבכה"ג שהיה שעת פטור באמצע ונסתלק ממנו חיוב המצוה, מברך שנית להתעטף אחרי שחוזר ונתחייב אף שהוא כבר עטוף. (ובכלל, הרי לדעת המחבר גם על טלית - קטן הברכה היא להתעטף). 

ויעויין ברמב"ם הלכות ברכות (פי"א ה"ה) שכתב וז"ל: העושה מצוה ולא בירך אם מצוה שעדיין עשייתה קיימת מברך אחר עשייה. כיצד, הרי שנתעטף בציצית ולא בירך תחילה חוזר ומברך אחר שנתעטף אשר קדשנו במצותיו וצונו להתעטף בציצית, עכ"ל. הנה שפתי הרמב"ם ברור מללו שאפשר לברך להתעטף אפילו אח"כ שכבר הוא לבוש בטלית לא רק בשעת עטיפה. אבל י"ל שטעם הרמב"ם הוא שמברך עכשיו על מעשה הלבישה שהיה קודם ועדיין לא בירך, וזה לא שייך בנידון דידן, שכבר בירך על הלבישה של קודם, אלא שהפקיר טליתו והיה שעת פטור באמצע, או שהיה באמצע לילה שהוא שעת פטור ונסתלק ממנו חיוב ציצית ומתחיל עתה חיוב חדש. 

ובזה יש מקום להערתו של כת"ר דאיך יברך עכשיו כשהוא עטוף כבר ברכת להתעטף, וברכה זו לא תקנו אלא על שעת עטיפה בלבד. ויעויין בתוס' יבמות (דף צ ע"ב, ד"ה כולהו). 

והנה, כעין הערתו של כת"ר שלא תקנו ברכה רק על שעת עטיפה בלבד מצינו בשו"ת הגרעק"א (סוף סימן ט) בשם הברכי יוסף לענין מזוזה. במי שיצא מדירתו לבית אחר לדור שם ויש שם מזוזה מכבר שהניח הדייר הראשון דמחויב הוא לברך על המזוזה דזהו מצוה חדשה לו בבית זה. (ובפתחי תשובה יו"ד (סימן רצא ס"ק ד) כתב שכוונת הגרעק"א היא, שיברך לדור בבית שיש בו מזוזה, דברכת לקבוע ל"ש בזה וכמ"ש המג"א (סימן יט ס"ק א), דאפשר דאם קבע מזוזה קודם שדר בתוכו, כשנכנס לדור מברך אקב"ו לדור בבית שיש בו מזוזה). ומסיים שם הגרעק"א, דה"ה ההולך מביתו על איזה ימים לכאורה הדין דכשחוזר לביתו יברך על המצוה, דהא בנתיים שלא היה בדירת ביתו לא היתה עליו חובת מזוזה ומתחיל עתה חיוב חדש וכו'. ושוב הביא שם בשם הברכ"י או"ח (סימן יט) שכתב בפשיטות בדין הראשון דשוכר בית שיש בו מזוזה דאינו מברך, דלא תקנו הברכה רק על שעת קביעת המזוזה, וממילא גם בדין השני בנוסע מעירו ואח"כ חוזר לביתו אינו מברך, וסיים דצריך עיון לדינא, עיי"ש. ולכאורה קשה על הברכ"י מהא דלן בטליתו בלילה שצריך לברך עליו בבקר להתעטף, ומהא דלבש טליתו קודם שיאיר היום, וכו'. 

ונ"ל לתרץ כל הנ"ל, בהקדם מה דאמרינן במס' מכות (דף כב) היה לבוש כלאים כל היום וכו' אמרו לו אל תלבש אל תלבש, חייב על כל או"א ואמר רב אשי עלה בגמרא לא פושט ולובש ממש, אלא אפילו שהה כדי ללבוש נמי חייב. ופירש הריטב"א בשם התוס', דאע"ג דקיי"ל דלאו שאין בו מעשה אין לוקין עליו מ"מ מחויב על כל התראה והתראה אעפ"י שאינו עושה מעשה בכל פעם, מ"מ מאחר שהיה בידו לתקן להסיר האיסור מעליו, בכל פעם שמתרים בו חשיב כלובש. וכך הוא לשון השאגת אריה (סימן לב): דחשבינן ליה כעושה מעשה בשהה כדי לפשוט וללבוש וחייב הואיל ותחילת הלבישה ע"י מעשה באה נגרר כל משך זמן הלבישה אחר תחילתה לחייב עליו בשהה כדי לפשוט וללבוש כאילו בכל שהייה ושהייה עשה מעשה. וכן אמרינן לגבי שהייה במקדש בטומאה, עיין שבועות (דף כד). ולפי"ז מסתבר שגם לענין מצוה נגרר כל משך זמן הלבישה אחר תחילתה ובכל שהיה ושהיה הוי כאילו עשה מעשה לבישה, וממילא שפיר יכול לברך להתעטף. ובזה מוסברים דברי הרמב"ם שאם נתעטף בציצית ולא בירך בתחילה, חוזר ומברך אחר שנתעטף להתעטף בציצית, משום דבכל שהיה ושהיה הוי כעושה מעשה העטיפה. וכן מיושבים בזה דברי התוס' במנחות (דף לו ע"א), במי שלובש טליתו קודם שיאיר היום שמברך אח"כ כשיאיר היום, ואע"ג דתחילת הלבישה היה בפטור, מ"מ מצינו כן לענין איסור שגם אם תחילת המעשה היה בהיתרא היא גורמת לאיסור דלאחר שעה להיות נחשב כאילו עשה מעשה בידים. ויעוין בשאגת אריה שם שהוכיח כן מהא דאיתא בנזיר (דף מ), גבי היכא דטומאה וביאה בהדדי אתיין דחייב משום טומאה ומשום ביאה ומוקי לה מר ר"א כגון שנכנס בשידה תיבה ומגדל ובא חבירו ופרע עליו את המעזיבה חייב אעפ"י שתחילת הכניסה בהיתרא, כי בא אח"כ איסורא אעפ"י שאינו עושה מעשה בשעה שבא עליו האיסור, מ"מ תחילת המעשה של היתרא גורם לאיסור שלאחר שעה להיות נחשב כעושה מעשה בידים. ועוד ראיה מההיא דשבועות בבעיא דשהייה למלקות גבי טומאת מקדש עיי"ש. ויעויין עוד בשאג"א שם שהקשה בזה על דברי תוס' יבמות (דף צ) שכתבו דמה"ט הו"ל סדין בציצית שב ואל תעשה אע"ג דמכסי בטלית דבת חיובא היא ואין בה ציצית הרי הוא עובר בידים משום דבשעת עיטוף אכתי לא מחייב עד אחר שנתעטף. והקשה ע"ז השאג"א, דמה בכך, הא מ"מ כיון דחיוב ציצית חל עליו ע"י מעשה של העיטוף אעפ"י שלא נתחייב בשעת עשיית מעשה העיטוף וכו' ע"כ. וכיון שלענין איסור אמרינן שגם אם תחילת המעשה היה בהיתרא היא גורמת לשהיה של איסור דלאחר שעה להיות נחשב כאילו עשה מעשה בידים, מסתברא שה"ה דיש לומר כן לענין מצוה, דאף כשמעשה הלבישה היתה בפטור מ"מ היא גורמת לשהית הלבישה אח"כ של חיוב להיות נחשב כאילו עשה אז מעשה הלבישה בחיוב ושפיר מברך להתעטף. ומתורץ בזה גם דינו של המחבר בלן בטליתו בלילה שצריך לברך עליו, וגם דינו של הארצות החיים לענין הפקיר טליתו וחזר וזכה בו, דנהי שבאמצע שהותו בעטיפה והלבישה היה שעת פטור ונפסק חיוב המצוה, מ"מ עצם מעשה הלבישה לא נפסק כל זמן שלא פשט את הטלית וממשיך בלבישתו ולא גרע ממעשה לבישה שהיתה בפטור דאמרינן שנגרר כל משך זמן הלבישה אחר תחילתה ובכל שהיה ושהיה שבאה אח"כ בחיוב חשיב כאילו עשה מעשה העטיפה והלבישה ושפיר מברך להתעטף. וכ"ז לא שייך אלא בשהיית גופא בלבישה דאיסורא של כלאים או בלבישה של מצוה דציצית, או בשהיית גופו בטומאה, שבזה שמשהה את גופו בלבישת מצוה או איסור חשיב כמלביש את גופו אבל לענין קביעת מזוזה דאין שם שהיית הגוף, שפיר כתב הברכ"י דלא תקנו הברכה רק על שעת קביעת המזוזה.
