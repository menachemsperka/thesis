 שו"ת שואל ונשאל חלק ב - יו"ד סימן קעג 

שאלה ראובן נתן לשמעון סך פלוני פרקמטיא לעשיית יין והתנו שבעל המעות יקח השליש ושמעון שני שלישים בהריוח ולא דיברו אם יהיה הפסד איך יתחלק ולקח עליו שט"ח (כמפיאל) על הסך שנתן לו ואחר איזה זמן שנעשה היין נתפשרו שיתן שמעון לראובן הקרן ואיזה דבר בעד הריוח ולפי דברי ראובן שחשב שמה שנתן לו שני שלישים בריוח מצד שכר טרחו ואפשר שביאר לו זה בביאור ובאו שניהם לשאול אם יש רבית במה שנתפשרו בסך מה ריוח: 

תשובה אפילו אם היו מתנים בפירוש שההפסד יתחלק באופן זה שיפסיד המתעסק רק ב' שלישי הריוח שהיה מרויח וכמ"ש מרן ז"ל ביו"ד סי' קע"ז ס' כ"ז שעי"ז הוי כאלו התנו שיקח שכר טרחו כמבואר שם מ"מ איסורא עבדי במה שכתב המתעסק לבעל המעות כמפיאל בקרן וכמ"ש מרן ז"ל שם ס' כ"ד [עי' [ל]החכ"א כלל קמ"ב סי"ג והדרכ"ת ס"ק ע"ד] ומבואר באחרונים דלאו דוקא שטר אלא ה"ה לכמפיאל הנקרא וויכסיל. והטעם הוא פשוט דהרי קי"ל דנותנים לכמפיאל דין שטר בנאמנות כמ"ש ה' ישא ברכה ז"ל בתשובותיו ע"ש. אלא דלכאורא בדיעבד שכבר עשו כן מאי דעבוד עבוד ואין להשיב. אמנם מבואר בהרשב"ש ז"ל סי' רכ"ה דאפילו אם כבר נטל יחזיר דהו"ל רבית קצוצה וסיים שם דלית דינא ולית דיינא דה"ל רבית קצוצה וכ"ש שסברת הרמב"ן היא כך ע"ש ולא ראינו מי שחולק עליו בזה להדיא. והגם דהמשד"ר למהרש"א ז"ל כתב דלדברי הרשב"א אפילו אם יש ב' עדים על התנאי הו"ל רבית קצוצה וכתב דלהש"ך ס"ק ב' בשם הב"ח אם יש עדים על זה על כרחו יקיים תנאו. ושאף אם אין עדים אם המקבל בעצמו רוצה לקיים תנאו שהתנו ביניהם ש"ד ע"ש ולזה הדעת נוטה וגו' מ"מ הרי מצדד שם דגם להש"ך רק רבית קצוצה לא הוי אבל אבק רבית מיהא הוי כיעו"ש ונמצא דבפלוגתא מתניא אי הוי רבית קצוצה או אבק רבית ובכה"ג כבר כתב הפת"ת סי' קע"ז ס"ק ח' בשם ה' ברית אברהם ז"ל יו"ד סי' מ"ט אות א' שהכריע דצריך להחזיר ע"ש ומתבאר מדבריו דצריך מדינא להחזיר ואפילו מי שאינו רוצה לצאת ידי שמים דאי בבא לצאת ידי שמים הלא בלא"ה אפילו אם אין סוברים דרבית קצוצה הוי רק לכ"ע אבק רבית מ"מ לצאת ידי שמים חייב להחזיר כמ"ש מרן ז"ל סי' קס"א ס"ב ע"ש. ועוד דכיון דלהרשב"ש פשיטא ליה מילתא דרבית קצוצה הוי ולה' משחא דרבותא רק הדעת נוטה כמו שיראה המעיין בדבריהם בכל כה"ג אמרינן אין ספק מוציא מידי ודאי ובתר הרשב"ש אזלינן וכ"ש שכן דעת הרמב"ן ז"ל וכמ"ש הרשב"ש ע"ש ועוד דכל דברי המשד"ר שם סובבים על דברי הב"ח ז"ל וכבר הוא עצמו תמה על דבריו מניה וביה והניח זה בצריך חיפוש בש"ס ובפוסקים וכיון דלא גמר המשד"ר ז"ל לימודו בזה והרשב"ש ז"ל פשיט"ל אין ספק מוציא מידי ודאי. ועוד דכבר כתב ה' דגול מרבבה דדברי הב"ח עצמו סתרי לדבריו בח"מ ששם העתיק לתשובת הרא"ש שכתב אסור ליקח הרבית ע"ש. והגם דה' דגול מרבבה עצמו כתב שם להוכיח להקל ע"ש כבר הפת"ת סי' קע"ז ס"ק ו' כתב משם מהר"ש דדברי הדגמ"ר ז"ל מושגים מדברי הרא"ש כלל פ"ח ופ"ט והעלה דאיסור גמור הוא ליקח רבית אם כתוב בשטר רק הקרן אפילו אם הלוה רוצה ליתן לו כמפורש בתשובת הרא"ש הנ"ל. ושכן ה' חמדת שלמה השיג על הדגמ"ר בזה ע"ש והדברים מוכיחים להדיא מדברי הדגמ"ר עצמו שלא היה בידו תשובת הרא"ש ולכן כתב שם דהרא"ש קאי רק על הצירוף מה שאין זה כלל בדברי הרא"ש כיעו"ש. וגם הב"ח ז"ל נראה דלא היה בידו תשובת הרא"ש דכלל פ"ח כמו שהיא בידינו רק כמו שזכר הוא בח"מ סי' קע"ו ע"ש. ובעיקר ראית הדגמ"ר מתשו' הרא"ש כלל ס"ו סי' יו"ד לאחר המחי"ר המעיין בדברי הרא"ש שם יראה דמיירי בנדוניא ובנדוניא כבר כתב מור"ם ז"ל סי' קע"ו ס' ו' דאין בשטר כזה רבית ע"ש וא"כ אין הוכחה כלל משם [אה"ב. ראיתי לידי"ן ר' יחזקאל מאזוז נר"ו שהעיר על זה דטעם מור"ם דבנדוניא ליכא רבית היינו משום דהוי כמוסיף לו חתנו על הנדוניא כיעו"ש ולא כן בנדון הרא"ש. והנה הרואה בתשובת הרא"ש יראה שלא נתבאר איך היה הנדון וכפי המתבאר קצת ששמעון נתחייב לאשת ראובן השלמת נדונייתה בשטר וכשמתה ירשו בניה כתובתה והשטר וטוען ראובן שהיה התנאי שהכסף הזה החסר להשלמת הנדוניא יעבוד בהם שמעון ויתן הריוח ודייק קצת שהתנאי הזה היה עם ראובן שכתב שכך התנה עמו וא"כ הוי דומה בדומה למ"ש מור"ם. הש"ך ס"ט] ועוד דכיון דדברי הרא"ש לפנינו להדיא דאסר על כרחך דלא דמי לבסי' ס"ו ובזולת זה כתבתי עוד איזה חילוקים מלבד החילוק של הנדוניא באופן דאם היה ביד הב"ח והדגמ"ר דברי הרא"ש הנ"ל ודאי דלא היו מקילים כלל וכידוע מכללי הפוסקים בהגלות דברי הראשונים שלא היו ביד האחרונים: 

והיה נראה לכאורא לומר דממ"ש מרן ז"ל בב"י סי' קע"ז על דברי הרמב"ם בפ"ה מה' מלוה ומשמע לי מדבריו דאה"ן שנוטל ריוח כפי מה שהתנו וגו' דדעתו כדעת הרמב"ם ז"ל דשרי ליטול וא"כ כיון דבש"ע סי' קע"ז סכ"ד קאי כשיטת הרמב"ם יכול ליטול הריוח אבל זה אינו דכבר הוא ז"ל סיים שם ורבינו אפשר וגו' וצ"ע וכיון דמספק"ל בדעת הרמב"ם א"כ מה שפסק בש"ע כדעת הרמב"ם היינו ג"כ דמה שתפרש בדעת הרמב"ם יתפרש בדעתו וכמ"ש ה' יד מלאכי ז"ל כללי הש"ע אות י"א והגם דכתב מרן ז"ל בב"י לשון משמע לי אין זו החלטה כלל ומתורת ספק מיהא לא נפיק וכמ"ש ה' הנז' ז"ל כללי שאר פוסקים אות כ"ו וכ"כ השד"ח בכללי הפוסקים סי' ט"ז אות ג' וכיון דבב"י בספק קאי מסתמא גם בש"ע הכי הוי ועוד דמדלא ציין מרן ז"ל על דברי הטור תשובת הרא"ש כלל פ"ח להראות מקום לדבריו בזה כדרכו תמיד נראה דלא היה בידו אז התשובה הנז' ואלו היתה בידו לא היה כותב גם מש' ליה דהמעיין בתשובה הנז' יראה דגם הרא"ש מפיק לה מהא גופא דשמא ימות ויליף מינה דהוי רבית דבתר שטרח אזלינן כיעו"ש וא"כ גם לדברי הרמב"ם הכי הוי וכדיליף הרא"ש מזה דתרתי איתנהו וכן מבואר בהלבוש ז"ל דתפס ב' הטעמים וכ"ן דעת הפת"ת ז"ל ס"ק ו' ושאר הפוסקים שזכר שם דסברי דגם לדעת מרן ז"ל אסור דהא קאי על דברי מרן וכתב כנ"ל. והמורם מכל זה דאין מי שסובר דמותר אלא דלהרשב"ש ולהרמב"ן הוי רבית קצוצה ולהמשד"ר דעתו נוטה ויש לצדד דהוי א"ר. וכן יש להוכיח לע"ד מדברי תשובת הרא"ש שזכר הב"ח בח"מ סי' קע"ו שאינה נמצאת בשאר ספרים כיעו"ש. אלא דאין לדחות מזה דברי הרשב"ש חדא דהוא ז"ל מסתמיך ואזיל על דברי הרמב"ן ז"ל ועוד דתשובה זו אינה נמצאת בשאר ספרים וכמ"ש הב"ח עצמו ואולי הספרים שבידינו עיקר. וא"כ הדרינן להכרעת ה' ברית אברהם שזכר הפת"ת סי' קע"ז ס"ק ח' דבכה"ג צריך להחזיר אפי' מי שאינו רוצה לצאת ידי שמים וכאמור. ומ"מ נראה דאין הב"ד כופין אותו על זה אלא דגם מדינא חייב ועי' לה' שלחנו של אברהם סי' קס"א אות דמר אמר חדא ומר אמ"ח ולא פליגי [נראה דכוונתו אמ"ש השש"א או"ג להוכיח דדעת מרן בספק רבית קצוצה אין מוציאין מיד המוחזק ע"ש והוא לכאורא נגד דברי ה' ברית אברהם שכ' דצריך להחזיר אבל לפי מ"ש ההמ"ח ז"ל דאין כופין אותו הב"ד אלא דאומרים לו דהוא חייב גם בדיני אדם להחזיר ותו לא מידי הוי שפיר מוסכמים דבריהם אהדדי ומר אמר חדא ומר אמר חדא ולא פליגי דהשש"א מיירי בכפיית הב"ד והברי"א מיירי בחיובו מדיני אדם] וכ"ז הוא אפי' אם היו מתנים להדיא על ההפסד באופן המותר. וכ"ש בנד"ד שלא התנו ונתפשרו קודם שידעו לא זה ולא זה מה שהדין נותן דההפסד משתנה מהריוח וא"כ י"ל דאדעתא דההפסד ג"כ יהיה רק שליש וכמ"ש השש"א [אות ל"ג] בשם הגו"ר ז"ל וה' משה"ר מע' [ע' אות פ'] דבעלמא ההפסד אזיל בתר ריוח ע"ש וא"כ אדעתא דהכי נתפשרו ונשאו ונתנו אדעתא דבעל המעות ואדעתא דהמתעסק שיש בעסקא זו איסור שהרי אינם נביאים דההפסד משתנה אלא דלא מצאתי כעת דין זה מבואר ומ"מ לע"ד יספיק הטעם הראשון לבדו להחזיר לו הריוח מפני שנכתב הכמפיאל. ומה שיש לצדד דכבר נתפשר ורצה והב"ח כתב דברצה שפי"ד כבר כתבתי דדברי הב"ח מושגים מד"ע ומדברי הרא"ש [וכ"כ הבינ"א כלל קמ"ב סי' יו"ד דמהרא"ש מוכח דאסור אף ברוצה]: 

ועוד דלא אמרה הב"ח אלא ברוצה גם עתה אבל בנתפשרו בטעות ועתה קצווח לא. גם מה שיש לצדד להקל ממ"ש הב"ח דכשיש עדים שרי אין מקום להקל כלל מצד דהודיע זה בעל המעות לאיזה אנשים חדא דגם להב"ח בעינן עדים ממש בשעת התנאי אבל הודעה בעלמא אינה כלום ותו דלהרשב"ש והרמב"ן לא מהני עדים וכמ"ש המשד"ר. ועוד דהחכ"א ז"ל [כלל קמ"ב אוי"ג] הוכיח מדברי הריטב"א דלא מהני עדים והוא אחד מהראשונים ואלו ראה דבריו הב"ח ודברי הרמב"ן הוה הדר ביה: 

ועתה נ"ל להוכיח מדברי הגמ' בבא מציעא דס"ח דקאמר אמר ליה מר בר אמימר לרב אשי אבא עביד הכי וכי אתו לקמיה מהימן להו. א"ל תינח היכא דאיתוה לדידיה אי שכיב ונפיל שטרא קמי יתמי מאי וגו' ע"ש ואי איתא דמכח שכתוב שטר בקרן ובפירות או בקרן אסור ואפי' בדיעבד דהו"ל קרוב לשכר ורחוק להפסד וכמ"ש הטור א"כ אפי' איתיה לדידיה נמי אסור וכ"נ דעת מרן הב"י בדעת הרמב"ם דבדיעבד שרי ולא ידעתי למה לא נסתייע מזה. וביותר קשה שהניח הדבר בצ"ע ואולי מ"ש וצ"ע אינו בעיקר הדין אלא בעיקר מה שיישב שם דעת הטור ע"ש. ומן האמור יש לתמוה על הרבנים שהזכרתי שאסרו וצי"ב עוד לעת הפנאי בעה"ו [עי' בתשובת הרא"ש כלל פ"ח סי' ז' דמבואר משם דמפ' כונת הגמ' כך דכיון דאם מת היורשים גובים בו רבית אפי' לא מת אסרי' ליה ליקח הריוח ע"ש] זהו הנלע"ד בזה. הכו"ח ביום ח"י תמוז יה"ל שנת התרצ"ה לפ"ג. ע"ה כלפון משה הכהן סילט"א:
