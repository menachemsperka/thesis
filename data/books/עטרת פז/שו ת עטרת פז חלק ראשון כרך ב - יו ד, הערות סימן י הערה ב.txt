 שו"ת עטרת פז חלק ראשון כרך ב - יו"ד, הערות סימן י הערה ב 

וראה לגאון מחנה אפרים (בהל' נדרים סימן ח) שתפס שם בפשיטות גמורה שכן הוא ג"כ דעת הרמב"ן ז"ל דלא אמרינן בצדקה אלכ"ל, שכתב שם וז"ל: והאומר ליתן דינר לעני אפי' הרמב"ן והרשב"א דכתבו דלא אמרו אמירתו לגבוה כמסירתו להדיוט אלא בהקדש בדק הבית אבל בצדקה לעניים לא וכו'. עכ"ל ע"ש. והרי דהגאון מחנה אפרים מעמיד את דעת הרמב"ן ז"ל דס"ל נמי כהרשב"א בענין זה דלא אמרינן אלכ"ל בענין צדקה. הן אמת דהגאון מחנה אפרים ז"ל לא אילפנו בינה מנין שאב מקור זה דהרמב"ן עומד בשיטת הרשב"א בענין זה, ועד עתה עדין לא הוברר לי איזה מקור כדברי הרמב"ן לענין זה. וביותר נתעורר אצלי הענין אגב מה שכתב הגאון איש מצליח בתשובותיו (כרך ג חאו"ח סוס"י ז) שהביא שם את דברי הרשב"ץ שהם סתומים באותו ענין שדן התם, וכתב ע"ז הרב דכיון שדברי הרמב"ן מפורשים באותו ענין א"כ יש לומר שגם הרשב"ץ עומד בשיטתו זאת שהרי הרשב"ץ היה מזרעו של הרמב"ן. יעו"ש. [וכבר הבאנו את דבריו אלה של הגאון איש מצליח בדברינו בשו"ת עטרת פז ח"א חאו"ח (סימן יד סוף עמוד שלג, וראה גם בדברינו במילואים שם עמוד שעא) ועמדנו לבאר איך הרמב"ן ז"ל היה קרובם של הרשב"ץ ובנו הרשב"ש. יעו"ש]. ומינה לכאורה י"ל גם איפכא דכל כה"ג כנידון דהכא דדברי הרמב"ן סתומים הם בענין זה ולעומת זה מצינו כמעט מפורש דהרשב"ץ והרשב"ש הנ"ל (רשב"ץ ח"ב סי' קנא, והרשב"ש סי' קסא) שסתמו דבריהם שם כדעת הסוברים דאמרינן אלכ"ל גם בצדקה, וכן מצאתי כיו"ב בדברי הגאון חקרי לב חיו"ד ח"ג (הל' צדקה סי' קיא) ד"ה גם בתשובת מימוני וכו' שתפס שם בפשיטות בדעת הרשב"ץ והרשב"ש דס"ל נמי כדעת הרי"ף ודעימיה דאמרינן שפיר גם בצדקה אלכ"ל מכך שהביאו בסתם את שיטת הסוברים הכי וסתמו כדבריהם. ע"ש. וממילא כיון שכך דמתבאר מדבריהם של הני תרתי צנתרי דדהבא אותו ואת בנו הרשב"ץ והרשב"ש דס"ל דאמרינן גם בצדקה אלכ"ל, מסתמא שדעת מור זקנם אדונינו הרמב"ן ז"ל נמי הכי הוי ובפרט שלעת עתה לא מצינו בדבריו להיפך מדבריהם, וא"כ מזה תגדל ותוסיף כמיהתינו ורצוננו העז לדעת ולהבין את דבריו של הגאון סבא דמשפטים המחנה אפרים ז"ל דמנין לקח מקור לדבריו אלה דהרמב"ן ס"ל דלא אמרינן אלכ"ל בצדקה. וצ"ב. 

והנה הלום נדפ"מ ספר מחנה אפרים ע"י ידידינו הגר"י סופר יצ"ו בתוספת הארותיו על ספר זה הכי קרא שמו "ציון יעקב", וראיתי שם (בהל' נדרים סימן ח) שבאמת נתעורר כיו"ב במש"כ המחנ"א דהרמב"ן ס"ל דלא אמרינן בצדקה אלכ"ל וכתב ע"ז וז"ל: גם דע דמש"כ רבינו הגאון ז"ל שדעת הרמב"ן כהרשב"א ז"ל דלא אמרו אלכ"ל אלא בהקדש בדק הבית אבל לא אמרו כן בצדקה לעניים, נעלם ממני היכן כתב כן רבינו הרמב"ן, ואדרבה רבינו ז"ל עצמו לקמן בהל' צדקה (סי' ב) כתב שהרמב"ן סובר כהרי"ף שבאמירה בעלמא נעשה נכסי עניים. ע"ש. וגדולה מזו מצאתי לגאון האדיר אור היר"ח ז"ל בספרו חקרי לב יו"ד (ח"ג סי' קיא) שבא לו בארוכה בזה והעלה שאף דעת הרשב"א כן דאף בהקדש עניים אמרינן אלכ"ל. עכ"ד. ע"ש. והרי דגם הרב הנז' עמד בתימא דמנין לרב מחנ"א דהרמב"ן ס"ל דלא אמרינן בצדקה אלכ"ל, ויתרה מזו הביא את דברי המחנ"א גופיה שכתב בהל' צדקה דהרמב"ן ס"ל דבצדקה שפיר אמרינן אלכ"ל. 

ואמנם לחומר הענין אולי היה אפשר להגיה בדברי המחנ"א (הל' נדרים סי' ח) הנ"ל במקום "אפילו הרמב"ן והרשב"א דכתבו דלא אמרו אלכ"ל אלא בהקדש ב"ה וכו'" ולגרוס אפילו דבתשובות הרמב"ן והרשב"א דכתבו וכו' והיינו דבתשובות המיוחסות להרמב"ן ז"ל לכאורה מתבאר דלא אמרינן בצדקה אלכ"ל (עי' היטב שם בסי' ערה), כלומר דנתכוין כאן המחנ"א לומר, דהרשב"א כתב כן גם בתשובות המיוחסות (סי' ערה) וגם בתשובותיו הנ"ל (סי' תקסג) דבצדקה לא אמרינן אלכ"ל, דהרי כידוע תשובות המיוחסות להרמב"ן הרשב"א כתבם וכמש"כ מרן ז"ל (בהקדמתו לבית יוסף) וראה גם בדברי הש"ך חו"מ (סי' עב ס"ק קט) ובמגן אברהם (סי' תרפט) ובדברי השד"ח בכללי הפוסקים (סימן י' אות ט) ובכללים (מערכת ח כלל מ), וכבר הארכנו בענין זה בס"ד במקום אחר. וממילא כאמור זהו כוונת הדברים דאפילו שהרשב"א בתשובות המיוחסות להרמב"ן וכן בתשובותיו דיליה כתב דלא אמרינן אלכ"ל וכו'. ואלא שלפי זה צריך להגיה במקום דכתבו דכתוב, דאם אמרינן דכתבו שוב מורה הדבר על רבים ודעת הרמב"ן שוב נכנס כבפני עצמו, והרי אנן אמרינן דאחד הם והכל הוא רק דעת הרשב"א. וק"ל. וקושטא קאי דאע"ג שיש בדבר זה דוחק שאנו עושים הגהות ושינויים בדברי הגאון מחנ"א ז"ל, אולם כבר כתב רבינו הב"י ז"ל בבית יוסף חיו"ד (סי' רכח, דף קלט ע"ב) בענין נדרים דעסיק ביה התם וז"ל: ואע"פ שפשט הלשון מורה על פירוש הריב"ש כיון שהענין אינו מתיישב לפירושו מוטב לדחוק הלשון ולכוונו עם הענין ולשון הירושלמי משונה לפעמים וכו'. ע"ש. וראה בשו"ת איש מצליח (כרך א חאו"ח סי' טו, דף סד ע"ב) שהביא לכלל זה, וראה גם במילואים בשו"ת איש מצליח כרך ג' (אות יד) שכתב שם הרב העורך יצ"ו שזהו כלל גדול בדברי רבותינו, והגאון מר בריה דרבינא נאמ"ן שליט"א הוסיף והביא (שם) עוד ציונים והוספות לכלל זה. וראה גם בשו"ת איש מצליח הנ"ל שכבר רמז על זה הרב המחבר וכתב ובני נאמ"ן יצ"ו הביא עוד מקורות לזה. ע"ש. וממילא הכ"נ י"ל דעדיף לדחוק ולומר דט"ס נפלה כאן בדברי המחנ"א, מלומר דיש סתירה כ"כ מפורשת בדבריו דפעם מביא בהל' נדרים דהרמב"ן ס"ל דלא אמרינן אלכ"ל בענין צדקה ובהל' צדקה כתב דהרמב"ן ס"ל כהרי"ף דאמרינן גם בצדקה אלכ"ל, וע"כ כאמור אמרינן דט"ס נפלה בדבריו, וכך יש לגרוס אפילו דבתשובות הרמב"ן והרשב"א דכתוב דלא אמרו אלכ"ל אלא בהקדש וכו', ולפי גירסא זו הכל יבוא על מקומו בשלום שהרי כאמור בתשובת הרמב"ן הנ"ל סי' ערה מבואר דלא אמרינן אלכ"ל ושכן הוא נמי בתשובת הרשב"א ח"א סי' תקסג. וכבר הבאנו לעיל ג"כ את דברי הב"י בחו"מ (רס"י יג) שכתב שם: ויותר נראה לי דט"ס הוא וצריך למחוק וא"ו דוהם וכו'. ע"ש. והרי דהוה קשה התם למרן ז"ל בדברי הרמב"ם ועמד להגיה, וה"נ י"ל כי היכי דלא תהיה סתירה גלויה בדברי המחנ"א, אולי ט"ס יש בדברים. הן אמת דעדין בכ"ז יש לעיין בדברי תשובת המיוחסות להרמב"ן הנ"ל בסי' ערה אם אכן מתבאר מהתם דלא אמרינן אלכ"ל בענין צדקה ולומר דאליה נתכוין המחנה אפרים, כי באמת אפשר לפרש בדברים התם דלעולם גם בענין צדקה אמרינן אלכ"ל. יעו"ש היטב. וראה נמי בשו"ת שואל ומשיב מהדורא תנינא (ח"א סי' כז, דף יח סוף עמוד ד) תחילת ד"ה ומ"ש מעלתו דאף וכו'. ע"ש. דוק ותבין. וראה גם בדברי הרשב"א גופיה בתשובות המיוחסות להרמב"ן (סי' קפג). יעו"ש. ועוד צ"ב. 

והן אמת כי הנה מדברי שו"ת בית יצחק שמעלקיס חיו"ד ח"ב (סי' פה אות ב) שהזכרנו לעיל שהביא שם את דברי הרדב"ז הנ"ל (ח"ג סי' תקכב וסי' תתב) דדעת הרשב"א יחידאה היא במאי דס"ל דלא אמרינן אלכ"ל בענין הקדש לעניים, והעיר ע"ז כאשר כתבנו דהרי לכאורה מצאנו דאיכא עוד עומדים בשיטתו, וז"ל שם: ובאמת רבים הם העומדים בשיטתו הרז"ה וגם במלחמות אין מבואר להיפך כמ"ש בתומים סי' סו. עכ"ל. ע"ש. ומתבאר עוד בדברי הבית יצחק כאן דרוצה להעמיד את דעת הרמב"ן בשיטת הרשב"א, דס"ל להרמב"ן דבצדקה לא אמרינן אמלכ"ל מכיון שהרמב"ם במלחמות סתם את דבריו. והנה לפ"ז אולי י"ל נמי גבי מה שעמדנו לעיל בדברי המחנ"א שכתב כיו"ב דהרמב"ן עומד בשיטת הרשב"א, וכתבנו דלא ידוע מנין שאב מקור לדבריו אלה, ועתה דילמא יש ליישב דהמחנ"א ס"ל נמי הכי כדכתב הבית יצחק, כי מהא דסתם הרמב"ן במלחמות שם ולא העיר בזה כלום אפשר לבאר דס"ל כהרשב"א. אלא שעדין תעמוד התימא במקומה שהרי כאמור המחנ"א גופיה (בהל' צדקה סימן ב) כתב דהרמב"ן עומד בשיטת הרי"ף דאף דצדקה אמרינן אלכ"ל. וצ"ב בענין זה.
