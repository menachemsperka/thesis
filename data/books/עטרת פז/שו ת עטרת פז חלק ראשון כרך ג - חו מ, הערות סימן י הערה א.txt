והנה מקור דבריו של מרן ז"ל בהאי דינא, הוא מדברי הרא"ש בפסקיו לבב"מ שהזכירו הטור (שם), דהרא"ש כתב בב"מ שם (פרק ט סי' כח) על מאי דאמרינן בגמ' דישראל שמכר השדה לעכו"ם שמותי משמתינן ליה עד דמקבל עליה כל אונסי דאתי ליה מחמתיה, דהוא כשיכול למכור לישראל באותן דמים שקנאו העובד כוכבים, אבל אינו מחוייב למכור לישראל בפחות. ע"ש. ובאמת כ"כ ג"כ התוס' בב"מ (שם ד"ה משמתינן) וז"ל: דוקא שישראל רוצה ליקח בדמים שנתן העכו"ם. ע"כ. וראה למרן בב"י הו"מ (שם אות ט) שהביא לעוד ראשונים דס"ל הכי. ע"ש. ולפי דרכנו דנקטינן דשווה האי הלכתא דבר מצרא, לדינא דהתו"כ דבעינן לאקדומי ישראל לעכו"ם בענין מקח וממכר, אין להקשות לפ"ז דכיצד זה שמדברי התוס' בע"ז (הנ"ל) עולה ומתבאר, דבהפסד מועט יש להקדים הישראל לעכו"ם. ומדבריהם הכא במס' בבא מציעא מתבאר דאפילו בהפסד מועט יכול להקדים העכו"ם לישראל, ורק בשוה בשוה שהישראל רוצה ליקח בדמים שנותן העכו"ם יקדים הישראל, אבל בפחות אפילו בדבר מועט לא, וכמו שביאר כן נמי הפרישה בטור (שם סק"ס). ע"ש. 

דהא לא קשיא, דכידוע בעלי התוס' רבים היו, ומר קאמר חדא ומר קאמר חדא, וכאשר כתב כבר רבינו החיד"א בס' שם הגדולים (ח"ב מער' ת אות כ) דאין להקשות מהתוס' שבמסכתא זו לתוס' שבמסכתא אחרת, אלא כאשר שני המקומות הם בשם ר"י או בשם ר"ת וכיוצא, אבל כששני המקומות הם בסתם אין להקשות, דאין מופת חותך דמחבר אחד חיבר תוספות השני מקומות. אמנם כששני המקומות הם במסכתא אחת זו היא קושיא וטעונה תירוץ, וזה פשוט. ע"ש. וכ"כ כיו"ב גם בקשר אמיץ (שבסוף שו"ת יוסף אומץ) וז"ל: אין להקשות דברי התוס' ממסכתא אחת למ"ש התוס' במסכתא אחרת דלא נודע לנו בבירור מי הוא הרב המחבר של תוס' במס' זו ומי הוא הרב המחבר התוס' של מסכתא אחרת, והגם דהרב מהרש"ל כתב דתוספות ממסכתות גדולות הוא מחבר אחד, אין מופת חותך כאשר נתברר לי הצעיר שעמדתי הרבה בענינים אלו. יעו"ש. וראה בזה עוד בדבריו בשו"ת חיים שאל (ח"א סי' כג) שגם שם נתבאר כיו"ב. ע"ש. וכיו"ב מבואר ג"כ בדברי הרב יד מלאכי (בכללי התוספות אות כ) שכתב שם, מה שמצינו בכמה מקומות שדברי התוס' סותרים זה את זה ממקום למקום, הסיבה היא שבמקום אחד הם דברי ר"ת ובמקום השני הם דברי ר"י וריב"א ויתר החולקים. ע"ש. וראה גם למהרש"א בחידושיו לגיטין (עד ע"א) שכתב שם, ואין זה תימה שהתוס' בשני מקומות, דהיינו הכא בגיטין והתם בקידושין, סותרים זה לזה. ע"ש. וביאור דבריו הוא ע"פ המבואר, דהיות ובעלי התוס' רבים היו, דזה כתב תוספות מסכת זו וזה כתב תוספות מסכת זו, שפיר איכא למימר גברא אגברא קא רמית, בתמיה. וכיו"ב כתב גם מהרי"ל בתשובותיו החדשות (סי' ריא) שכתב שם, דאין לתמוה על שינוי בתוס' נדה מההיא דיבמות, דתוס' יבמות יסד רבינו אליעזר מטוך הוא ע"פ רבינו שמשון משאנץ וכו', ובנדה מיוסד ע"פ תום' רבינו ברוך שירפנטין בעל התרומה. ע"ש. וראה גם בשו"ת מהר"י מינץ (סי' ה) שג"כ העלה הכי, דהשינויים מתוס' לתוס' אין לתמוה כ"כ ע"ז, דהתוס' ביבמות והתוס' בכתובות אפשר דלא חד גאון תיקנם. ע"ש. וכ"כ גם הגאון רבי עקיבא איגר בתשובותיו מהדו"ק (סי' קצד), והגאון רבי חיים פלאג'י בס' עיני כל חי בחי' עמ"ס מנחות (סו ע"א), ובס' כל החיים (דף כח ע"א), והגאון ר"ש קלוגר בשו"ת ובחרת בחיים (סי' קא), והגאון מהר"י אסאד בשו"ת יהודה יעלה חאו"ח (סס"י קח) ובחיו"ד (סי' קפד), והגאון ר"ש נתנזון בשו"ת שואל ומשיב מהדו"ק (ח"א סי' ז) ובמהדורא תנינא (ח"א סס"י פט) ובמהדורא רביעאה (ח"א סי' מו), וכ"כ גם האדמו"ר מסוכטשוב בשו"ת אבני נזר חאו"ח (סי' קלח אות ב), ומהרש"ם בתשובותיו ח"ב (במפתחות לסי' קמה), ובס' בדעת תורה חיו"ד (בפתיחה להל' שחיטה אות ו), וכ"כ גם בשו"ת תורת חסד מלובלין חאו"ח (סי' מג סוף אות ב) והאבה"ע (סי' יג), וכן העלה גם הגר"א אלפנדרי בשו"ת הסבא קדישא (ח"א סי' כא) ועוד. וע"ע למו"ר שליט"א בשו"ת יביע אומר ח"ד (חיו"ד סי' טו אות ב) שג"כ הביא שם כיו"ב דאין להקשות מתוס' שבמסכתא זו לתוס' שבמסכת אחרת, כי בעלי התוס' רבים היו ומר קאמר חדא ומר חדא, דכ"כ בשו"ת צפיחית בדבש (סי' כו דף מו ע"ד), ובשו"ת בנין ציון החדשות (סי' קמב), ובשו"ת עולת האיש (חיו"ד סי' י). ע"ש. וראה בזה עוד בס' ברית יעקב (סי' יב עמוד קלג) שג"כ עמד שם בענין זה, ופתה שערים בתבונה להציב ציונים רבים לרבותינו העומדים באימרה ושיטה זו דאין להקשות מדברי התוס' ובפרט ממסכתא למסכתא כיון שבעלי התוס' היו רבים ולאו כל דברי התוס' נאמרו מחכם אחד. ע"ש. 

ויש לציין כי הנה הפלפולא חריפתא בפסקי הרא"ש בב"ק (קיד ע"א, פרק י אות ס) עמד להעיר שם, דאיכא שינוי בין לשונו של הרא"ש בפסקיו לבב"ק לפסקיו בב"מ, דבב"ק כתב, ומשמתינן וכו', אומר ר"ת דדוקא אם הישראל רוצה ליתן כמו העכו"ם. ונראה דדוקא עד כדי דמיהן, אבל אם העכו"ם רוצה להוסיף מדמי שוויה ימכרנה לישראל בדמיה, כי אולי העכו"ם מוסיף דמים להיות שכן לישראל ולגרשו מנחלתו. אבל בב"מ נתב הרא"ש, והוא שיכול למכור לישראל באותן דמים שקנאה העובד כוכבים, אבל אינו חייב למכור לישראל בפחות, ואי חזינן וכו', [דמשמע אפילו שהגוי רוצה לשלם יותר מכדי דמיהן], והטור בחו"מ (סי' קעה) העתיק את דברי הרא"ש הללו דבב"מ. ובסוף כתב הפל"ח להעיר שם, כי התום' בבב"מ כתבו כדברי הרא"ש דבב"מ, ולכן קשה דהב"י בחו"מ (רס"י קנו) כתב בשם התוס' כלשון שכתב הרא"ש בב"ק שביאר דדוקא עד כדי דמיהן, אבל אם העכו"ם רוצה להוסיף מדמי שוויה ימכרנה לישראל בדמיה וכו'. ע"ש. ואיברא דלפ"ז לכאורה יש להעיר כיו"ב גם על דברי הסמ"ע (הנ"ל), שהרי על מה שהביא הטור (בסי' קעה) את לשונו של הרא"ש בבב"מ, הוסיף בדרישה (שם סק"ס) את דברי הרא"ש דבב"ק, שדוקא שהעכו"ם אינו מעלה אותה בדמים אלא שנותן עליה כפי שוויה, וכן נמי על דברי מרן בשו"ע (שם סעיף מא) שג"כ נקט כלשון הטור שהביא את דברי הרא"ש בבב"מ, הוסיף הסמ"ע (שם ס"ק עג) ולא כשהכותי מעלה בדמים וכו'. יעו"ש. ולפי מה שהעיר הפלפולא חריפתא דדברי הרא"ש בב"מ ודברי הרא"ש בב"ק, ב' דברים נפרדים הם, קשיא כיצד זה דעשאם לשפה אחת ודברים אחדים וערבינהו ותנינהו בהדי הדדי. 

אולם האמת היא, דהמעיין בדברי הדרישה (שם) יראה, שהוא מטה גם את דברי הרא"ש דפרק המקבל בב"מ לדבריו אשר כתב בגוזל ומאכיל בב"ק, שביאר שם את דברי הרא"ש שאמר בפרק המקבל, אבל אינו חייב למוכרה לישראל בפחות, דהיינו פחות מדמי שוויה, אבל לעולם היכא שהישראל נותן כדי שוויה אפילו שהגוי רוצה ליתן יותר א"צ להקדים העכו"ם. וא"כ נראה לפרש לפי דברי הדרישה דהא דהרא"ש בב"ק המשיך וכתב אחרי דברי ר"ת שאמר דדוקא אם הישראל רוצה ליתן כמו העכו"ם, ונראה דדוקא עד כדי דמיהן אבל אם העכו"ם רוצה להוסיף מדמי שוויה ימכרנה לישראל בדמיה כי אולי העכו"ם מוסיף דמים כדי להיות שכן לישראל לגרשו מנחלתו. ע"כ. דאין הדברים סותרים למש"כ בב"מ, כי באמת זה כלול במש"כ הרא"ש בב"מ, שכתב שם, ואי חזינן שמתכוין העובד כוכבים לקנות במיצר ישראל, כדי להשחית נחלת ישראל כגון בשכונת היהודים, או שיש ערמה בדבר הכל לפי ראות הדיינין. והכא אמר חדא והכא חדא ולא פליגי, שדבריו בב"ק באו להשלים לדבריו בב"מ דהיכי דידעינן שיש איזה ערמה בדבר, כגון שרואים שהנכרי מוסיף על השדה יותר מדמי שוויה, זהו סימן שיש איזה ערמה בדבר שרוצה להשחית את נחלתו של הישראל, אולם בעיקרם של דברים גם מש"כ הרא"ש בב"מ אבל אינו מחוייב למכור לישראל בפחות, היינו בפחות מדמי שוויה, אבל לא פחות ממה שהגוי רוצה לשלם יותר מדמי שוויה, רק שבב"מ סתם ולא פירש מאי חשיב ערמה, ובב"ק פירש מהו ערמה כגון שרוצה לקנות ביותר משוויה. וק"ל. כך נראה ליישב את דברי הדרישה לחומר הקושיה. וא"כ אולי גם רבינו הב"י הבין כן בדברי התוס' במש"כ בב"מ (קח ע"ב) ובב"ק (קיד ע"א) דכתבו, שדוקא שהישראל רוצה ליקח בדמים שנתן העכו"ם, דהיינו כשנותן העכו"ם כדי שוויה ולא יותר מדמי שוויה. דהא דכתב הרא"ש בב"ק על דברי ר"ת, ונראה דהיינו דוקא וכו', כלומר דגם ר"ת ס"ל הכי, והרא"ש לפרש דבריו אתא. והוא אשר הב"י בחו"מ (סי' קנו) הביא בשם התוס' והרא"ש, דכל שנותן הישראל דמים ראויים לשויה של השדה יש להקדימו, דמשמע מינא דאף שהגוי רוצה ליתן יותר ממנו. וק"ל. ועוד צ"ע. 

ושו"ר בחקרי לב חחו"מ (סי' קלט) ומצאתי בס"ד שם, בד"ה אך את זה וכו', שהביא את דברי הרא"ש דבב"ק הנ"ל שסיים בדבריו, ונראה דדוקא עד כדי דמיהן וכו' כי אולי הנכרי מוסיף דמים לגרש הישראל מנחלתו, ועמד לדייק בדבריו, דעד כאן לא קאמר הרא"ש טעמא, אלא משום דשמא רוצה להזיק לישראל לכן לא ימכור לו, אבל לעולם אה"נ אי לא מטי ליה לישראל נזק במה שימכור לנכרי לא מחייבינן שימכור לישראל ולהניח את הגוי שמוסיף על שוויה. ע"ש בדבריו. ומתורף לשונו מתבאר דס"ל שאפילו כשהגוי מוסיף על השדה יותר מדמי שוויה דיכול להקדימו לישראל היכא דליכא לחשש דהערמה, וכמתבאר מדבריו בב"מ. ומזה משמע כדברים אשר אמרנו בס"ד לעיל דאין סתירה מדברי הרא"ש בב"מ לדבריו בב"ק, רק דהוא ס"ל לאידך גיסא דבב' המקומות ס"ל להרא"ש שאין להקדים הישראל אפילו כשרוצה ליתן דמי שוויה של השדה כל שהגוי רוצה לשלם יותר אע"פ שהוא יותר מכדי שוויה. הן אמת דבסוף דבריו הביא שם החקרי לב את דברי השלטי גיבורים שכתב לבאר דאיכא פלוגתא בין התוס' והרא"ש, דלהתוס' כל שהעכו"ם נותן יותר אפילו ביותר משויה שרי למכור לו, וקיבל החק"ל את הדברים בשתיקה ולא העיר בזה כלום. ומדשתיק לכאורה ש"מ דניחא ליה. ע"ש. א"כ הדרא קושיה לדוכתא מדוע מרן ז"ל בב"י חו"מ (רס"י קנו) ערבינהו ותנינהו לדעת התוס' והרא"ש כחדא. וצ"ב.
