 שו"ת עטרת פז חלק ראשון כרך ג - חו"מ, הערות סימן ח הערה ב 

ואיברא דקיי"ל כלל גדול בדברי הרמב"ם ז"ל, דדרכו הוא לתפוס לשון התלמוד גמרא משנה וברייתא בקיצור ומה שנפרש בלשון התלמוד נפרש נמי בלשונו וכמ"ש כן מרן ז"ל בב"י יו"ד (סי' שלד ד"ה נדוהו בפניו), וז"ל שם: ואין נראים דבריו שכך דרכו של הרמב"ם להעתיק לשון הגמרא ומה שנפרש בלשון הגמרא אנו יכולים לפרש בדבריו. ע"ש. ואמנם דהכנה"ג (בכללי הרמב"ם אות כד) עמד להעיר דאיכא בדבר זה סתירה בדברי מרן, דהא בב"י יו"ד (סי' רא ד"ה ומ"ש ואף אם תטבול) כתב: אע"ג דהרמב"ם סידרא ולישנא דמתני' נקט ליכא למימר דכי היכי דדחקיה בלשינא דמתני' הכי נדחוק בפירוש דברי הרמב"ם דבתראה הוא והו"ל לפרש, ומזה מבואר היפך דבריו במש"כ בסי' שלד. ע"ש. אולם עי' בשד"ח בכללי הפוסקים (סימן ה אות א) שכתב, דנראה דאית לן למינקט עיקר כמש"כ מרן בסי' שלד מפני שכן כתבו נמי הרבה מהאחרונים כדברי מרן הללו דדרכו של הרמב"ם להעתיק לשון הגמרא ומה שנפרש בלשון הגמרא אנו יכולים לפרש בדבריו, דכ"כ הב"ח בחחו"מ (סי' רז אות יא) והש"ך והגינת ורדים והרדב"ז בתשובותיו ח"ה (סי' שא ס"ט) ועוד, וכן הוא מפורש נמי בדברי מרן גופיה כמה וכמה פעמים הן בכסף משנה (פ"י מהל' ס"ת ה"ד) וכן בהל' גרושין (פי"ג ה"א), והן בב"י מקומות הרבה או"ח (סי' ו) וביו"ד (סי' קלב קצד קצו קצח) ובאבה"ע (סי' יז) ובחו"מ (סי' ח) ועוד, ולכן אין ספק דהכי אית לן למינקט. ע"ש. [וראה בזה גם לרבינו החיד"א בשו"ת חיים שאל (ח"א סי' נב) שכתב: ידוע דהרמב"ם דייק בלישניה טובא היכא דהם דברי עצמו ולשונו, אמנם כי נקיט לישנא דמתניתין וברייתא ושמעתא אמרינן צדיק עת"ק לשון חכמים בלשון המשנה כידוע. יעו"ש]. ואולם נראה לומר בזה דעד כאן לא אמרינן דכל דהרמב"ם סתם כלשון התלמוד יש לנו לפרש בדבריו כל אשר אנו מפרשים בתלמוד, דהיינו דוקא בדבר העולה על הדעת וניתן להתפרש מאליו ואין בו דבר פירוש עמוק מחודש ונסתר, אבל כל כי הא דנידון דידן שבגמ' אמרינן "עד שיסלקם" דהפשט הפשוט הוא לסלק את היזק הנזק בכל בדרך שהיא וכגון כלב דסגי לקושרו או למוכרו לעכו"ם וכל כיו"ב, ולית מאן דמעלה בדעתו לפרש בסתם דהיינו "שיהרגם" כמו שפירש רש"י, כל כי הא בודאי דעל הרמב"ם לפרש דהאי סילוק הריגה הוי. וק"ל. ועוד צ"ע. 

ולפי מה שהבאנו לעיל לפרש דמשו"ה ס"ל דהאי יסלקנו היינו יהרגנו, משום דחזינן בגמ' דבב"ק (מו ע"א) דאמרינן התם דטעמא דר' אליעזר גבי שור המועד דאין לו שמירה אלא סכין, דאמר אביי דהוא משום הא דר' נתן מנין שלא יגדל כלב רע וכו', וממילא מהכי משמע דר' נתן ס"ל דסילוק כלב רע היינו הריגה וכי הא דבשור אין לו תקנה אלא סכין. ועי' בשד"ח בכללי הפוסקים (סי' ה אות א) שכתב לפרש דכל האי דאמרינן דהרמב"ם דרכו לסתום כלשון התלמוד וכל אשר נפרש בש"ס נפרש בדבריו, במקום שדברי הש"ס סתומים וההכרח לפרש בהם את הפירוש הזה הוא משום מה שמבואר אח"כ בגמ', א"כ היכא דהרמב"ם כתב רק דברי הש"ס הסתומים ואת דברי ההכרח לא כתב, בכה"ג שפיר יש לפרש את דברי הרמב"ם כפשוטם כיון שלא כתב ההכרח, ועי' בדבריו של השד"ח (באסיפת דינים מער' ה אות כג) שהביא לזה דוגמא מגבי הא דקאמר ר' שמעון בברייתא גבי קדשים קלים דאין טעונין מריקה ושטיפה אלא מדיחן מפני שנותן טעם, ומפרשינן ד"מדיחן" דנקט הכא היינו מגעילן, דהא קאמר מפני שנותן טעם ובזה בודאי דלא סגי הדחה, והרמב"ם נקט בהאי דינא נמי "מדיחן", וכתב השד"ח דהכא אין לומר דמאי מפרשינן בש"ס נפרש בדבריו שהרי בש"ס יש הכרח דהיינו הגעלה שהרי נקטינן התם "מפני שנותן טעם", ואולם הרמב"ם הכא לא נקט ליה הכי להאי טעמא ואין לנו את ההכרח הזה בדבריו, לכן בכה"ג שלא כתב את ההכרח י"ל דלא אמרינן דמאי דמפרשינן בש"ס נפרש בהרמב"ם. ע"ש. וממילא לכאורה גם הכא נימא הכי בנידון דידן דכיון שאין הכרח בדברי הרמב"ם דהאי סילוק היינו הריגה, שהרי להלכה גבי הסוגיא דדף מא ע"ב הרמב"ם לא פסק כדברי ר' אליעזר דאין שימור לשור אלא סכין, אלא פסק כדעת ר' יהודה דמתניתין שם, עי' בדבריו הל' נזקי ממון (פ"ז ה"א). יעו"ש. ולכן י"ל דכיון דאין את ההכרח בדברי הרמב"ם גבי הסוגיא דר' אליעזר (דדף מו ע"א) דסילוק ושמירה היינו הריגה, הכ"נ יש לפרש דבריו כפשוטם דסילוק דנקט היינו סילוק סתם. 

ברם אולי יש לדחות זאת לפי המבואר בתוס' רבינו פרץ בב"ק שם (מו ע"א) שעמד על דברי רש"י שם שכתב גבי האי דאמרינן "אלא אמר אביי היינו טעמיה דר' אליעזר כדתניא ר' נתן אומר מנין" וכו' ופרש"י: "הכי נמי האי לא ישמרנו לא יקיימנו הוא". ע"כ. והקשה ר"פ, צ"ע אמאי לא נפקא לן מ"לא תשים דמים" כמו כלב רע וסולם רעוע, וכתב ליישב: ושמא משום דשור מועד לא הוי מזומן כל כך כמו הני וגם אית ליה תקנתא טפי בשמירה, ומהאי טעמא נמי ניחא דרבנן דפליגי עליה דרבי אליעזר בשור מועד ואמרי דמותר לקיימו לא פליגי עליה דרבי נתן דכלב רע וסולם רעוע. עכ"ד. ע"ש. ומבואר עולה מדבריו אלה של רבינו פרץ דאפילו מאן דפליג על דברי ר' אליעזר גבי שור מועד ולא סבירא ליה להאי מילתא דקאמר דאין לו שמירה אלא סכין, מ"מ גבי כלב כו"ע ס"ל דאית ביה האי דינא דצריך להורגו ואין לקיימו. וממילא יוצא דלפי ר"פ אין קשר בין האי דינא דאין תקנה לשור מועד אלא סכין להאי דינא דכלב דהסילוק בו היינו הריגה, כיון דבכלב פשיטא בלא"ה דהוי ביה האי דינא דהריגה. וגם עולה מדברי רבינו פרץ דסילוק כלב בהריגה הוא דבר פשוט המובן מאליו ממאי דאמרינן שצריך לסלקו. וק"ל. ואולם באמת הנה הפנ"י בב"ק (שם) עמד נמי בהערה זו דהרר"פ מדוע רש"י לא למד דין שור מועד מקרא דלא תשים וגו', וכתב ליישב דמקרא דלא תשים וגו' לא שמעינן אלא איסורא בעלמא אבל חיובא אם קיימו בשמירה מעולה לא ידעינן דחייב, להכי בעי קרא ד"ולא ישמרנו" שחייב לשלם. ע"ש. וממילא לפי הפנ"י שפיר י"ל שוב דרק למ"ד דגבי שור אמרינן אין לו שמירה אלא סכין הוא דס"ל נמי דגבי כלב סילוקו בהריגה. וק"ל. ועוד צ"ב.
