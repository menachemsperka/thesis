 שו"ת מנחת יצחק חלק ב סימן קטז 

בנדון הפרשת תו"מ המבואר בספרי (ח"א סי' פ"ד ופ"ה), ובאם מותר להפריש מקליפי תפו"ז על הפרי. 

ב"ה יום ד' וישלח י"א כסלו תשי"ח לפ"ק מנשסתר יצ"ו. לכבוד הרה"ג וכו' מוה"ר אברהם יצחק זסלנסקי שליט"א. בעיה"ק ירושלים תובב"א, אחדשת"ה, את גי"ה קבלתי בדברי אי אלו הערות לספרי מנחת יצחק, ותודה רבה לכת"ה על אשר העיר אותי בקשר למה שכתבתי (שם סי' פ"ד פ"ה) בנוגע להפרשת תו"מ, מפירות שבאים מארץ לחו"ל, שאין לצרף הפקוח מהממונים להשגיח על הפרשת תו"מ, כי אין מפקחים במשקים במקום גדול הפירות, אלא מפקחים בכל עיר על פירות השוק, אבל לא על מה שנשלח לחו"ל, הי' קצת פקוח לפני מלחמת העולם הראשונה, אבל כל תקופת המנדט וגם עכשיו אין פקוח על התוצרת שנשלח לחו"ל, אבל יש לסמוך על שאר צדדי היתר' שכתבתי (וכת"ה הוסיף עוד נופך בזה), להקל עכ"פ בפירות הנשלחים לחו"ל, להפריש עפ"י נוסח ההפרשה מפי גאון גדול ירושלמי זצ"ל, אמנם לא כולם הסכימו לו, לפחות בשביל חו"ל זה מספיק, (כמבואר בנוסח ההפרשה שכתב לי וכמבואר בספר מעדני ארץ (ח"א) מהגרש"ז אברביך שליט"א בירושלים עיה"ק), ומטעם כי בארץ יש אפשרות להפריש מן הרע על היפה, כיון דלאבוד אזלא, אבל בחו"ל כל הפירות יפים הם, יש הפסד רב, וגם הזהירות בתרומה שלא יבואו לאוכלה היא רבה מאד, ואולי יש מקום לסדר הפרשת תו"מ מקליפי התפו"ז כפי שמובא חקירה זו בספר שו"ת ציץ אליעזר (ח"א סי' א'), וע"כ למי שרוצה להחמיר יפריש מהקליפות בלי ברכה עכת"ד. 

(א) והנה בנוגע לנוסח ההפרשה הנ"ל, אין לי הספר הנ"ל שמבואר שם כל המו"מ, ע"כ א"א לי לפלפל בזה, ולמעשה כבר נדפס בעירנו נוסח ההפרשה בלשון המדינה ובקיצור, בדרך שכתוב בספרי הנ"ל, אכן בנוגע להפריש מהקליפות התפו"ז, לכאורה עצה נכונה היא מאד, וגם בנוגע לזה אין לי הספר ציץ אליעזר הנ"ל, אבל כבר מבואר בדברי כת"ה די נימוק וטעם לזה, ואבוא בזה על סדר דברי כת"ה. 

(ב) ואקדים שכבר איתמר דבר זה בבי מדרשא, ויש שלשה שיטות בדבר, והמה בנויים על דברי הש"ס בצלף, אם הקפריסין דהיא קליפת הפרי של האביונות לכ"פ, חייב במעשר, ופליגו בזה ר"א ור"ע (במתני' דמעשרו' פ"ד מ"ו), ר"א מחייב במעש' ור"ע פוטר, והובא (במס' ברכות ל"ו) לענין ברכה, וברמב"ם (פ"ב דתרומות ה"ד) פסק כר"ע דפטור ממעשר, וכן נראה דעת הש"ע (ביו"ד סי' של"א סעי' קכ"ז) שלא הזכיר שם רק אביונות, וכן משמע ממה דס"ל (באו"ח סי' ר"ב סעי' ו') לענין ברכה, דמברך בפה"א, ומהרמב"ם נראה דס"ל גם בערלה כן (עי' בב"י יו"ד סי' רצ"ד) אף בא"י, וכן נראה ממש"כ בפי' המשניות (מעשרות שם) דהלכה כר"ע מחביריו, וכן משמע דס"ל לה"ג והרי"ף (ברכות שם), ממה דגרסו שם והלכתא זורק את האביונות וכו', אבל התוס' והרא"ש שם ס"ל, דדוקא גבי ערלה בחו"ל הלכה כר"ע, משום דהמיקל בארץ הלכה כמותו בחו"ל, וכמו דאיתא בש"ס שם מקודם, ולפי מש"כ הרבנו יונה שם גם דעת הרי"ף כן, וע"ע בד"ח ומעיו"ט על הרא"ש שם, ובמעשר י"ל דקיל, משום דמעשר אילן, או בכלל מעשר בזה"ז ל"ה רק דרבנן, עי' בפנ"י ברכות שם, במש"כ לפרש דברי הש"ס שם דמעשר דרבנן, לשיטת הרמב"ם דמעשר אילן מה"ת, ומש"כ בזה בקונטרס עוללות אפרים (מהגאון מוה"ר אפרים זלמן מרגליות דין ו'). 

(ג) ובקליפי תפו"ז, דעת הטו"ז (או"ח סי' ר"ד ס"ק ט"ו), דיש להם דין קפריסין, וע"כ לפי מה דס"ל להטור וש"ע בקפריסין דברכתו בפה"א, גם בקליפי תפו"ז כן, ולפי"ז בודאי פטור גם במעשר כמו בקפריסין, וזה שיטה א', שיטה הב' הוא המג"א (סי' ר"ב ס"ק י"ז) דפליג וס"ל דקליפי תפו"ז עדיף מקפריסין, וברכתן בפה"ע, ומטעם דהוי מגוף הפרי, ונטעי אינשי אדעתא לאוכלן בקליפיהן ע"י רקוח, ועוד מ"מ ראוי לאכלן חיים, אלא שאינם טוב כ"כ, וכ"ז ראוי להצטרף למאן דס"ל כן גם בקפריסין עייש"ה, וכ"כ (בס"ק ל"א), ועי' פמ"ג שם, שיטה הג' הוא הא"ר (סי' ר"ג סק"ג) בשם הע"ת, דס"ל דעוד גרע מקפריסין, וברכתן רק שהכל, ומטעם דלא נטעי להו אינשי אדעתא לאכלן ע"י רקוח, גם עינינו רואות שא"א לאוכלן חיים כלל, וע"כ מברך רק שהכל, וכ"פ בפמ"ג שם, וא"כ לדידי' מכ"ש דפטור ממעשר, אכן החיי"א (בנשמת אדם כלל נ"ג סי' ו') מצדיק שיטת המג"א, דעיקר נטיעתם ג"כ אדעתא דקליפיהם, שהרי מביאים ממנו אלפים ורבבות ליטרות קליפות בפעם א', ומה"ט חייב גם במעשר, עיי"ש, אולם בבאר היטב שם כתב שבמלכות ישמעאל מנהגם לברך שהכל, והיינו כדברי הא"ר, ועי' בס' עוללות אפרים (מהגאון מוהר"ז מרגלית /מרגליות/ על ה' ברכת הפירות דין ו') מה שפלפל בזה, ובהגהות חת"ס (על המג"א שם) כתב דנראה כהא"ר. 

(ד) ובכן אבוא לדברי כת"ה, מש"כ באות א' לפשיטות דקליפי תפו"ז יש להם דין קליפי אתרוג, שהרמב"ם פסק להלכה (בפי"א מה' תרומות ה"ה), עפ"י מקור מהתוספתא (ריש פ"י מתרומות) שאסורים לזרים, וא"כ ברור שהם חייבים במעשרות עכת"ד, הנה גם בנשמת אדם שם הביא לראי' מאתרוג של תרומה, דאמר (בסוכה ל"ה) מפני שמפסיד הקליפה עיי"ש, אכן ראיתי בס' חזון איש (הנדפס מחדש ה' ברכת הפירות סי' ל"ג אות ג') שהביא מהתוספתא דתרומות הנ"ל באתרוג, ומ"מ כתב כיון דמצינו בקליפי אגוזים ושקדים שאינו אוכל כדתנן (במעשר שני פ"א מ"ג), ומצינו באתרוג דהוי אוכל כנ"ל, ושאר קליפין לא שמענו, וילמוד סתום מן המפורש, וכל שאין אוכלין אותו כלל לאו אוכל הוא, ובסוף (ס"ק ב') כתב לענין קליפי תפו"ז, לתמוה על המג"א הנ"ל, דצ"ע כיון דאין מרקחין רק מיעוטן ומיעוט אנשים ורוב אזיל לאשפה, לא מצינו מקור לזה דחשיב פרי, ואין נפקותא בין המיעוט השכיח או לא שכיח, ומה שהשיג הנ"א על הא"ר כנ"ל, אין זו תשובה, דאותן שמשליכים מרובים עליהם עשר פעמים, ול"ד לקפריסין דהתם אוכלין אותן, אלא שאינו עיקר הפרי וטעמו פחות, אבל קליפי תפו"ז אין אוכלין כלל חיין, ולפ"ז הן פטורין ממעשר עכת"ד, והרי תפס גם הוא את שיטת הא"ר הנ"ל לעיקר, ועי' גם בתשו' כנף רננה (או"ח סי' י"ט), דמחלק בין קליפת תפו"ז ובין קליפת אתרוג עיי"ש, וא"כ הלכה זו אינה ברורה, ומ"מ לדאתאן לנד"ד בנוגע לחו"ל, אפשר למצוא מקום להקל מטעמים הנ"ל, זולת מה שיש לדון משום דלהלן. 

(ה) דהנה יש לדון, דאף אם הקליפה חייב במעשר, אם מותר להפריש מהקליפה על הפרי, ולכת"ה אף דפשיטא לי' באות ב', דאין שום מקום בסברא להחליט, דבזמן דאזלא לאיבוד, לחלק בין הקליפה לעצם הפרי דכל עיקר המניעה משום הרע על היפה, וכשהולך לאיבוד אין איסור בזה כמבואר (ביו"ד סי' של"א), אבל שוב חזר על משנתו באות ד', וכתב דיש מקום לומר דלהטו"ז (בסי' ר"ד) הנ"ל, דמברכין על הקליפי תפו"ז בפה"א, דאולי י"ל דלשיטתו אינם חייבים במעשרות, (הנה כי כן נראה ממה שכתבתי למעלה), או לפחות שלא לתרום מהם על הפירות, ולא מטעם מהרע על היפה, אלא דלפי"ז יש להם דין ירק, והוי ממין על שאינו מינו עכת"ד, הנה מה דמספקא לכת"ה מפורש בירושלמי (מעשרות פ"ה ה"ה) וז"ל, הקפרס והתמרות מין אחד הן מעשר מקפרס על התמרות ומן התמרות על הקפרס, אבל לא מהן על האביונות ומן האביונות עליהן, ופי' הפני משה דאף לר' אליעזר דמחייב במעשר, מ"מ אין מעשרים מהן על האביונות דהוי ממין על שאינו מינו עיי"ש, וא"כ הכי נמי בקליפי תפו"ז, דאף אם נאמר דמחייב במעשר, אבל לא עדיף מקפריסין לר' אליעזר דמחייב במעשר, ומ"מ הוי ממין על שאינו מינו, אלא דבאמת לא ידעתי טעמא מאי הוי אינו מינו, וגם י"ל דלהמג"א הנ"ל עדיף מקפריסין, אכן מצאתי בפי' מהר"י בן מלכי צדק (על המתני' דמעשרות פ"ד מ"ד) הנ"ל, דהביא דברי הירושלמי הנ"ל, ומפרש כוונתו, דהוי מדבר גמור על שאינו גמור, ומדבר שאינו גמור על הגמור, וכמפורש בירושלמי (ספ"ק דתרומות), ועי' בפיה"מ להרמב"ם שם, וא"כ פשיטא דגם בקליפי תפו"ז שייך טעם זה, אך ד"ז הוי רק מדרבנן ועיקר הקרא למין על שאינו מינו הוא דאתי כמ"ש הש"ך (ביו"ד סי' של"א סעי' נ"ד סק"פ), ובדיעבד אם תרם תרומתו תרומה כדאיתא במתני' שם, וכ"פ בש"ע שם, ובזה אחתום בכל חותמי ברכות, וגם בתודה על מה שהעתיק לי תשובתו שפרסם בקו"ת תמוז - אב תשי"ז, בקשר למה שכתבתי (בספרי סי' ס"ג וס"ד), אולם מחמת אפיסת הפנאי א"א לי לחזור עו"ה על משנה זו, ואי"ה עוד חזון למועד. והנני דושת"ה כה"י. יצחק יעקב ווייס.
