 שו"ת מנחת יצחק חלק א סימן קיא 

בענין סעודת מרעים בימי הספירה 

נשאלתי, אם מותר לעשות בימי הספירה סעודת מרעים עם כלי זמר, בלא ריקודים ומחולות, לטובת מגבית לדבר מצוה. 

(א) תשובה: הנה זה אין ספק דהמנהג שלא להתענג מכלי זמר בימי הספירה, וגדול כח המנהג, עד די"ל דנוגע באיסור דאורייתא דהוי כמו נדר, כמ"ש החתם סופר (ביו"ד סי' ק"ז) דהביא מר"ן נדרים (דף ד') דכל שמקבל עליו לעשות מצוה, הוי מן התורה, כדכתיב מפיך זו צדקה, וגם קבלת האבות מחייב את הבנים, עיי"ש, ויש להביא ראיה לדברי הח"ס מדברי הירושלמי (ר"פ מקום שנהגו), אעין דשיטין הוו במגדל, צבעייה, אתון ושאלון לר' חנינא חבריהון דרבנן, מהו מיעבד בהן עבודה, אמר להן מכיון שנהגו בהן אבותיכם באיסור, אל תשנו מנהג אבותיכם נוחי נפש, ועוד שם, בני מישא קבלו עליהו, שלא לפרוש בים הגדול, אנו מה אנו, אמר להן, מכיון שנהגו אבותיכם באיסור, אל תשנו מנהג אבותיכם נוחי נפש, ומפורש שם דהוי משום נדר עיי"ש, ועיין פסחים (נ' ע"ב) ובמג"א (סי' תס"ח) וח"י ופר"ח שם, וטו"ז (סי' תקנ"א סק"ט), והגר"א שם, הביא את הירושלמי הנ"ל, וכתב דכל זה אסור לנו משום נדר עיי"ש, והא דאיתא בירושלמי (ביכורים פ"א ה"ה) בפלוגתא דראב"י ור"י, בבת גרים, דחד כהן נסב בת גרים, אתי עובדא קמי' ר' אבהו, וארבעי' על ספסילא, אמר לי' רב ביבי, לא כן אלפן רבי, הלכה כר' יוסי, א"ל ולא כהנים נהגו סלסול בעצמן כר"א, א"ל ועל מנהג לוקין וכו', עיי"ש, ומוכח דהיכא דהוי רק מנהג בעלמא, אין לוקין עליו מכות מרדות, י"ל דשאני התם, דכהנים נהגו בפירוש רק לסלסול בעלמא, וכמ"ש בדע"ת (ה' שחיטה סי' כ"ג סק"ד) עיי"ש, אכן בפמ"ג בפתי"כ (ח"א סוף אות מ"א) כ', דעל מנהג אין לוקין מ"מ עיי"ש, מ"מ מה שנוגע לענין ביטולו עי' ריש כתובות, +בש"ה, בני היקר הרב החו"ב מוה"ר ישכר דוב שליט"א, העיר קושי' עצומה בדברי הש"ס שם (דף ג' ע"ב) דאיתא שם, ואב"א מאי מחמת האונס, כדתניא הרי שהיה פתו אפוי, וטבחו טבוח, ויינו מזוג וכו', וקשה לפי מה שפירש רש"י מקודם, על הא דאמר, דשקדו חכמים על תקנת בנות ישראל, שיהא אדם טורח בסעודה, שלשה ימים, אחד בשבת ושני בשבת וכו', ומסכנה ואילך, נהגו העם לכנוס בשלישי, ולא מיחו בידם חכמים, ובשני לא יכנוס ואם מחמת האונס מותר וכו', ופי' רש"י ובשני לא יכנוס בשביל אותה סכנה, ואין צריך לעקור בשבילה תקנת חכמים אלא יום אחד, ויהי' טורח מיהת שני ימים בצורכי סעודה, עכ"ל, אלמא דרש"י ס"ל דעיקר הטעם דבשני לא יכנוס, משום דליכא שקדו, א"כ כשפריך על זה מאי אונס, מאי משני ואב"א כדתניא וכו' כנ"ל, הרי בכה"ג דהוי פתו אפוי, וטבחו טבוח, אז כבר הי' שקדו, ושיטא דמותר לכנוס בשני, אפילו בלא אונס, כמו דמתרץ הש"ס שם (/כתובות/ ג' ע"א) על הא דפריך הא בעינן שקדו ומשני דטריח לי', אלמא היכא דטריח לי' מותר לכנוס לכ"ע, ומאי רבותא אשמיענן הש"ס בזה, עכ"ד, ותרצתי לבני נ"י עפי"מ דאיתא בח"ס בחידושים לכתובות (משנת תקס"ב), דהא דמשני הש"ס דטריח לי', הא דוקא היכא דהב"ד קבועים בכל יום, ולפי' התוספות (ד"ה שב"ד), אינם קבועים כלל, אלא כל הצריך לב"ד, מכניף לי' ג' לדינו, ומשמע מתוספות (ד"ה אי איכא וכו'), דבכה"ג ליכא ט"ב כלל, דלא יוציא קול, ולא יבואו עדים, א"כ אין כאן שקדו, רק משום כבודה בעלמא, לזה מהני טריח לי', אבל בזמן שב"ד קבועים בב' וה', ואיכא משום אקרורי דעתא ואיכא נמי בטרחות הסעודה, משום חזקה אין אדם טורח וכו', אז לא מועיל טריח לי', התקנה קבוע שיהיה טורח ג' ימים אחר השבת, ולא חלקו בין גברא לגברא, עכת"ד, ובזה מיושב ג"כ הערת בני שליט"א כמובן. (ב) ועוד הקשה בני החו"ב שליט"א, בסוגיא הנ"ל, (דף ד' ע"ה). דאיתא שם, למימרא דאבילות קילא לי' מנדה, והאמר רבי יצחק ב"ח אר"ה, כל מלאכות שהאשה עושה לבעלה נדה עושה חוץ ממזיגת הכוס וכו', והקשה השטמ"ק, תקשי לי' אבילות מאבילות, דיחוד אסרינן באבלות אף על פי שבעל, ומזיגת הכוס, שהוא יותר הרגל עברה שרינן, ותירץ דלעולם ביחוד איכא הרגל עבירה טפי, שהרי יחוד א"א אסור, ול"מ איסור מזיגת הכוס בא"א, מיהו בנדה שרינן, משום דא"א שלא להתאחד עמה, דכתיב סוגה בשושנים, עיי"ש, וכוונתו למה דאיתא בסנהדרין (ל"ז), דגלי קרא בנדה סוגה בשושנים, שאפילו כסוגה בשושנים, לא יפרצו בהם פרצות, וקשה לפי"ד, דא"כ לק"מ גם קושי' הש"ס, דלעולם אימא לך אבילות חמירא לי' מנדה, והא דאוסרים יחוד באבלות, משום דביחוד יש יותר הרגל עבירה, ממזיגת הכוס וכו', ורק יחוד גבי נדה שאני, אף דאסרו גם מזיגת הכוס, דל"ש כ"כ הרגל עבירה בזה, משום דביחוד, א"א שלא להתאחד עמה, ומשום דכתיב סוגה בשושנים כנ"ל, וכה"ג הקשה בש"ך (יו"ד סי' שמ"ב ס"ק ה') על הראב"ד, ועי' מהרש"א בסוגיין וצע"ג.+ דלא בטלו תקנתא דבתולה נשאת ליום הרביעי, אף בחשש איסור, דתבעל לטפסר תחילה, וגם לענין שמנו של גיד, דהוא רק מנהג ישראל קדושים, ואפי"ה גם בפסח נהגו בו איסור, אף דיש בזה לאו דלא תותירו, ועי' בשו"ת אמרי יושר (ח"ב סי' קפ"ו) מזה. וע"כ אף אם יהיה מנהג זה של איסור כלי זמר בימי הספירה, רק תוספת חומרא, על המבואר בש"ע ונו"כ (בסי' תצ"ג), די לנו, שלא להתירם מטעם הנ"ל. 

(ב) אמנם באמת, כבר הובא מנהג זה, בכתבי הדע"ק (הביאו בהגהות מהרש"ם הנדפס בס' אורחת חיים סי' תצ"ג) לענין נשואין בל"ג בעומר, דכ' דיש להקדים א"ע, ולעשות גם הסעודה ביום ל"ג, ושלא לעשות ריקודים וכלי זמר בליל ל"ד עיי"ש, הרי אף בנשואין, והתחיל בהיתר בל"ג בעומר, דעתו להחמיר, וכ"ש בשל רשות. ולדעתי הוי איסור כלי זמר, בכלל מה דאיתא במג"א (סי' תצ"ג סק"א), דכתב דאסור לעשות ריקודין ומחולות, בימי הספירה, ואף דלא הזכיר כלי זמר, מ"מ כלי זמר הוי כל שכן, דהוי שמחה יתירה, מריקודין ומחולות, ועי' בטור וש"ע (או"ח סי' של"ח), דיש מתירים, לומר לא"י, לנגן בכלי שיר בחופות, דאמירה לא"י במקום מצוה שרי, ואין שמחת חתן וכלה בלא כלי שיר ועיין רמ"א שם, והב"י (בסי' תק"ס) הביא זאת לענין היתר בכלי שיר, בשמחת חתן וכלה בזמה"ז, עיי"ש, ובמג"א (סי' של"ח סק"ד) הביא, בשם מהרי"ל, שפעם אחת גזר המושל שלא ינגנו בכלי שיר, וצוה להוליך חתן וכלה לעיר אחרת, לעשות שם החתונה, כדי לנגן בכלי שיר עיי"ש, ומוכח מזה, דכלי שיר הוי שמחה היותר גדולה, שאפשר לעשות לחתן וכלה. 

(ג) ובזה י"ל, מה דאיתא בספר חסידים (סי' תקל"ו), באחד שצוה לבנו, לעשות נשואין שלשים יום למיתתו, ואמר החכם לעשות, כאשר צוה, אבל כלי שיר לא ישמע בבית עיי"ש, ומשמע, הא ריקודים ומחולות שעושין לפני הכלה, כדאיתא בש"ע א"ע (סי' ס"א) ובב"ש (סי' כ"ח סעי' ה') שרי, וע"כ משום דכלי השיר הוי יותר שמחה, מרקודין ומחולות, (ואגב, צ"ע ממה דאיתא בהגהות אמרי ברוך ליו"ד (סי' שצ"ב) לענין הדין המבואר (שם סעי' ג'), מי שלא קיים פו"ר ושידך אשה, ואחר שהכין צרכי חופה מת אחי המשודכת, מותר לכונסה, ולבא עליה אחר שבעה, דיש להתיר לעשות גם ריקודין ומחולות, ולנגן בכלי זמר, ואף שמ"ש, שמנהגנו שאין שכיח סעודת חתונה בלא כלי זמר, אפשר שלא היה המנהג כן בזמן הס"ח, אמנם מ"ש, כיון שכל האיסור מחמת שמחה, הרי אנו מתירין לה שמחת נשואין, גם יש לחוש לשמחת החתן שאינו אבל עיי"ש, כ"ז שייך גם בנידון הס"ח שרק הוא אבל ולא היא וצ"ע). 

(ד) והדרן לנד"ד דבכלל איסור ריקודין ומחולות, שכתב המג"א, הוי גם כן כלי זמר, מטעם כש"כ, ועוד יש להוכיח כן, מהא דאיתא במג"א (סי' תקנ"א סק"ט), דאסור לעשות ריקודין ומחולות מי"ז בתמוז ואילך, עיי"ש, הרי לא הזכיר רק ריקודין ומחולות, ומ"מ הלכה פסוקה היא בפי גדולי האחרונים, דגם כלי זמר בלבד אסור דעל דברי מג"א אלו, הביא הפמ"ג הנדון, לענין ישראל שפרנסתו כלי זמר, אצל אינו ישראל בבית המשתה, וראה כדי פרנסה שרי ודאי, עיי"ש, ואף גם בזה מבואר בספר דה"ח (סי' קכ"ט), דדוקא מי"ז בתמוז עד ר"ח, אבל מר"ח עד אחר התענית אסור, עיי"ש, ועי' בת' מהר"מ שיק (חיו"ד סי' שס"ח) לענין אם רשאים ללמוד חכמת המוזיק בשעת אבלם, כיון דמתכוון לידע האומנות כדי להחיות נפשו ולא לשמחה, והביא ראיה מדברי הפמ"ג הנ"ל עיי"ש, וראיתי מובא בשם שו"ת זכר שמחה (סי' ס"ז) שהתיר מכח זה לתלמידים ללמוד חכמה זו, מי"ז בתמוז עד שבוע שחל בו ת"ב, דאבלות ישינה קיל מאבלות סתם עיי"ש, ובודאי כוונתו בלומד לצורך פרנסה, (ועי' בזכור לאברהם ח"ג יו"ד ה' אבל אות שמחה ובשד"ח מע' אבלות אות רט"ו), וש"מ דזולת כל הנ"ל, פשיטא להם, דאף כלי זמר לבד, בלא ריקודים ומחולות אסור, ואף דבמקור הדין לא איתא רק ריקודין ומחולות, מכל מקום פשיטא להם, דכלי זמר לא יבצר מהם, וא"כ הה"ד לענין ימי הספירה. 

(ה) ועי' בת' פנים מאירות (ח"ג סי' ל"ז) דכתב, לענין מה דמבואר ברמ"א (סי' תקנ"א), דמילה שהוא מר"ח עד ת"ב, נוהגין שהמוהל ובעל ברית ואבי הבן לובשים בגדי שבת, ונשאל אם מותרים לגלח זקנם, וע"ז הוכיח ממה דאיתא ברמ"א (סי' תצ"ה), דמי שהוא בעל ברית, או מוהל, מותר להסתפר בספירה, לכבוד המילה, ואם כן לא מיבעי', בשבוע שלא חל ת"ב בתוכו, דלא אסרי מהדין הש"ס, אלא מכח מנהגא, ואם כן כי היכי דבימי הספירה, דאסור בתספורת ממנהגא, מותרים להסתפר, הכא נמי כיון דיו"ט שלו הוא, ואין איסור תספורת, אלא מצד מנהגא, דהא (בסוף תענית) פסק כרשב"ג, דאין אסור אלא באותו שבת שחל ת"ב בלבד, ואינו אסור מר"ח אלא מצד המנהג, עכת"ד, ובזה צ"ע, דברי הפמ"ג (בסי' תקנ"א א"א סק"ו), לענין עבר ונשא, קודם ר"ח אב, די"ל ז' ימי המשתה יו"ט שלו, א"ד לדידן עכ"פ עבר, קנסינן, ואין לובש בגדי שבת עיי"ש, הרי מסופק אם עונשין אותו כיון שעבר ונשא בין המצרים, אף שהיה קודם ר"ח, ובסימן תצ"ג (סעי' א') לענין ימי הספירה, מבואר, דאם עבר וכנס אין עונשין אותו עיי"ש, ובודאי לא עדיף, מי"ז בתמוז עד ר"ח, מהתם, דהרי אף אחר ר"ח, בשבוע שלא חל ת"ב, לא עדיף מהתם, וכמ"ש הפנים מאירות, וכבר עמד בזה, בספר תורת חיים, וכתב דכוונת הפמ"ג, אם קנסינן אותו, שלא ללבוש בגדי שבת, לאו משום קנס, אלא דמה שעבר עבר, ולמה יחזיר ויעבור על דברי חז"ל, וללבוש בגדי שבת עיי"ש, וש"מ דבדבר שנאסר בתרווהו, אין סברא להחמיר יותר מר"ח אב, בשבוע שלא /חל/ בו ת"ב, ומכ"ש מי"ז בתמוז עד ר"ח, מימי הספירה, ואם כן הוא הדין לענין שמחה, שנאסר בתרווהו, כיון שמבואר דכלי זמר אסור מי"ז בתמוז, ומכ"ש אחרי ר"ח אב, אף בשבוע שלא חל בו ת"ב, אף לצורך פרנסה כנ"ל, זה בעצמו כדאי להוכיח, דגם בימי הספירה נאסר דלא גרע, מימי המצרים כנזכר, ומכ"ש לבטל המנהג בפרהסיא, עי' בסוף תשובת שבות יעקב (ח"ב סי' ל"ה) עיין שם.
