 שו"ת מנחת יצחק חלק ד סימן לז 

עוד בנוגע להמקוה בקענזעס סיטי להגה"צ הנ"ל. 

ב"ה יום ג' תצא תשכ"ג לפ"ק מנשסטר יצ"ו. שוכט"ס לכבוד ידי"נ הגה"צ המפורסם בכל קצוי ארץ במעשיו התרומיות לקו"ט ישראל וכו' כקש"ת מוה"ר חי"ל דייטש שליט"א אבד"ק העלמעץ וכעת באה"ב יצ"ו. אחדשת"ה כמשפט לאוהבי שמו, כעת לקחתי מעט פנאי, לעיין בכתבי הרבנים הגאונים, ששלח לידי כהד"ג שליט"א, בנדון המשאבה במקוה דקענזיס סיטי, שכתב הרב דשם בזה"ל: בצד שמאל של המשאבה מהודק צנור פלסתיק /פלסטיק/, והצנור ההוא נכפף ויורד לתוך הכותל, המבדיל בין המקוה והחדר המיוחד למשאבה, הצנור הפלסטיק נגמר בקרקעית המקוה בקרן זוית, אשר יש שם גומא בעומק ב' אצבעות עכ"ל, ואני הבנתי מדבריו, דלמעלה בהמשאבה מחובר צנור פלסטיק, אשר צד אחד יורד לתוך כותל המבדיל וכו', וצד אחד יורד לתוך המקוה עד קרקעיתה, אבל לא כן הבין הרב הגאון מוהר"י גרינוואלד שליט"א אבד"ק פאפא, אלא שהכוונה שהכותל של המקוה גופא מנוקבת סמוך לקרקעיתה עד לחדר המשאבה, ובנקב הזה נכנס צנור הפלסטיק לתוך המקוה, והצנור מחובר למעלה אל המשאבה, ובמה שתלוי במציאות א"א לפלפל, ורק אם הכוונה כמ"ש הרה"ג הנ"ל מה דינו. 

(א) הנה הרה"ג הנ"ל, כתב, דיש לחוש, דנהי דהצנור פלסטיק אינו דבר המקב"ט =המקבל טומאה=, אבל כיון שהיא מחוברת עם המשאבה, וקיי"ל דעץ המשמש את המתכת הכל הולך אחר עיקר הכלי, ומקבל טומאה, עי' (שבת נ"ט ע"ב), וכיון שבלא הצנור, היו המים יוצאים, הו"ל מעמיד בדבר המקב"ט, ואפילו אם הצנור מחוברת לכותלי המקוה, הרי עכ"פ תלוי בדין חקקו ולבסוף קבעו, דידועים דברי הנוב"י, שהרעיש ע"ז דבודאו' ל"מ אפילו קבעו ולבסוף חקקו עכת"ד, - והנה בנוגע למחובר לקרקע, הרי לכאורה הדין פשוט בש"ע (סי' ר"א סעי' מ"ח), דמהני, אם מחובר לקרקע, וגם נעשה לשמש את הקרקע, כמ"ש הנוב"י תנינא (סי' קל"ז), ובדגול מרבבה בגליון הש"ע שם, אך יש לחוש להחמיר, מטעם דברי הרא"ש (ה' מקואות סי' י"ב ובתשו' סי' ל"א), שכתב גם בכה"ג, דהרוצה לצאת מידי ספק וכו' עיי"ש, דש"מ דיש מקום להחמיר גם בכה"ג, ועי' בספרי מנחת יצחק (ח"א סי' קל"ט), וגם מטעם דאיתא בספרי הנ"ל (ח"ב סי' ק"ה), ואף דבכה"ג שנעשה רק להעמיד הזחילה, י"ל דקיל, דתלוי בהשיטות בפי' המשנה (מקואות פ"ה מ"ה), ובשתי דעות בש"ע (סי' נ') שם, ועי' בחת"ס (יו"ד סי' קצ"ט ורי"ו), מ"מ הרי גדולי האחרונים החמירו גם בברזא /בברז/ מטעמים הנ"ל, אף דהוי רק להעמיד הזחילה, וכמבואר בספרי (ח"ב) שם, - ומכ"ש אם מחובר רק ע"י הברגה, דל"ה חיבור, כמבואר בגי"ט (סי' ט"ו) והביאו בתשו' מהר"ם שיק (חיו"ד סי' ר"ד), ועי' בכת"ס (חיו"ד סי' צ"ז), המובאים גם בקונטרס של הרב הגאון מוהר"ב שטערן שליט"א, אבד"ק עדת ישראל במעלבארן יצ"ו. 

(ב) והנה אם הצנור אינו מחובר לקרקע, והמשאבה מחובר לקרקע, כמו שישנם באיזה מקואות, (ורק מחברים אותם למעלה במשאבה, ויורדים לתוך המים בעת הרקת המקוה, ויש לדון במציאות שלעיל, דהמקוה מנוקבת והצנור סותמו מהו), הרי יש מקום לומר דאף להמקילים במחובר לקרקע, ונעשה לשמש את הקרקע, מ"מ כיון שהצינור אינו מחובר, הוי כמו ברזא שאינה מחוברת, שכתב המהר"ם שיק שם, בהסבר פלוגתת החת"ס (שם סי' רי"ח) והגי"ט, דהחת"ס ס"ל, דברזא ל"ה כלי, משום דהוי רק כמסמרים המבואר ברמב"ם (פ"י דכלים), והגי"ט ס"ל, דחשיבא חלק מן הצנור, דע"י תרווהו נעשה כלי הראוי להשתמש, והוי כעין מספורת של פרקים, (בשבת נ"ח ע"ב) עיי"ש, והמשמעות, דמיירי דהצנור מחובר לקרקע, ומ"מ נפסל הברזא עי"ז, וכעין שכתב בגי"ט (סי' ט"ו), בכלי שקורין (קראנע /ברז/), והיא הנקרא בספר האגודה, ובפי שאר אחרונים תרנגול (צאפף האהן /ברז מגולף בצורת תרנגול/), והוא מחובר משני חלקים, החלק הא' היא כעין קנה חלול, ונקרא מינקת (בפ"ט דכלים), והחלק השני, הוא הברזא הסובבת בתוך המינקת, ע"כ אם רוצה לסתום את המקוה בכלי כזה, צריך שיאמר להאומן שיעשה אותו ע"מ לקובעו בקרקע, ויקבענו שם בקביעות גמורה דמהני וכו', וגם יזהר מאד שתהא הברזא מחוברת בחיבור גמור בתוך המינקת, שלא תהא נשמטת כלל וכו', אבל אם יכול לשומטה, אז ודאי מקבל טומאה, דכיון דאפשר להשתמש בה בפ"ע הוי כשאר כלי תשמיש של אדם וכו' עיי"ש, ובשמלה (אות צ') העיר ע"ז, דצע"ק, כיון דהמינקת מחובר בחיבור גמור, א"כ למה לי תו בחיבור הברזא, הרי הברזא זו ל"ה רק מקצת כלי, והרמב"ם כתב (בפ"ט מכלים ה"א), דכל כלי מתכת שיש לו שם לוי, אינו מתטמא בפ"ע, מפני שהוא כמקצת כלי, וי"ל, עיי"ש, - ועי' בכת"ס שם (חיו"ד סי' ק"א ד"ה הנה וד"ה אלא), דפשיט' לי' כדברי השמלה הנ"ל, וכתב כן בישוב דברי החת"ס עיי"ש, - אבל מ"מ חזינן לדעתי' של הגי"ט להחמיר בזה, והשמלה חשש לשיטתו עיי"ש, הרי דאף דאינו מקבל טומאה, רק ע"י הצטרפתו לחלק השני המחובר, מ"מ נחשב למעמיד בדבר המקב"ט, וא"כ הה"ד בנד"ד, - אבל לכאורה הא בורכא, דשם דייק הגי"ט דכיון דאפשר להשתמש בה בפ"ע, הוי כשאר כלי תשמיש אדם כנ"ל משא"כ הכא הרי בפ"ע, ל"ה מקב"ט, רק ע"י הצירוף לחלק השני המחובר לקרקע, וממ"נ, אם מחמת עצמו, ואם מחמת החלק השני, אינו מקב"ט, - אבל גם זה אינו ברור לי, אם צנור של פלסטיק בפ"ע אינו דבר המקב"ט, כאשר גליתי דעתי מכבר לפני כהד"ג שליט"א (ונדפסה בספרו הבהיר על או"ח ח"ר בדקל"ו מדה"ס), ויש לדון רק מפני שהוא כמקצת כלי, ושוב אנו באין לכל הנ"ל, - וחוץ מזה, הרי אף בפשוטי פלסטיק יש לדון דהוי מקב"ט עיי"ש. 

(ג) והנה הרה"ג מוהר"ב שטערן שליט"א בתשובתו, יצא לדון בדבר החדש, לומר דעפי"ד הש"ס ופוסקים (במס' שבת שם מ"ח ע"ב), במספורת של פרקים וכו', דשלא בשעת מלאכה, אינו מקבל טומאה רק מדרבנן, א"כ ג"כ הכא, אם הדרך לפרק הצנור מהמשאבה כדי לנקותו וכיב"ז לכ"ע, או להרמב"ם גם אם אין דרך לפרקו, אינו מקבל טומאה רק מדרבנן, כיון דבשעת טבילה, הוי לעולם שלא בשעת מלאכה, ויש להקל בקבעו ולבסוף חקקו עכת"ד, - הנה אם היינו אומרין כן, הי' מקום להקל אף באינו מחובר, לפי"מ שפסק בש"ע שם (סעי' מ"ח), דבדבר המקב"ט רק מדרבנן, אין איסור משום הויתו ע"י טומאה, ועי' בש"ך (ס"ק ק"ד) שם, ומש"כ בחת"ס שם (סי' קצ"ט ורי"ח), אבל למעשה מחמירים גם בהוי' בדהמק"ט מדרבנן, עי' בלחו"ש שם ובש"פ, - וגם לפימ"ש לעיל, ל"ה דבר פשוט, שצנור פלסטיק בפ"ע, אינו מקב"ט, ואולי גם מדאו', - ועוד בעיקר יסודו של הרה"ג הנ"ל, הגם שהמה נאמרים בהשכל ודעת תורה, מ"מ למעשה קשה לסמוך על חילוקים דקים, בדבר שלא נאמר בפירוש, דהרי מצינו ג"כ בהרבה כלים של פרקים, דלא מחלקים בין שעת מלאכה, לשלא בשעת מלאכה, ואדרבה גם כשהם מפורקים מתטמאים, וכמו דאיתא במתני' (פי"ח דכלים) וברמב"ם (פ' כ"ו דכלים), וע"כ אנו צריכים לחילוקים, וכמבואר ג"כ בדברי הרה"ג הנ"ל בשם כ"פ, ובכלי שלא מבואר בהדיא דינו, יש לדון בדימוי מילתא למילתא, וכי מפני שאנו מדמין נעשה מעשה, - וכמו שאנו חזינן דגם בברזא כנ"ל דפשיטא להו לכמה גדולים דל"ה מקב"ט, מ"מ לא רצו לסמוך למעשה כנ"ל. 

(ד) ועוד יש לדון היכא דהצנור מתחזק ע"י המשאבה, דבלא המשאבה, לא הי' מתקיים במקומו, דאז יש לדון בתרתי, א' עפי"מ דאיתא בש"ע שם (סעי' מ"ח) והיא מדברי הרא"ש, דלא יאחוז אדם דף ויעבירם עליו וכו', והיינו מפני שהאדם מקבל טומאה, ומטעם זה איתא בחת"ס שם (סי' קצ"ט ד"ה ומה), בנוגע לסילון של עץ בסופו, דצריך שיעשה באופן שיכול לעמוד בלא דבר המקבל טומאה, אבל אם א"א לעמוד, אלא ע"י אותו דבר המקב"ט, לא יועיל כלום עיי"ש, וא"כ הה"ד המונע הזחילה, אפילו לסמוך את הדבר הסותם בדבר המקב"ט אסור, וכ"כ בהדיא בספר מי השילוח שהביא בשמלה (ס"ק פ"ט), אך בשמלה שם מסיק להקל במניעת הזחילה כה"ג עיי"ש, מ"מ הרי הוא בעצמו (באות צ' ד"ה ועכ"פ), כתב לענין ברזא, דגרע ממעכב הזחילה, שהוא מעמיד כל המקוה עיי"ש, וא"כ הה"ד בנד"ד, ב' הרי לפי מציאות זו, הצנור משמש שתי תשמישין, להרקת המים, ולסתום הנקב, ואם גם לסתימת הנקב, צריכה להחזקת המשאבה, אז הוי החיבור בשעת מלאכה גם בעת הטבילה לצורך הסתימה, ונחשב כלי המקב"ט לכ"ע, - וא"כ ל"מ אם המשאבה אינה מחוברת לקרקע דפסול, אלא אף אם מחוברת באנו לדין קבעו בקרקע הנ"ל, דאף אם נעשה בכל התנאים יש להחמיר. 

סו"ד, דלא מיבעי' אם הצנור צריך להמשאבה בשעת סתימה דאסור, אלא אף בלא"ה יש להחמיר כנ"ל, ע"כ יש לעשות באופן שהבנתי שהמציאות כן, וכדברי הרה"ג מוהר"י שליט"א כנלענ"ד. (והנה הגם שהרה"ג מוהר"ב שטערן שליט"א, כתב עוד טעמים להיתר, אבל כנראה שנשמטו משלוח אצלם אלי, כי בקונטרס מהנ"ל שנשלח אלי, אינו רק קצת מאות ג' וקצת מהחתימה, וכאשר שלמעשה בלא"ה אינו כדאי שיהי' המקוה מנוקבת בקרקעיתה, מטעם שאפשר להגיע לידי זוחלין, שמה"ט אנו נמנעים כל היכא דאפשר, בהרקת המים ע"י נקב וברזא למטה, רק ע"י משאבה, וכן השכילו לעשות במקוה הנ"ל, ומה הועילו בתקנתן, אם עדיין יש נקב, ואנו צריכים לסתימת הצינור, דבודאי לא עדיף מנקב הסתום בברזא המועיל עפ"י דינא, ע"כ אסתגר בזה כעת). והנני בזה ידידו דושת"ה וחותם בכל חותמי ברכות ובברכת כוח"ט ושנה טובה ומבורכת, יצחק יעקב ווייס 

בש"ה: ושוב נשלח לי מאת כהד"ג שליט"א העתקה ממכתב הרב מקענזיס סיטי נ"י, ונהנתי מאד לראות, כי לא מעדו אשורי, כאשר הבנתי מקודם, דהמציאות היא, שאף בלא צינור הפלסטיק, לא היו המים יוצאים לחוץ, וממילא פשיטא דליכא מידי למיחש בזה כלל, וכמבואר כה"ג בתשו' ידידי הרה"ג דעדת ישראל במעלבארן שליט"א (אות ד') שהי' נחסר בההעתקה ששלח מקודם, ונספח בתוך המכתבים מאת כהד"ג שליט"א ששלח כעת.
