 שו"ת יחווה דעת חלק א סימן ה 

שאלה: מי שיש לו מכונית פרטית, האם חייב לבדוק אותה מחמץ אור ליום י"ד, לאור הנר? 

תשובה: הדבר ברור שחיוב בדיקת חמץ נוהגת בכל מקום שאפשר שהכניסו לשם חמץ, בין בבתים בין בכלים, וברור שצריכים לבדוק החמץ במקרר ובארונות המטבח שרגילים לתת שם חמץ. וכן צריכים לבדוק בילקוטי הילדים שהולכים לבית הספר, ובכיסי הבגדים, פן נשאר שם חמץ. וכמ"ש בשו"ת מהרי"ו (סימן קצ"ג). [וסמך לכך כתב בשו"ת מהרי"ו שם מהגמרא (מס' שבת יב ע"א), תניא, חייב אדם למשמש בבגדיו בערב שבת. אמר רב יוסף, הלכתא רבתי לשבתא. (כלומר, תקנה גדולה יש כאן להפרישו מאיסור שבת, שמא ישכח ויוציא מרשות לרשות בשבת). וכ"כ בשמו הגאון רבי יוסף ענגיל בספר גליוני הש"ס (שבת יב ע"א). ע"ש]. וכן פסק הרמ"א בהגה /א"ח/ (סוף סימן תל"ג), ונודע שאף על פי שאנו נוהגים כדברי המרדכי (פסחים פרק א' סימן תקל"ו), לכבד החדרים ולנקותם ניקוי יסודי מכל חשש חמץ עוד לפני ליל י"ד בניסן, עם כל זה אנו חייבים לבדוק החמץ בליל י"ד לאור הנר, וכמו שכתבו הטור והשלחן ערוך /א"ח/ (סי' תל"ג סעיף יא). וכן כתב בשו"ת תרומת הדשן (סימן קל"ג). וכן הדין בענין בדיקת חמץ במכונית הפרטית, שרגילים לקחת לתוכה מצרכי מזון של חמץ, שאף על פי שמנקים היטב את המכונית לפני ליל י"ד לבל ישאר בה שום חשש חמץ, אף על פי כן צריך לבדקה בליל י"ד לאור הנר, כמצות חז"ל. ברם, אין לחזור ולברך על ביעור חמץ כשבא לבדוק במכונית, אלא די במה שמברך בביתו על ביעור חמץ בשעת הבדיקה, ועל סמך ברכה זו יחזור ויבדוק החמץ במכוניתו. ואף על פי שיש מרחק מסויים ביניהם, אין ההליכה נחשבת הפסק. וכמבואר בדברי הבית חדש א"ח (סי' ח'), ובט"ז שם (ס"ק י"א), שאין ההליכה נחשבת להפסק, אלא אם כן הפסיק בשיחה ובהיסח הדעת בינתים. וכן הסכים בספר תפלה לדוד (סי' ל"א). וכן כתב בספר בית הלל יו"ד (סי' רפ"ט סק"א). וכן פסק החק יעקב (סי' תל"ב). וכתב המאמר מרדכי (שם סק"ד), שכן המנהג פשוט לבדוק החמץ בבתים ובחנויות בברכה אחת, אפילו אם החנות רחוקה מן הבית, הואיל וההליכה אינה נחשבת להפסק. וראה עוד בספר בית מאיר או"ח (סוף סי' תרל"ט). וכן פסק בשו"ת רב פעלים ח"ב (חאו"ח סי' ה'), מטעם שספק ברכות להקל. (וע' בשו"ת מהרש"ם ח"ג סי' רצ"א. ודו"ק). ומכאן מודעה רבה לבעלי אוטובוסים ציבוריים, וכן מטוסים של חברות תעופה ישראליות, שחייבים לבדוק החמץ מהמכוניות והמטוסים, ולנקותם מכל חשש חמץ, כדין בתי כנסיות ובתי מדרשות שהם של צבור, וראוי ונכון לקיים בהם מצות בדיקת חמץ לאור הנר בליל ארבעה עשר, כמצות חז"ל +אמנם הגאון מהר"י שפירא בחידושי פסחים (בתשובה סי' י"ב) כתב, שרוב המוני העם מקילים בבדיקת החמץ בליל י"ד, ואין מקפידים לבדוק יפה בחורין ובסדקים, כמצות חז"ל מדין התלמוד, אלא בודקים דרך העברה בעלמא כמה חדרים בשעה מועטת, ויש להם על מה שיסמוכו, מכיון שמקודם לכן הנשים רוחצות ומשפשפות ומנקות כל החדרים היטב מן החמץ. ע"כ. (וזה להיפך ממ"ש בס' דגלי הודיה והמצוה על תהלים דף ק"ט ע"ב, שהעושה כן מברך ברכה לבטלה. ואינו מוכרח). ומכל מקום נ"ל שלכתחילה צריך לבדוק היטב בליל י"ד כהלכה. וכן מוכח ממ"ש מרן הש"ע /או"ח/ (סימן תל"ג סי"א): המכבד חדרו בי"ג בניסן ומכוין לבדוק החמץ ולבערו, ונזהר לא להכניס שם עוד חמץ, אעפ"כ צריך לבדוק בליל י"ד. ומקורו מהמרדכי בשם הראב"ן. וכדבריו כתב מהר"י בן מיגאש, הובא בס' יבין שמועה להרשב"ץ (כ"ו ע"ג). וכתב הב"ח, שאפילו בדק בליל י"ג לאור הנר כראוי, אעפ"כ צריך לחזור ולבדוק בליל י"ד, שלא לחלק בין ביעור לביעור. ואכן הפר"ח כ' שאף שכן מוכח מדברי הראב"ן והש"ע, מ"מ מהירושלמי נראה שא"צ לחזור ולבדוק בליל י"ד. מכל מקום הגאון ר' חיים אבולעפייא במקראי קדש (דק"נ ע"א) הסכים לד' הב"ח. וכ"כ מהר"י עייאש במטה יהודה, ושכן דעת מרן הש"ע. (וע' חק יעקב וכסא אליהו). ועכ"פ כשבדק רק ביום י"ג, בודאי דלכ"ע צריך לחזור ולבדוק היטב בליל י"ד לאור הנר. וכן נהג הגר"א מווילנא, ע' מעשה רב (אות קע"ה), וכן הזהיר ע"ז בס' שני אליהו (די"ח ע"ג). וכן נהג להחמיר מאד הגאון בעל חתם סופר, וכמ"ש בשו"ת מכתב סופר ח"א (ס"ס =סוף סימן= י"ז). ע"ש. - והנה החק יעקב (בס"ק כ"ז) דייק מדברי הרמ"א שכתב: הכיסים ובתי ידים של בגדים צריכים בדיקה, שלפעמים נותנים שם חמץ, שמשמע שצריך לבדוק בליל י"ד, אלא שהעולם אין נזהרים בזה, ורק בשעת הביעור ביום י"ד מנערים הכיסים שלהם. ע"ש. וכ"כ הגר"ז והמקור חיים. וע"ע בשו"ת שבט הלוי (חאו"ח סימן קל"ו). ולפ"ז לכאורה יש לדון שלא להצריך בדיקה במכונית בליל י"ד לאור הנר. אבל נראה שיש לחלק בין זה לזה, כאשר עיני המעיין ישר יחזו פנימו. ואפילו אם אינו רוצה להשתמש במכונית בכל ימי הפסח, צריך לבדוק אותה מן החמץ. וכמ"ש כיו"ב בשו"ת חלק לוי (סימן קמ"א). וע' בדעת תורה (סי' תל"ג). ובשו"ת יגל יעקב (דף קנ"א ע"א). ודו"ק.+
