 שו"ת קול מבשר חלק א סימן סח 

להרב המאוה"ג מו"ה מאיר נ"י האבד"ק מילניצא. 

אחדש"ה. הן בהיותי במח"ק מדי דברי עם כ"ת בענין מה שהחמיר הנוב"י דגט צריך ב"ד של שלשה וכ"ת העיר מד' התוספות במגילה גבי מרדכי ואסתר ואני אמרתי על אתר כי כבר העירו בזה האחרונים כמדומה לי בס' גט פשוט או בתשו' חתם סופר ואחרי בואי לביתי ונחתי מעמל הדרך בינותי בספרים וראיתי כי זכרוני לא כיזב לי והוא בתשו' חת"ס אה"ע ח"ב סי' ס"ו ד"ה אך לסדר בב"ד של ג' זה הדבר צריך עיונא רבה בנוב"י תנינא חאה"ע סי' קי"ד מייתי כמה מקומות וראיות ברורות מש"ס ופוסקים שאין צריך לגרש בבי דינא. ועיין חידושי רשב"א סופ"ק דמגילה וכו' יעו"ש. ובאמת מדברי התוספות במגילה דף ט"ו ע"א ד"ה כשם אין ראי' כ"כ די"ל דחדא נקטי דצריך עדים ויתפרסם עי"ז למלכות וה"ה דהוו מצי למימר משום דצריך ב"ד, ובפרט בתוס' סנהדרין דף ע"ד ע"ב ד"ה והא כתבו בסתם שהי' ירא פן יודע הדבר למלך יעו"ש ולא נקטו בלישנא עדים כלל ואפשר לומר דבאמת כוונתם משום הא דצריך ב"ד, אולם בחדושי הרשב"א במגילה שם /סופ"ק/ הקשה על תירוץ התוס' דאכתי קשה דהי' יכול לכתוב בכתב ידו את הגט שהוא כשר מדאורייתא בלא עדים (כדאמרינן במשנה דגיטין דף פ"ו ע"א) והרשב"א הניח בקושיא וכתב שאין מתיישבין על דברי אגדה יעו"ש, ואי נימא דצריך ב"ד של שלשה לק"מ קושית הרשב"א ומזה ראי' ברורה דדעת הרשב"א דא"צ ב"ד, ולזה רמז בקצרה בתשו' הח"ס סי' ס"ו הנ"ל. ובאמת תימה על הנוב"י שבסי' קי"ד שם ד"ה והנה בירושלמי הביא בעצמו מגיטין דף פ"ו לענין כתב ידו דכשר בלא עדים ומוכח דא"צ ב"ד וע"ז כתב החילוק בין ארוסה לנשואה דגט ארוסה אין בו חיוב כתובה ול"צ ב"ד והלא הנוב"י בעצמו במהדו"ת חיו"ד סי' קס"א פלפל בדברי הרשב"א דמגילה הנ"ל הרבה יעו"ש, והרי אסתר היתה נשואה למרדכי ולא ארוסה ואפ"ה כתב הרשב"א דבכתב ידו הי' יכול לגרשה בצינעא בלא עדים ובלא ב"ד. איברא שדברי הרשב"א תמוהין קצת דלכאורה הא עכ"פ מדרבנן הגט פסול ורק אם נשאת בדיעבד הולד כשר אבל לכתחילה הרי הי' אסור למרדכי להחזירה לינשא לו כמבואר בגיטין דף פ"ו שם, וכפי הנראה נשמר מזה הרשב"א בתחלת דבריו שכתב וז"ל: אבל יש קושיא לתירוץ התוס' שיכתוב לה גט בכתב ידו שהוא כשר דאורייתא וטוב הי' לו לעשות כן דהא אסתר היכי עבדא דהשתא הכי ברצונה נבעלה ותהרג ואל תעבור ואלו הי' מגרשה בכתב ידו לא היתה עוברת על איסור עריות החמור עכ"ל דהשתא קושיתו עולה יפה דאף דלא יועיל הגט בכת"י לענין להתיר למרדכי להחזירה אבל עכ"פ יציל אותה בזה מאיסור א"א =אשת איש= שלא תעבור עליו אסתר ברצון, אך בסוף דברי הרשב"א הדר תברי' לגזיזי' וסיים ולא היתה נאסרת למרדכי יעו"ש וארכבה אתרי רכשי והרי למרדכי היתה אסורה לינשא לכתחילה גם אחר הגט שנכתב בלא עדים כנ"ל, ועיין בנוב"י מהדו"ת חיו"ד סי' קס"א שלא הרגיש בזה ועיין בתשו' שבות יעקב ח"ב חאה"ע סי' קי"ז, ועיין מהרש"א במגילה שם /סופ"ק/ שנעלמו ממנו דברי הרשב"א ועיין גם במהרש"א סנהדרין ע"ד ע"ב. אמנם נראה דהרשב"א נסתפק אם גזירה זו ותקנתא דרבנן לפסול גט בלא עדים כבר היתה בימי מרדכי שהי' מאנשי כנסת הגדולה בדורו של עזרא ונחמי' ועיין בשבת דף קכ"ג ע"ב אר"ח בימי נחמי' בן חכלי' נשנית משנה זו והרמב"ן בהשגותיו לס' המצות סוף השורש השני כתב שכל השבותין נגזרו אז יעוש"ה וגדולה מזו כתב הסמ"ג דגזירת שבות דהזאה נגזרה עוד בימי משה ולכך אמרו הטמאים למה נגרע שלא יכלו להזות שחל ז' שלהם בשבת והפנ"י רפ"ד דר"ה כתב שגם גזירה דשמא יעבירנו בתקיעה נתקנה עוד בימי משה ובמס' שבת דף ל' ע"א אמרינן משה גזר כמה גזירות ותיקן כמה תקנות ע"ש ובזה מיושבת תמיהת הרמב"ן בפירושו לתורה פ' בהר פסוק בעשור לחודש ביום הכפורים תעבירו שופר בכל ארצכם על רש"י שם שהעתיק הת"כ שלפ"ד הרמב"ן הוא רק אסמכתא יעו"ש, ולכך לא הספיק לו להרשב"א להקשות מטעם השני דהי' לו לגרשה כדי שלא תהא נאסרת למרדכי דהא י"ל דאף אם יגרשה בכתב ידו בלא עדים (דלא סגי בלא"ה שלא יתפרסם למלך) אכתי תהא נאסרת לינשא לו מדרבנן די"ל דכבר היתה גזירה זו בימיו, ולכך העמיד הרשב"א עיקר תמיהתו על הטעם הראשון דהי' לו לגרשה בכת"י דכשר מדאורייתא ויצילה בזה מאיסור עריות בהליכתה לאחשורוש ברצון ולא תעבור על איסור תורה, אך גם הטעם הראשון לא הספיק לו דהא י"ל כמ"ש בתשו' מהרי"ק שורש קס"ז דכיון שעשתה אסתר להצלת כל ישראל לא עברה בזה שום איסור אף דנאסרה לבעלה והב"ש אהע"ז סי' קע"ח סק"ד מביאו קצת בקצרה יעו"ש במהרי"ק שמביא ראיות עצומות לזה. לכך ארכבה הרשב"א אתרי רכשי וצורף גם הטעם השני כיון די"ל דאכתי בימי מרדכי לא נגזרה עדיין גזירה זו לפסול גט בלא עדים בכתב ידו משום דילמא אתי לאכשורי בכתב סופר כמ"ש רש"י בגיטין דף פ"ו ע"א ד"ה ואם ניסת או משום דהוי כאין בו זמן (ויש בו משום פרי או שמא יחפה על בת אחותו) כשיטת התוס' בגיטין דף י"ג ע"ב ד"ה כתב יעו"ש. ואכמ"ל. 

והנה במהרי"ק שהבאתי לעיל אחר שהוכיח דאסתר שפיר עבדא לעבור על איסור עריות משום הצלת כלל ישראל צירף עוד טעם אחר וז"ל: ובפרט דקרקע עולם היתה יעו"ש. ונוראות נפלאתי, דהכא גבי כאשר אבדתי אבדתי דהי' ברצון ואיך שייך קרקע עולם לנבעלת ברצון הרי חייבת מיתת ב"ד וכרת כמו האיש ורק גבי אונס מהני טעמא דקרקע עולם דל"א =דלא אמרינן= תהרג ואל תעבור. וכן תמה הנוב"י מה"ת אהע"ז סי' ק"נ ד"ה ואמנם על תשובת בית יוסף סי' י' לענין ישראלית הנבעלת לכותי יעו"ש, אולם באמת סברת הב"י נכונה כיון דבב"ק דף ל"ב ע"א אמרינן איהי לא עבדא מעשה רק לענין חטאת ומלקות ומיתת ב"ד רחמנא אחשבה להנאה כמעשה אבל גבי קנאין פוגעין בו י"ל דלא נחשב כמעשה דבזה לא גילתה התורה דהוא הלכה ואין מורין כן. אולם כ"ז למאן דלא דריש ותהי לבת לבית ואסתר לא היתה א"א רק משום ישראלית הנבעלת לנכרי, כמבואר בב"י יו"ד סי' קנ"ז ד"ה כתב הר"ן במס' יומא יעו"ש וכ"כ בעה"מ והמלחמות לרמב"ן ס"פ בן סורר, אבל מהרי"ק דאזיל למאן דדריש כאשר אבדתי אבדתי דהיתה אסתר א"א אין מקום לטעמא דקרקע עולם וצע"ג. 

מה שהקשה כ"ת בהא דאמרינן בר"ה דף כ"ט ע"ב גבי ר"ה שחל בשבת אמר רבא אי מדאורייתא היא במקדש היכי תקעינן דלפ"ד ר"ת וראב"ד דכלאים בבג"כ =בבגדי כהונה= הותרו אף שלא בשעת עבודה כיון דאישתרי לצורך עבודה א"כ ה"נ גבי תקיעה אף אי נימא דהוי מלאכה מדאורייתא מ"מ י"ל דמותר לתקוע במקדש על הקרבן בשיר, במח"כ טעה בזה טעות גדולה שהרי מבואר בתוס' זבחים דף ס"ח ע"א סוד"ה כשהוא ובמנחות דף כ"ח ע"א סוף ד"ה חצוצרות ובע"ז דף מ"ז ע"א ד"ה קרנו שעל הקרבן לא היו תוקעין בשופר אלא בחצוצרות יעו"ש וכן מבואר בגמ' ר"ה דף צ"ט ע"א ובערכין דף ג' ע"ב הכל חייבין בתקיעת שופר כהנים לויים וישראלים וכו' כהנים איצטריכא ליה ס"ד אמינא הואיל וכתיב יום תרועה יהיה לכם וכו' והני כהנים הואיל ואיתנהו בתקיעת דכל השנה דכתיב ותקעתם בחצוצרות יעו"ש וכן מבואר בגמ' ר"ה דף כ"ט ע"א ובערכין דף ג' חצוצרות והכא שופר יעו"ש הרי למדנו כשם שאין יוצאין בשופר לקרבן כך אין יוצאין בחצוצרות לתקיעת ראש השנה ועיין תוס' ר"ה דף כ"ו ע"ב ד"ה ר' יהודא שכתבו והא דבעינן שופר בראש השנה וכו' והא דאיצטריך לי' קרא דתקעו בחודש שופר משום דיליף נמי ראש השנה ממדבר והוה אמינא דבחצוצרות כמדבר עכ"ל יעו"ש במשנה דר"ה דף כ"ו ע"ב ובסוגיא דגמ' שם ועיין ברש"י שבת דף ל"ו ע"א ד"ה לשופר של ר"ה וכן במשנה פ"ב דערכין דף י' ודף י"ג קחשיב כל כלי שיר שהיו משתמשין בהם במקדש ולא חשיב שופר וכן ברמב"ם פ"ג מה' כלי המקדש ועיין מל"מ שם. והשתא כיון דשופר וחצוצרות שני מינים הם ושופר אינו ראוי לעבודה שעל הקרבן והרי גם ר"ת מודה בבגדים שאינן ראויין לעבודה דאסור ללובשן כשהן כלאים אף במקדש והוא מפרש בזה מאי דאמרינן בגמ' דערכין דף ג' נהי דאישתרו בעידן עבודה שלא בעידן עבודה לא אשתרי כמ"ש התוס' דף מ"א ע"א ד"ה תכלת ובחולין דף ק"י ע"ב ד"ה טלית ובתוס' ישנים יומא ס"ט ע"א ד"ה תיפוק לי' משום כלאים יעו"ש וא"כ אף דהותרה התקיעה לצורך עבודה בחצוצרות מ"מ בשופר שאינו ראוי לעבודה אסור לתקוע אף במקדש ושפיר פריך רבא אי מדאורייתא הוא במקדש היכי תקעינן וזה פשוט מאוד. 

ועוד נראה ברור בעיני דאף בחצוצרות הראויין לעבודה אסור לתקוע שלא בשעת עבודה אף במקדש ולא דמי כלל לכלאים בבג"כ וראיה לזה מדברי הרמ"א בש"ע או"ח סי' תקצ"ו דאחר שיצאו בשופר של ר"ה אסור לתקוע בחנם והיינו משום דיש בו שבות וכמו שהאריך להוכיח בתשו' חכם צבי סי' ל"ה אף דלצורך מצוה הותר ול"א הואיל ואישתרי אישתרי אע"ג דלד' ר"ת אפי' מדרבנן אין איסור ללבוש כלאים בכסות לילה משום הואיל ואישתרי כמבואר בתוס' חולין ק"י ע"ב גבי טלית שאולה יעוש"ה. ואמינא טעמא בזה דהא התוס' במנחות דף מ"א כתבו בטעמו של ר"ת כיון דכתיב לא תלבש שעטנז גדילים תעשה לך לומר שבמקום גדילים תלבש שעטנז משמע דכל לבישה שריא רחמנא יעו"ש וכן כלאים דבגדי כהונה ילפינן נמי מיני' דכל לבישה שרי רחמנא כמו אצל ציצית כמ"ש השאג"א בסי' כ"ט לחלק מטעם זה באיסור הנאה דבגדי כהונה דא"א למילף מכלאים דכל לבישה שרא רחמנא דמה לכלאים דקיל שכן שרי בציצית משא"כ איסור הקדש הלכך מיבעיא לן שפיר אי בגדי כהונה ניתנין ליהנות בהן ביומא דף ס"ט. וא"כ לפי"ז פשיטא דלא ילפינן איסור תקיעה בשבת ויו"ט מכלאים דנימא הואיל ואישתרי אישתרי כמובן. וגם לפמ"ש הרא"ש בהלכות ציצית סוף סימן ב' בשם ר"ת דלבישה הותרה בבג"כ בכלאים אף שלא בשעת עבודה לפי שלא ניתנה תורה למלאכי השרת דכשיגמור עבודתו יפשיטן מיד ולא ישתהו עליו כלל הלכך כשהתירה תורה לבישה לגמרי הותרה אפילו שלא בשעת עבודה ולכך בהצעה תחת ראשו דהוא שלא כדרך לבישה דל"ש האי טעמא אסור אף במקדש כדאמרינן ביומא ס"ט דמניחין אותן כנגד ראשיהן ולא תחת ראשיהן ממש משום דמתהני מכלאים יעו"ש, וכ"כ התוס' ישנים ביומא שם /דף ס"ט/ והתוס' במנחות מ"א וחולין ק"י לחלק בין הצעה ללבישה דגבי הצעה מודה ר"ת דאסור. וא"כ גבי תקיעה דלא שייך האי טעמא דלא ניתנה תורה למלאכי השרת דהא אחר גמר עבודת הקרבן יוכל ברגע אחד להפסיק לתקוע וכן אחר גמר מצות תקיעת שופר של ראש השנה לפ"ד הרמ"א או"ח סי' תקצ"ו ותשו' ח"צ הנ"ל וא"צ שהייה כמו בהפשטת בגדי כהונה כמ"ש הרא"ש הנ"ל ולכך אם חוזר ונוטל את החצוצרות שלא בשעת עבודה (או את השופר שלא לצורך מצוה דיש בו משום שבות עכ"פ) עושה איסור ול"א בזה הואיל ואישתרי אישתרי. וזה ברור מאוד וסברא זו אמרתי לכת"ה בע"פ על אתר בקצרה אך כעת הרחבתי הדברים בהמתקת והסברת הענין. 

ובהיותי בענין זה אזכיר מילתא דתמיהא לי מסוגיא דע"ז דף מ"ז שהבאתי לעיל המשתחוה לבהמה צמרה מהו לתכלת קרני' מהו לחצוצרות על דברי הלח"מ פ"ח מה' לולב ה"א שכתב דל"ש מהב"ע =מצוה הבאה בעבירה= רק במידי דדמי לקרבן והא בגמ' ע"ז שם /דף מ"ז/ פריך תכלת דמאי אי לציצית היינו בעיא דריש לקיש יעוש"ה ודו"ק. ובחידושי הארכתי בזה ואכ"מ. 

מה שהעיר כ"ת דגזירה שמא יתקן כלי שיר אי שייך במקדש תליא בפלוגתא דתנאי אי עיקר שירה בפה או בכלי דלמ"ד בכלי הוי כלי שרת כדאמרינן בסוכה דף נ' וכלי שרת שנפגם אין מתקנין אותו משום דאין עניות במקום עשירות. הנה י"ל לפ"ד רש"י בזבחים פ"ח ע"א ד"ה נפגמו אין מתקנין אותן פירוש פגימה גדולה שתיקון השחזה ניכר יעו"ש מבואר דהיכי דהקלקול מועט ואין הפגם ניכר כ"כ ל"ש לומר אין עניות במקום עשירות וא"כ גבי כלי שיר דאין הפגם ניכר ע"פ רוב דגם בקלקול מועט נשתנה הקול מותר לתקנן. ועי' בתוס' חולין דף ג' ע"א ד"ה כגון שכתבו דסכין של שחיטה אינו כלי שרת יעו"ש ובזבחים שם /דף פ"ח/ מבואר דסכין שנפגמה אין מתקנין אותה ומוכרח לומר דבכל כלי שמשתמשין בו להקדש לעבודה אף שאינו כלי שרת כשנפגם אין מתקנין אותן דלא כתירוץ השני שבתוס' שבועות דף י"א דדוקא בכ"ש =בכלי שרת= שייך לומר א"ע =אין עניות= במקום עשירות ולא במכתשת שהיא רק קד"ד =קדושת דמים= יעו"ש, וכן מוכרח לומר לשיטת ריטב"א קדושין דף נ"ד הובא במל"מ פ"ח מה' כלי המקדש דבג"כ הוא רק קד"ד דאינם כל"ש יעו"ש, ובזבחים אמרינן בבג"כ דא"ע במק"ע =בבגדי כהונה דאין עניות במקום עשירות= וע"כ דלאו דוקא בכ"ש אמרינן כן. וא"כ גבי כלי שיר אף למ"ד עיקר שירה בפה וכלי הוי רק לבסומי קלא ול"ה כ"ש מ"מ כיון שמשתמשין בו במקדש לעבודה כשנפגם אין מתקנין אותו ודו"ק. ועפי"ד ריטב"א הנ"ל יש לומר דבבג"כ ל"ש לומר דבר הלמד בהיקש אינו חוזר ומלמד בהיקש דהא לא אמרו כן רק בקדה"ג =בקדושת הגוף= ולא בקד"ד כמ"ש התוספות ביומא ס"ח ד"ה תן ויש ליישב בזה קושיית תוספות יומא כ"ג ד"ה ילבש ועיין תוספות יומא דף ל"ב ד"ה ומה יעו"ש. ודו"ק כי קצרתי. משלם ראטה
